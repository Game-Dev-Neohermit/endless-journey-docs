Task 3: WeaponValidation — spread, range, ammo transitions​

Files: Modify src/ReplicatedStorage/SharedModules/WeaponValidation.luau

  • Step 1: Before the final return WeaponValidation, add:
-- Apply a random spread cone to an aim direction. `spreadDegrees` is the max
-- half-angle. Returns a unit direction. Used client-side for visuals/feel; the
-- server raycasts with the client-supplied direction (range-capped) and does its
-- own line-of-sight, so spread is cosmetic, not a trust boundary.
function WeaponValidation.applySpread(direction: Vector3, spreadDegrees: number): Vector3
	if spreadDegrees <= 0 then
		return direction.Unit
	end
	local unit = direction.Unit
	-- Build an orthonormal basis around `unit`.
	local up = math.abs(unit.Y) < 0.99 and Vector3.yAxis or Vector3.xAxis
	local right = unit:Cross(up).Unit
	local realUp = right:Cross(unit).Unit
	local angle = math.rad(spreadDegrees) * math.sqrt(math.random())
	local theta = math.random() * math.pi * 2
	local offset = (right * math.cos(theta) + realUp * math.sin(theta)) * math.tan(angle)
	return (unit + offset).Unit
end

-- True if `hitPos` is within `maxDistance` studs of `origin`.
function WeaponValidation.isWithinRange(origin: Vector3, hitPos: Vector3, maxDistance: number): boolean
	return (hitPos - origin).Magnitude <= maxDistance
end

-- Pure ammo transition: consume one round. Returns the new magazine count.
function WeaponValidation.ammoAfterShot(mag: number): number
	return math.max(0, mag - 1)
end

-- Pure ammo transition: reload. Moves rounds from reserve into the magazine up to
-- `magazineSize`. Returns (newMag, newReserve).
function WeaponValidation.ammoAfterReload(mag: number, reserve: number, magazineSize: number): (number, number)
	local needed = math.max(0, magazineSize - mag)
	local moved = math.min(needed, reserve)
	return mag + moved, reserve - moved
end

Task 6: GunHandler (client)​

Files: Create src/StarterPlayer/StarterPlayerScripts/WeaponHandlers/GunHandler.luau

Predicts firing/ammo, draws the shooter's own tracer + muzzle flash, handles the automatic fire loop and reload. Local ammo is a module table reconciled by WeaponAmmoSync.

  • Step 1: Write the module:
--!strict

--[=[
    @class GunHandler
    @client
    Category handler for hitscan guns. Predicts fire-rate and ammo for snappy
    feel, draws the shooter's tracer/muzzle FX, and sends `ReqFireWeapon` for the
    server to apply authoritative damage. Supports semi and automatic fire (held
    via OnActivate/OnDeactivate) and reload (OnReload). Local ammo is reconciled
    by the server's `WeaponAmmoSync`.
]=]

local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local RunService = game:GetService("RunService")
local UserInputService = game:GetService("UserInputService")

local SharedModules = ReplicatedStorage.SharedModules
local WeaponValidation = require(SharedModules:WaitForChild("WeaponValidation"))
local WeaponTypes = require(SharedModules:WaitForChild("WeaponTypes"))
local Packets = require(SharedModules:WaitForChild("Packets"))
local Signals = require(SharedModules:WaitForChild("Signals"))
local SoundManager = require(SharedModules:WaitForChild("SoundManager"))
local MotionMaster = require(SharedModules:WaitForChild("MotionMaster"))

local Janitor = require(ReplicatedStorage.Packages:WaitForChild("Janitor"))

-- Reuse the Plan 1 handler context shape.
local MeleeHandler = require(script.Parent:WaitForChild("MeleeHandler"))
export type HandlerContext = MeleeHandler.HandlerContext

local player = Players.LocalPlayer

-- Module-level predicted ammo, keyed by weapon name. Reconciled by WeaponAmmoSync.
local localAmmo: { [string]: { mag: number, reserve: number } } = {}
local lastShot: { [string]: number } = {}
local firingTokens: { [string]: number } = {} -- cancels the auto-fire loop on release

-- One-time connection: server ammo authority overrides local prediction.
do
	Packets.WeaponAmmoSync.OnClientEvent:Connect(function(weaponName: string, mag: number, reserve: number)
		localAmmo[weaponName] = { mag = mag, reserve = reserve }
		Signals.WeaponAmmoChanged:Fire(weaponName, mag, reserve)
	end)
end

local GunHandler = {}

local function getMuzzle(tool: Tool): (Vector3, BasePart?)
	local handle = tool:FindFirstChild("Handle") :: BasePart?
	if not handle then
		local character = player.Character
		local root = character and character:FindFirstChild("HumanoidRootPart") :: BasePart?
		return (root and root.Position) or Vector3.zero, nil
	end
	local muzzle = handle:FindFirstChild("Muzzle") :: Attachment?
	if muzzle then
		return muzzle.WorldPosition, handle
	end
	return handle.Position, handle
end

-- Aim point under the crosshair (screen center), via a camera raycast.
local function getAimPoint(ignore: { Instance }, maxDistance: number): Vector3
	local camera = workspace.CurrentCamera
	local viewport = camera.ViewportSize
	local ray = camera:ViewportPointToRay(viewport.X / 2, viewport.Y / 2)
	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	params.FilterDescendantsInstances = ignore
	local result = workspace:Raycast(ray.Origin, ray.Direction * maxDistance, params)
	return result and result.Position or (ray.Origin + ray.Direction * maxDistance)
end

local function spawnTracer(from: Vector3, to: Vector3)
	local distance = (to - from).Magnitude
	if distance < 1 then
		return
	end
	local tracer = Instance.new("Part")
	tracer.Anchored = true
	tracer.CanCollide = false
	tracer.CanQuery = false
	tracer.CanTouch = false
	tracer.Material = Enum.Material.Neon
	tracer.Color = Color3.fromRGB(255, 226, 150)
	tracer.Size = Vector3.new(0.1, 0.1, distance)
	tracer.CFrame = CFrame.lookAt(from, to) * CFrame.new(0, 0, -distance / 2)
	tracer.Parent = workspace
	MotionMaster.Play(tracer, { Transparency = 1 }, { Duration = 0.08, Model = "EaseOutQuad" })
	task.delay(0.12, function()
		tracer:Destroy()
	end)
end

local function fireOnce(context: HandlerContext)
	local config = context.config :: WeaponTypes.GunWeaponConfig
	local now = os.clock()
	local interval = 60 / config.RPM

	if not WeaponValidation.canFire(lastShot[config.ToolName] or -math.huge, interval, now) then
		return
	end

	local ammo = localAmmo[config.ToolName]
	if not ammo or ammo.mag <= 0 then
		SoundManager.PlaySFX(config.DrySound)
		return
	end

	lastShot[config.ToolName] = now
	ammo.mag = WeaponValidation.ammoAfterShot(ammo.mag)
	Signals.WeaponAmmoChanged:Fire(config.ToolName, ammo.mag, ammo.reserve)

	local character = context.character
	local origin, handle = getMuzzle(context.tool)
	local aimPoint = getAimPoint({ character }, config.MaxDistance)
	local direction = WeaponValidation.applySpread((aimPoint - origin), config.Spread)

	-- Local visuals + fire sound (replicated so others hear it).
	if handle then
		SoundManager.PlaySFXAt(config.FireSound, handle.Position, { Replicate = true })
	end

	-- Predict the hit for tracer endpoint + hitmarker.
	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	params.FilterDescendantsInstances = { character }
	local result = workspace:Raycast(origin, direction * config.MaxDistance, params)
	local endpoint = result and result.Position or (origin + direction * config.MaxDistance)
	spawnTracer(origin, endpoint)
	if result then
		local model = WeaponValidation.resolveHumanoidModel(result.Instance)
		if model then
			Signals.WeaponLocalHit:Fire(model)
		end
	end

	Packets.ReqFireWeapon:Fire(origin, direction)
end

function GunHandler.OnEquip(_context: HandlerContext)
	-- Initial ammo arrives via WeaponAmmoSync from the server on equip.
end

function GunHandler.OnActivate(context: HandlerContext)
	local config = context.config :: WeaponTypes.GunWeaponConfig
	if not config.Automatic then
		fireOnce(context)
		return
	end
	-- Automatic: loop until OnDeactivate bumps the token.
	firingTokens[config.ToolName] = (firingTokens[config.ToolName] or 0) + 1
	local token = firingTokens[config.ToolName]
	task.spawn(function()
		while firingTokens[config.ToolName] == token do
			fireOnce(context)
			RunService.Heartbeat:Wait()
		end
	end)
end

function GunHandler.OnDeactivate(context: HandlerContext)
	local config = context.config :: WeaponTypes.GunWeaponConfig
	firingTokens[config.ToolName] = (firingTokens[config.ToolName] or 0) + 1 -- cancel loop
end

function GunHandler.OnReload(context: HandlerContext)
	local config = context.config :: WeaponTypes.GunWeaponConfig
	local ammo = localAmmo[config.ToolName]
	if not ammo or ammo.mag >= config.MagazineSize or ammo.reserve <= 0 then
		return
	end

	Signals.WeaponReloadStarted:Fire(config.ToolName, config.ReloadTime)
	SoundManager.PlaySFX(config.ReloadSound)

	Packets.ReqReloadWeapon:Fire():andThen(function(accepted: boolean, reloadTime: number)
		if not accepted then
			Signals.WeaponReloadComplete:Fire(config.ToolName)
			return
		end
		task.delay(reloadTime, function()
			-- Server sends authoritative WeaponAmmoSync at completion; predict here too.
			local current = localAmmo[config.ToolName]
			if current then
				local newMag, newReserve = WeaponValidation.ammoAfterReload(current.mag, current.reserve, config.MagazineSize)
				current.mag, current.reserve = newMag, newReserve
				Signals.WeaponAmmoChanged:Fire(config.ToolName, newMag, newReserve)
			end
			Signals.WeaponReloadComplete:Fire(config.ToolName)
		end)
	end)
end

function GunHandler.OnUnequip(context: HandlerContext)
	local config = context.config :: WeaponTypes.GunWeaponConfig
	firingTokens[config.ToolName] = (firingTokens[config.ToolName] or 0) + 1 -- stop any loop
end

return GunHandler

Packets.ReqReloadWeapon:Fire():andThen(...) follows the project's :Response(...) packet pattern (the request returns a promise-like with andThen). If the local Packet library returns the values directly instead, replace with local accepted, reloadTime = Packets.ReqReloadWeapon:Fire(). Confirm against Packets/Packet usage in ReqShredderStart when implementing.

local function onFireWeapon(player: Player, origin: Vector3, direction: Vector3)
	local _tool, config = getEquippedWeapon(player)
	if not config or config.Category ~= "Gun" then
		return
	end
	local gun = config :: WeaponTypes.GunWeaponConfig
	if gun.FireMode ~= "Hitscan" then
		return -- projectile guns (rocket) handled in Plan 3
	end

	local now = os.clock()
	local interval = 60 / gun.RPM
	if not WeaponValidation.canFire(WeaponState.GetLastFire(player, gun.ToolName), interval, now) then
		return
	end
	if WeaponState.IsReloading(player, gun.ToolName, now) then
		return
	end

	local mag, reserve = WeaponState.GetAmmo(player, gun.ToolName)
	if mag <= 0 then
		return
	end

	local character = player.Character
	local root = character and character:FindFirstChild("HumanoidRootPart") :: BasePart?
	if not character or not root then
		return
	end
	-- Anti-exploit: claimed muzzle origin must be near the character.
	if (origin - root.Position).Magnitude > MAX_ORIGIN_DISTANCE then
		return
	end

	WeaponState.SetLastFire(player, gun.ToolName, now)
	WeaponState.SetAmmo(player, gun.ToolName, WeaponValidation.ammoAfterShot(mag), reserve)

	-- Authoritative raycast (server's hit wins; range-capped; ignores the shooter).
	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	params.FilterDescendantsInstances = { character }
	local unit = direction.Unit
	local result = workspace:Raycast(origin, unit * gun.MaxDistance, params)

	local endpoint = origin + unit * gun.MaxDistance
	local isHit = false
	if result then
		endpoint = result.Position
		local model, humanoid = WeaponValidation.resolveHumanoidModel(result.Instance)
		if model and humanoid then
			local amount = gun.Damage
			if result.Instance.Name == "Head" then
				amount *= (gun.HeadshotMultiplier or 1)
			end
			isHit = CombatService.ApplyToModel(model, humanoid, amount, gun.DamageType, player)
		end
	end

	Packets.WeaponFired:Fire(player, gun.ToolName, origin, endpoint, isHit)

	local newMag, newReserve = WeaponState.GetAmmo(player, gun.ToolName)
	Packets.WeaponAmmoSync:FireClient(player, gun.ToolName, newMag, newReserve)
end

local function onReloadWeapon(player: Player): (boolean, number)
	local _tool, config = getEquippedWeapon(player)
	if not config or config.Category ~= "Gun" then
		return false, 0
	end
	local gun = config :: WeaponTypes.GunWeaponConfig

	local now = os.clock()
	if WeaponState.IsReloading(player, gun.ToolName, now) then
		return false, 0
	end
	local mag, reserve = WeaponState.GetAmmo(player, gun.ToolName)
	if mag >= gun.MagazineSize or reserve <= 0 then
		return false, 0
	end

	WeaponState.SetReloadEnd(player, gun.ToolName, now + gun.ReloadTime)

	task.delay(gun.ReloadTime, function()
		local curMag, curReserve = WeaponState.GetAmmo(player, gun.ToolName)
		local newMag, newReserve = WeaponValidation.ammoAfterReload(curMag, curReserve, gun.MagazineSize)
		WeaponState.SetAmmo(player, gun.ToolName, newMag, newReserve)
		WeaponState.SetReloadEnd(player, gun.ToolName, 0)
		Packets.WeaponAmmoSync:FireClient(player, gun.ToolName, newMag, newReserve)
	end)

	return true, gun.ReloadTime
end
