Perk System Framework Implementation Plan

For agentic workers: REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (- [ ]) syntax for tracking.

Goal: Build the level-driven perk system framework — point allocation, persistence, party-max resolution, replicated buff folders, and a perk GUI — and wire the four perks whose mechanic already exists (Player HP, Passive Regen, Grabhook Reel, Junk Rarity).

Architecture: Mirrors the existing Tech Tree. PerkConfig declares perks + per-level values; PerkManager (ProfileStore-backed, like XPManager) owns allocation and derives points from XP level; PerkEffectsDispatcher publishes resolved buffs into replicated folders (PerkBuffs/<UserId> personal, PartyPerkBuffs party-max); consumer systems read one attribute. PerkController drives a code-built screen GUI, predicting then reconciling via packets.

Tech Stack: Roblox / Luau, Rojo sync, ProfileStore (session-locked DataStore), the project's Packet (typed nets) and Signals (in-process bus) modules, Janitor for cleanup.

Testing note: This project has no automated test suite — verification is done in Roblox Studio (per CLAUDE.md). Each task ends with a concrete Studio observation and a commit. Do not run rojo build or selene as verification (project preference); the user syncs live with rojo serve. For pure-logic modules, a Studio command-bar snippet is given as an optional sanity check.

Reference spec: docs/superpowers/specs/2026-06-12-perk-system-framework-design.md

Task 3: PerkState (pure shared logic)​

Files:

  • Create: src/ReplicatedStorage/SharedModules/PerkState.luau

  • Step 1: Create the pure module

--!strict

--[=[
	@class PerkState
	Pure, side-effect-free perk math shared by the client (button enable) and the
	server (authorization). Points are DERIVED, never stored: each level-up grants
	1 Personal point AND 1 Party point, so a pool's total points equal the player's
	level; spent = sum of that pool's perk levels; available = total - spent.
]=]

local ReplicatedStorage = game:GetService("ReplicatedStorage")
local PerkConfig = require(ReplicatedStorage.SharedModules.Configs.PerkConfig)
local PerkResultCodes = require(ReplicatedStorage.SharedModules.PerkResultCodes)

export type PerkLevels = { [string]: number }

local PerkState = {}

--[=[
	Sum of all spent points in one category for the given perk levels.
]=]
function PerkState.spent(perkLevels: PerkLevels, category: PerkConfig.PerkCategory): number
	local total = 0
	for perkId, level in perkLevels do
		local perk = PerkConfig.Perks[perkId]
		if perk and perk.Category == category then
			total += level
		end
	end
	return total
end

--[=[
	Points still available in one pool: playerLevel - spent (floored at 0).
]=]
function PerkState.available(perkLevels: PerkLevels, category: PerkConfig.PerkCategory, playerLevel: number): number
	return math.max(0, playerLevel - PerkState.spent(perkLevels, category))
end

--[=[
	Whether one point can be spent on `perkId`. Returns (ok, reasonCode).
]=]
function PerkState.canAllocate(perkLevels: PerkLevels, perkId: string, playerLevel: number): (boolean, number)
	local perk = PerkConfig.Perks[perkId]
	if not perk then
		return false, PerkResultCodes.UNKNOWN_PERK
	end
	local current = perkLevels[perkId] or 0
	if current >= perk.MaxLevel then
		return false, PerkResultCodes.AT_MAX_LEVEL
	end
	if PerkState.available(perkLevels, perk.Category, playerLevel) < 1 then
		return false, PerkResultCodes.NO_POINTS
	end
	return true, PerkResultCodes.OK
end

return PerkState

Task 7: PerkManager (server authority)​

Files:

  • Create: src/ServerStorage/SharedModules/PerkManager.luau

  • Step 1: Create the manager (ProfileStore lifecycle modeled on XPManager; packet handlers + derivation in one cohesive authority):

--!strict

--[=[
	@class PerkManager
	@server
	Authority for player perk allocation. Owns the `PlayerPerks` ProfileStore
	(one session per player, key = UserId), validates allocate/respec with the
	pure `PerkState` (reading the player's level from `XPManager`), and replicates
	the owning player's snapshot via `PerkStateSync`. Points are derived, never
	stored — only `PerkLevels` persists. Fires `Signals.OnPerkStateChanged` after
	every mutation so `PerkEffectsDispatcher` republishes the buff folders.
]=]

local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local ServerStorage = game:GetService("ServerStorage")

local SharedModules = ReplicatedStorage.SharedModules
local PerkConfig = require(SharedModules.Configs:WaitForChild("PerkConfig"))
local PerkState = require(SharedModules:WaitForChild("PerkState"))
local PerkResultCodes = require(SharedModules:WaitForChild("PerkResultCodes"))
local Packets = require(SharedModules:WaitForChild("Packets"))
local Signals = require(SharedModules:WaitForChild("Signals"))
local ProfileStore = require(ServerStorage.SharedModules:WaitForChild("ProfileStore"))
local XPManager = require(ServerStorage.SharedModules:WaitForChild("XPManager"))

local PROFILE_TEMPLATE = {
	PerkLevels = {} :: { [string]: number },
}

local PlayerStore = ProfileStore.New(PerkConfig.STORE_NAME, PROFILE_TEMPLATE)
local Profiles: { [Player]: any } = {}

local PerkManager = {}
local initialized = false

-- Fire the current snapshot to one player.
local function replicate(player: Player)
	local profile = Profiles[player]
	if not profile then
		return
	end
	local levels = profile.Data.PerkLevels
	local lvl = XPManager.GetLevel(player)
	local personal = PerkState.available(levels, "Personal", lvl)
	local party = PerkState.available(levels, "Party", lvl)
	Packets.PerkStateSync:FireClient(player, levels, personal, party)
end

--[=[
	@within PerkManager
	Spend one point on `perkId`. Returns (success, reasonCode). No-op (with the
	matching reason) if the profile is missing, the perk is unknown/at max, or the
	pool has no available points.
]=]
function PerkManager.AllocatePoint(player: Player, perkId: string): (boolean, number)
	local profile = Profiles[player]
	if not profile then
		return false, PerkResultCodes.NO_PROFILE
	end
	local levels = profile.Data.PerkLevels
	local ok, reason = PerkState.canAllocate(levels, perkId, XPManager.GetLevel(player))
	if not ok then
		return false, reason
	end
	levels[perkId] = (levels[perkId] or 0) + 1
	Signals.OnPerkStateChanged:Fire(player)
	replicate(player)
	return true, PerkResultCodes.OK
end

--[=[
	@within PerkManager
	Respec a pool. `pool`: 0 = both, 1 = Personal, 2 = Party. Clears the matching
	perk levels (points are derived, so they return to the wallet automatically).
]=]
function PerkManager.Respec(player: Player, pool: number)
	local profile = Profiles[player]
	if not profile then
		return
	end
	local levels = profile.Data.PerkLevels
	for perkId in levels do
		local perk = PerkConfig.Perks[perkId]
		if perk then
			local match = pool == 0
				or (pool == 1 and perk.Category == "Personal")
				or (pool == 2 and perk.Category == "Party")
			if match then
				levels[perkId] = nil
			end
		end
	end
	Signals.OnPerkStateChanged:Fire(player)
	replicate(player)
end

--[=[
	@within PerkManager
	The player's persisted perk levels (live reference; read-only by convention).
	Returns an empty table if no profile is loaded.
]=]
function PerkManager.GetPerkLevels(player: Player): { [string]: number }
	local profile = Profiles[player]
	return (profile and profile.Data.PerkLevels) or {}
end

local function onPlayerAdded(player: Player)
	local profile = PlayerStore:StartSessionAsync(tostring(player.UserId), {
		Cancel = function()
			return player.Parent ~= Players
		end,
	})

	if profile == nil then
		player:Kick("Perk profile failed to load — please rejoin.")
		return
	end

	profile:AddUserId(player.UserId)
	profile:Reconcile()

	profile.OnSessionEnd:Connect(function()
		Profiles[player] = nil
		player:Kick("Perk profile session ended — please rejoin.")
	end)

	if player.Parent == Players then
		Profiles[player] = profile
		replicate(player)
	else
		profile:EndSession()
	end
end

--[=[
	@within PerkManager
	Wire profile lifecycle, packet handlers, and the level-up resend. Call once.
]=]
function PerkManager.Initialize()
	if initialized then
		return
	end
	initialized = true

	for _, player in Players:GetPlayers() do
		task.spawn(onPlayerAdded, player)
	end
	Players.PlayerAdded:Connect(onPlayerAdded)
	Players.PlayerRemoving:Connect(function(player)
		local profile = Profiles[player]
		if profile then
			profile:EndSession()
		end
	end)

	-- Allocate / respec requests (authoritative re-validation inside the APIs).
	Packets.ReqAllocatePerk.OnServerInvoke = function(player: Player, perkId: string)
		if type(perkId) ~= "string" then
			return false, PerkResultCodes.UNKNOWN_PERK
		end
		return PerkManager.AllocatePoint(player, perkId)
	end
	Packets.ReqRespecPerks.OnServerInvoke = function(player: Player, pool: number)
		if type(pool) ~= "number" or pool < 0 or pool > 2 then
			return false
		end
		PerkManager.Respec(player, pool)
		return true
	end

	-- A level-up grows available points — resend the snapshot.
	Signals.OnPlayerLevelUp:Connect(function(player: Player)
		replicate(player)
	end)
end

return PerkManager

Response-handler convention (verified): a :Response(...) packet is handled server-side with Packet.OnServerInvoke = function(player, ...) return ... end and called client-side with local a, b = Packet:Fire(...) (it yields and returns the response tuple). This matches ReqShredderStart.OnServerInvoke in ShredderServer and ReqShredderStart:Fire(bench) in ShredderClient. The handlers above already use this convention.

Task 9: PerkEffectsDispatcher (buff folders + Junk Rarity bridge)​

Files:

  • Create: src/ServerScriptService/PerkEffectsDispatcher.server.luau

  • Step 1: Create the dispatcher (sole writer of both buff folders; recomputes on allocate/respec/join/leave):

--!strict

--[=[
	@class PerkEffectsDispatcher
	@server
	The single writer of the perk buff folders. Personal buffs go to
	`ReplicatedStorage/PerkBuffs/<UserId>` (per-player attributes); party buffs go
	to `ReplicatedStorage/PartyPerkBuffs` as the per-perk MAX level across all
	players in the server (the "only highest active" rule). Recomputes on
	allocate/respec (`Signals.OnPerkStateChanged`), player join, and player leave.
	Bridges the Junk Rarity party perk onto the existing `ApplyTierMultiplier`
	signal (which `JunkService` already listens for).
]=]

local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local ServerStorage = game:GetService("ServerStorage")

local SharedModules = ReplicatedStorage.SharedModules
local PerkConfig = require(SharedModules.Configs:WaitForChild("PerkConfig"))
local Signals = require(SharedModules:WaitForChild("Signals"))
local PerkManager = require(ServerStorage.SharedModules:WaitForChild("PerkManager"))

-- Rare junk tiers the Salvage Luck perk boosts (tiers 3-5 in JunkConfig).
local RARE_JUNK_TIERS = { 3, 4, 5 }

local function ensureFolder(name: string): Folder
	local f = ReplicatedStorage:FindFirstChild(name)
	if not f then
		f = Instance.new("Folder")
		f.Name = name
		f.Parent = ReplicatedStorage
	end
	return f :: Folder
end

local perkBuffs = ensureFolder("PerkBuffs")
local partyPerkBuffs = ensureFolder("PartyPerkBuffs")

-- Per-player personal buffs.
local function publishPersonal(player: Player)
	local holder = perkBuffs:FindFirstChild(tostring(player.UserId)) :: Folder?
	if not holder then
		holder = Instance.new("Folder")
		holder.Name = tostring(player.UserId)
		holder.Parent = perkBuffs
	end
	local levels = PerkManager.GetPerkLevels(player)
	for perkId, perk in PerkConfig.Perks do
		if perk.Category == "Personal" then
			holder:SetAttribute(perk.BuffKey, perk.PerLevel * (levels[perkId] or 0))
		end
	end
end

-- Party buffs = per-perk MAX level across all players.
local function publishParty()
	-- Compute max level per party perk.
	local maxLevel: { [string]: number } = {}
	for perkId, perk in PerkConfig.Perks do
		if perk.Category == "Party" then
			maxLevel[perkId] = 0
		end
	end
	for _, player in Players:GetPlayers() do
		local levels = PerkManager.GetPerkLevels(player)
		for perkId in maxLevel do
			local l = levels[perkId] or 0
			if l > maxLevel[perkId] then
				maxLevel[perkId] = l
			end
		end
	end
	-- Write resolved party buffs.
	for perkId, perk in PerkConfig.Perks do
		if perk.Category == "Party" then
			partyPerkBuffs:SetAttribute(perk.BuffKey, perk.PerLevel * maxLevel[perkId])
		end
	end
	-- Junk Rarity bridge: drive the rare-tier weight multipliers JunkService listens for.
	local rarityBonus = partyPerkBuffs:GetAttribute("JunkTierRarityBonus") :: number? or 0
	for _, tier in RARE_JUNK_TIERS do
		Signals.ApplyTierMultiplier:Fire(tier, 1 + rarityBonus)
	end
end

Signals.OnPerkStateChanged:Connect(function(player: Player)
	publishPersonal(player)
	publishParty()
end)

Players.PlayerAdded:Connect(function(player: Player)
	-- Wait until the manager has a profile, then publish this player's personal

				local activeLvl = math.floor(activeVal / perk.PerLevel + 0.5)

			else
				row.Level.Text = ("Lv %d/%d"):format(lvl, perk.MaxLevel)
			end
			local avail = perk.Category == "Personal" and personalAvailable or partyAvailable
			row.Plus.Visible = lvl < perk.MaxLevel and avail > 0
		end
	end
end
