Workbench Crafting Implementation Plan

For agentic workers: REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (- [ ]) syntax for tracking.

Goal: Wire the Studio-built WorkbenchGui to a new server-authoritative crafting system that turns Wood/Metal into weapons, armor, and placeables from config-defined recipes.

Architecture: Mirrors the Tech Tree's client-predicts / server-authoritative shape — a pure shared validator + frozen result codes + a typed ReqCraft packet + a server WorkbenchService that consumes materials and dispatches output through CraftOutputDispatcher. Recipes may be gated behind unlocked Tech Tree nodes, read from a new ReplicatedStorage/TechTreeUnlocked folder published by TechTreeService. Stash/Move reuse the existing PlaceableActions system unchanged.

Tech Stack: Roblox / Luau / Rojo. Networking via the project Packets registry; cleanup via Janitor; client audio via SoundManager; toasts via Signals.RequestNotification.

Verification note: This project has no automated test suite — verification is in Roblox Studio (per CLAUDE.md). Each task ends with a concrete Studio check, then a commit. Do not run rojo build or selene as verification (project preference); the user syncs live with rojo serve.

No project.json changes: every new file lands in a folder already mapped by default.project.json (ReplicatedStorage/SharedModules, ServerScriptService, ServerStorage/SharedModules, StarterPlayer/StarterPlayerScripts).

Task 3: WorkbenchValidation (pure shared validator)​

Files:

  • Create: src/ReplicatedStorage/SharedModules/WorkbenchValidation.luau

  • Step 1: Create the validator module

--!strict

--[=[
	@class WorkbenchValidation
	Pure, stateless validator shared by the Workbench client and server — no
	Roblox service calls. The single rules source: the client uses it to grey
	unaffordable/locked recipes; the server uses it to authorise crafts. It does
	NOT check proximity, output feasibility, or that the player physically holds
	Tools — those are server-only concerns layered on top in WorkbenchService.
]=]

local WorkbenchConfig = require(script.Parent.Configs:WaitForChild("WorkbenchConfig"))
local Codes = require(script.Parent:WaitForChild("WorkbenchResultCodes"))

local WorkbenchValidation = {}

--[=[ Returns the recipe for `recipeId`, or nil. ]=]
function WorkbenchValidation.getRecipe(recipeId: string): WorkbenchConfig.Recipe?
	if type(recipeId) ~= "string" then
		return nil
	end
	return WorkbenchConfig.Recipes[recipeId]
end

--[=[ Normalised cost map for a recipe, omitting nil/zero entries. ]=]
function WorkbenchValidation.costOf(recipe: WorkbenchConfig.Recipe): { [string]: number }
	local out: { [string]: number } = {}
	for matType, amount in recipe.Cost :: { [string]: number } do
		if amount and amount > 0 then
			out[matType] = amount
		end
	end
	return out
end

--[=[ True if `heldCounts[mat] >= cost[mat]` for every cost material. ]=]
function WorkbenchValidation.canAfford(recipe: WorkbenchConfig.Recipe, heldCounts: { [string]: number }): boolean
	for matType, need in WorkbenchValidation.costOf(recipe) do
		if (heldCounts[matType] or 0) < need then
			return false
		end
	end
	return true
end

--[=[ True if the recipe is ungated or its required node is unlocked.
	`isNodeUnlocked(nodeId) -> boolean` is supplied by the caller. ]=]
function WorkbenchValidation.isUnlocked(
	recipe: WorkbenchConfig.Recipe,
	isNodeUnlocked: (nodeId: string) -> boolean
): boolean
	local node = recipe.RequiresNode
	if node == nil then
		return true
	end
	return isNodeUnlocked(node)
end

--[=[ Full validation: recipe membership, gate, affordability.
	Returns (ok, code). ]=]
function WorkbenchValidation.validateCraft(
	recipe: WorkbenchConfig.Recipe?,
	heldCounts: { [string]: number },
	isNodeUnlocked: (nodeId: string) -> boolean
): (boolean, number)
	if recipe == nil then
		return false, Codes.UNKNOWN_RECIPE
	end
	if not WorkbenchValidation.isUnlocked(recipe, isNodeUnlocked) then
		return false, Codes.LOCKED
	end
	if not WorkbenchValidation.canAfford(recipe, heldCounts) then
		return false, Codes.INSUFFICIENT_MATERIALS
	end
	return true, Codes.NONE
end

return WorkbenchValidation

Task 6: CraftOutputDispatcher​

Files:

  • Create: src/ServerStorage/SharedModules/CraftOutputDispatcher.luau

  • Step 1: Create the dispatcher module

--!strict

--[=[
	@class CraftOutputDispatcher
	@server
	Server-side grant logic for a recipe's `Output`, dispatched on `Output.Kind`
	so each output type is isolated (mirrors TechEffectsDispatcher). `isFeasible`
	is checked by WorkbenchService BEFORE consuming materials, so a misconfigured
	recipe never costs the player materials; `grant` performs the actual award.
		- Weapon    → clone a Tool from ServerStorage/<WEAPON_TEMPLATES_FOLDER> to Backpack
		- Armor     → ArmorManager.SetTier(player, Tier)
		- Placeable → clone ReplicatedStorage/PlaceableTools/<Name> to Backpack
]=]

local ReplicatedStorage = game:GetService("ReplicatedStorage")
local ServerStorage = game:GetService("ServerStorage")

local SharedModules = ReplicatedStorage.SharedModules
local WorkbenchConfig = require(SharedModules.Configs:WaitForChild("WorkbenchConfig"))
local ArmorConfig = require(SharedModules.Configs:WaitForChild("ArmorConfig"))
local ArmorManager = require(ServerStorage.SharedModules:WaitForChild("ArmorManager"))

type Output = WorkbenchConfig.Output

local CraftOutputDispatcher = {}

-- Locate a weapon Tool template in the configured ServerStorage folder.
local function weaponTemplate(toolName: string): Tool?
	local folder = ServerStorage:FindFirstChild(WorkbenchConfig.WEAPON_TEMPLATES_FOLDER)
	local inst = folder and folder:FindFirstChild(toolName)
	if inst and inst:IsA("Tool") then
		return inst :: Tool
	end
	return nil
end

-- Locate a placeable Tool template (same templates the stash/move system grants).
local function placeableTemplate(placeableName: string): Tool?
	local folder = ReplicatedStorage:FindFirstChild("PlaceableTools")
	local inst = folder and folder:FindFirstChild(placeableName)
	if inst and inst:IsA("Tool") then
		return inst :: Tool
	end
	return nil
end

--[=[ True if this output can be granted right now (template/tier exists).
	Checked before materials are consumed. ]=]
function CraftOutputDispatcher.isFeasible(output: Output): boolean
	if output.Kind == "Weapon" then
		return weaponTemplate(output.ToolName) ~= nil
	elseif output.Kind == "Placeable" then
		return placeableTemplate(output.PlaceableName) ~= nil
	elseif output.Kind == "Armor" then
		return ArmorConfig.Tiers[output.Tier] ~= nil
	end
	return false
end

--[=[ Grant the output to the player. Returns true on success. Assumes
	`isFeasible` already passed; a false here is a defensive last resort. ]=]
function CraftOutputDispatcher.grant(player: Player, output: Output): boolean
	if output.Kind == "Weapon" then
		local template = weaponTemplate(output.ToolName)
		if not template then
			return false
		end
		local backpack = player:FindFirstChildOfClass("Backpack")
		if not backpack then
			return false
		end
		template:Clone().Parent = backpack
		return true
	elseif output.Kind == "Placeable" then
		local template = placeableTemplate(output.PlaceableName)
		if not template then
			return false
		end
		local backpack = player:FindFirstChildOfClass("Backpack")
		if not backpack then
			return false
		end
		template:Clone().Parent = backpack
		return true
	elseif output.Kind == "Armor" then
		return ArmorManager.SetTier(player, output.Tier)
	end
	warn(("[CraftOutputDispatcher] Unknown output kind for grant"))
	return false
end

return CraftOutputDispatcher

  • Create: src/ServerScriptService/WorkbenchService.server.luau

  • Step 1: Create the service script

--!strict

--[=[
	@class WorkbenchService
	@server
	Authoritative handler for `ReqCraft`. Validates (recipe membership, target is a
	tagged Workbench station, proximity, per-player cooldown, Tech Tree gate, output
	feasibility, then materials), consumes Wood/Metal Tools from the crafter's
	backpack, and dispatches the recipe output via CraftOutputDispatcher. Any
	failure returns a reason code and consumes nothing. Client input is never
	trusted: proximity and gate are re-checked from server-owned state.
]=]

local CollectionService = game:GetService("CollectionService")
local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local ServerStorage = game:GetService("ServerStorage")

local SharedModules = ReplicatedStorage.SharedModules
local WorkbenchConfig = require(SharedModules.Configs:WaitForChild("WorkbenchConfig"))
local WorkbenchValidation = require(SharedModules:WaitForChild("WorkbenchValidation"))
local Codes = require(SharedModules:WaitForChild("WorkbenchResultCodes"))
local Packets = require(SharedModules:WaitForChild("Packets"))
local CraftOutputDispatcher = require(ServerStorage.SharedModules:WaitForChild("CraftOutputDispatcher"))

local ReqCraft = Packets.ReqCraft

-- ── Tech Tree gate read (server-written folder; unforgeable by clients) ──
local function isNodeUnlocked(nodeId: string): boolean
	local folder = ReplicatedStorage:FindFirstChild("TechTreeUnlocked")
	return folder ~= nil and folder:FindFirstChild(nodeId) ~= nil
end

-- ── Material Tool helpers (same contract as TechTreeService) ──
local function countMaterial(player: Player, matType: string): number
	local count = 0
	local bp = player:FindFirstChildOfClass("Backpack")
	if bp then
		for _, item in bp:GetChildren() do
			if item:IsA("Tool") and item:GetAttribute("IsCraftMaterial") and item:GetAttribute("ItemType") == matType then
				count += 1
			end
		end
	end
	local char = player.Character
	if char then
		local equipped = char:FindFirstChildWhichIsA("Tool")
		if equipped and equipped:GetAttribute("IsCraftMaterial") and equipped:GetAttribute("ItemType") == matType then
			count += 1
		end
	end
	return count
end

local function consumeMaterial(player: Player, matType: string, amount: number)
	local remaining = amount
	local function tryConsume(container: Instance?)
		if not container or remaining <= 0 then
			return
		end
		for _, item in container:GetChildren() do
			if remaining <= 0 then
				break
			end
			if item:IsA("Tool") and item:GetAttribute("IsCraftMaterial") and item:GetAttribute("ItemType") == matType then
				item:Destroy()
				remaining -= 1
			end
		end
	end
	tryConsume(player:FindFirstChildOfClass("Backpack"))
	tryConsume(player.Character)
end

-- ── Target / proximity validation ──
local function isValidStation(bench: any): boolean
	if typeof(bench) ~= "Instance" or not bench:IsA("Model") then
		return false
	end
	if not CollectionService:HasTag(bench, WorkbenchConfig.STATION_TAG) then
		return false
	end
	return bench:IsDescendantOf(workspace)
end

local function isNearStation(player: Player, bench: Model): boolean
	local char = player.Character
	local hrp = char and char:FindFirstChild("HumanoidRootPart") :: BasePart?
	local bp = bench.PrimaryPart or bench:FindFirstChildWhichIsA("BasePart")
	if not hrp or not bp then
		return false
	end
	local limit = WorkbenchConfig.INTERACT_DISTANCE + WorkbenchConfig.INTERACT_DISTANCE_TOLERANCE
	return (hrp.Position - bp.Position).Magnitude <= limit
end

-- ── Per-player cooldown ──
local lastCraft: { [Player]: number } = {}
Players.PlayerRemoving:Connect(function(p)
	lastCraft[p] = nil
end)

-- ── ReqCraft ──
ReqCraft.OnServerInvoke = function(player: Player, bench: any, recipeId: any): (boolean, number)
	-- 1. recipe exists
	local recipe = WorkbenchValidation.getRecipe(recipeId)
	if recipe == nil then
		return false, Codes.UNKNOWN_RECIPE
	end

	-- 2. target is a tagged Workbench station
	if not isValidStation(bench) then
		return false, Codes.INVALID_TARGET
	end

	-- 3. proximity (server-checked; client never trusted)
	if not isNearStation(player, bench :: Model) then
		return false, Codes.TOO_FAR
	end

	-- 4. cooldown
	local now = os.clock()
	local last = lastCraft[player]
	if last and (now - last) < WorkbenchConfig.CRAFT_COOLDOWN then
		return false, Codes.COOLDOWN
	end

	-- 5. gate
	if not WorkbenchValidation.isUnlocked(recipe, isNodeUnlocked) then
		return false, Codes.LOCKED
	end

	-- 6. output feasibility (BEFORE consuming, so a bad recipe never eats materials)
	if not CraftOutputDispatcher.isFeasible(recipe.Output) then
		return false, Codes.OUTPUT_FAILED
	end

	-- 7. materials present
	local cost = WorkbenchValidation.costOf(recipe)
	for matType, need in cost do
		if countMaterial(player, matType) < need then
			return false, Codes.INSUFFICIENT_MATERIALS
		end
	end

	-- Consume → grant → cooldown
	for matType, need in cost do
		consumeMaterial(player, matType, need)
	end
	if not CraftOutputDispatcher.grant(player, recipe.Output) then
		-- Defensive: feasibility passed but grant failed (e.g. no Backpack mid-respawn).
		warn(("[WorkbenchService] grant failed for recipe %s after consume"):format(tostring(recipeId)))
		return false, Codes.OUTPUT_FAILED
	end
	lastCraft[player] = now
	return true, Codes.NONE
end

--[=[
	@class WorkbenchGuiController
	@client
	Drives the Studio-authored `WorkbenchGui`. Opens via a placed Workbench
	station's ProximityPrompt (CollectionService `WorkbenchStation`), builds the
	recipe grid from `WorkbenchConfig`, fills the detail pane on selection, sends
	`ReqCraft` on Accept, and wires Stash/Move via `PlaceableActions`. Greys
	recipes the player can't afford or that are gated by a locked Tech node;
	repaints live on Backpack and Tech-unlock changes. The recipe grid is
	filtered by category; the (future) custom dropdown drives the filter by
	firing `Signals.RequestWorkbenchCategory` — no dropdown UI exists yet.
]=]

-- Active dropdown filter. "all" shows every recipe; otherwise matches recipe.Category.

	table.insert(sortedIds, id)

-- ── Material helpers ──
local function countHeld(matType: string): number
	local count = 0
	local bp = LocalPlayer:FindFirstChildOfClass("Backpack")
	if bp then
		for _, item in bp:GetChildren() do
			if item:IsA("Tool") and item:GetAttribute("IsCraftMaterial") and item:GetAttribute("ItemType") == matType then
				count += 1
			end
		end
	end
	local char = LocalPlayer.Character
	if char then
		local equipped = char:FindFirstChildWhichIsA("Tool")
		if equipped and equipped:GetAttribute("IsCraftMaterial") and equipped:GetAttribute("ItemType") == matType then
			count += 1
		end
	end
	return count
end

local function heldCounts(): { [string]: number }
	return { Wood = countHeld("Wood"), Metal = countHeld("Metal") }
end

local function isNodeUnlocked(nodeId: string): boolean
	local folder = ReplicatedStorage:FindFirstChild("TechTreeUnlocked")
	return folder ~= nil and folder:FindFirstChild(nodeId) ~= nil
end

-- ── Detail pane ──
local function setMaterialRow(row: Frame, need: number, have: number)
	if need <= 0 then
		row.Visible = false
		return
	end
	row.Visible = true
	local countLabel = row:FindFirstChild("MaterialCount") :: TextLabel?
	if countLabel then
		countLabel.Text = ("%d/%d"):format(have, need)
		countLabel.TextColor3 = if have >= need then Color3.fromRGB(150, 230, 150) else Color3.fromRGB(230, 150, 150)
	end
end

local function refreshDetail()
	local id = selectedRecipeId
	if not id then
		acceptButton.Visible = false
		return
	end
	local recipe = WorkbenchConfig.Recipes[id]
	acceptButton.Visible = true

	recipeIcon.Image = recipe.Icon
	headerLabel.Text = recipe.DisplayName
	subHeaderLabel.Text = recipe.Description

	-- Tier pips: fill Tier1..TierN.
	for i = 1, 5 do
		local pip = tierLevels:FindFirstChild("Tier" .. i) :: Frame?
		if pip then
			pip.BackgroundTransparency = if i <= recipe.Tier then 0 else 0.7
		end
	end

	local cost = recipe.Cost
	local have = heldCounts()
	setMaterialRow(reqWood, cost.Wood or 0, have.Wood)
	setMaterialRow(reqMetal, cost.Metal or 0, have.Metal)

	local ok = WorkbenchValidation.canAfford(recipe, have)
		and WorkbenchValidation.isUnlocked(recipe, isNodeUnlocked)
	acceptButton.Active = ok
	acceptButton.AutoButtonColor = ok
	acceptButton.BackgroundTransparency = if ok then 0 else 0.5
end

-- ── Grid ──
local function recipeLocked(recipe: WorkbenchConfig.Recipe): boolean
	return not WorkbenchValidation.isUnlocked(recipe, isNodeUnlocked)
end

local function refreshGrid()
	for id, button in recipeButtons do
		local recipe = WorkbenchConfig.Recipes[id]
		local visible = currentCategory == "all" or recipe.Category == currentCategory
		button.Visible = visible
		button.ImageTransparency = if recipeLocked(recipe) then 0.6 else 0
	end
end

-- True if `cat` is one of the canonical dropdown categories.
local function isValidCategory(cat: string): boolean
	for _, c in WorkbenchConfig.Categories do
		if c == cat then
			return true
		end
	end
	return false
end

-- Apply a category filter (called by the future dropdown via RequestWorkbenchCategory).
-- Ignores unknown values; clears the detail pane if the selected recipe is now hidden.
local function setCategory(cat: string)
	if not isValidCategory(cat) or cat == currentCategory then
		return
	end
	currentCategory = cat
	if selectedRecipeId then
		local recipe = WorkbenchConfig.Recipes[selectedRecipeId]
		if recipe and currentCategory ~= "all" and recipe.Category ~= currentCategory then
			selectedRecipeId = nil
			refreshDetail()
		end
	end
	refreshGrid()
end

-- Dropdown seam: the custom category dropdown fires this signal when built.
Signals.RequestWorkbenchCategory:Connect(setCategory)

local function selectRecipe(id: string)
	selectedRecipeId = id
	refreshDetail()
end

local function buildGrid()
	for order, id in sortedIds do
		local recipe = WorkbenchConfig.Recipes[id]
		local button = itemTemplate:Clone()
		button.Name = id
		button.LayoutOrder = order
		button.Visible = true
		local icon = button:FindFirstChild("Icon") :: ImageLabel?
		if icon then
			icon.Image = recipe.Icon
		end
		button.Activated:Connect(function()
			selectRecipe(id)
		end)
		button.Parent = scroll
		recipeButtons[id] = button
	end
	refreshGrid()
end

buildGrid()

-- ── Accept ──
acceptButton.Activated:Connect(function()
	local id = selectedRecipeId
	local station = currentStation
	if not id or not station or not acceptButton.Active then
		return
	end
	acceptButton.Active = false
	local success, code = ReqCraft:Fire(station, id)
	acceptButton.Active = true
	if success then
		if WorkbenchConfig.Sounds.CraftSuccess ~= "" then
			SoundManager.PlaySFX(WorkbenchConfig.Sounds.CraftSuccess)
		end
		Signals.OnItemCrafted:Fire(id)
		refreshDetail()
	else
		if WorkbenchConfig.Sounds.CraftFail ~= "" then
			SoundManager.PlaySFX(WorkbenchConfig.Sounds.CraftFail)
		end
		notify(CODE_MESSAGES[code] or "Craft failed.")
	end
end)

-- ── Open / close ──
local function close()
	if not isOpen then
		return
	end
	isOpen = false
	gui.Enabled = false
	if actionBinding then
		actionBinding:Unbind()
		actionBinding = nil
	end
	currentStation = nil
end

local function open(station: Model)
	if isOpen and currentStation == station then
		close()
		return
	end
	if isOpen then
		close()
	end
	currentStation = station
	isOpen = true
	gui.Enabled = true
	actionBinding = PlaceableActions.bind(station, gui)
	-- Auto-close if this station is stashed/moved/destroyed while open.
	station.AncestryChanged:Connect(function(_, parent)
		if parent == nil and currentStation == station then
			close()
		end
	end)
	refreshGrid()
	refreshDetail()
end

-- ── Live repaint ──
local function repaintIfOpen()
	if isOpen then
		refreshGrid()
		refreshDetail()
	end
end

-- Backpack material changes.
local function watchBackpack(bp: Instance)
	bp.ChildAdded:Connect(repaintIfOpen)
	bp.ChildRemoved:Connect(repaintIfOpen)
end
local existingBp = LocalPlayer:FindFirstChildOfClass("Backpack")
if existingBp then
	watchBackpack(existingBp)
end
LocalPlayer.ChildAdded:Connect(function(child)
	if child:IsA("Backpack") then
		watchBackpack(child)
	end
end)

Signals.OnNodeUnlocked:Connect(repaintIfOpen)
do
	local folder = ReplicatedStorage:FindFirstChild("TechTreeUnlocked")
	if folder then
		folder.ChildAdded:Connect(repaintIfOpen)
	else
		ReplicatedStorage.ChildAdded:Connect(function(child)
			if child.Name == "TechTreeUnlocked" then
				child.ChildAdded:Connect(repaintIfOpen)
			end
		end)
	end
end

-- ── Station prompt wiring ──
local stationJanitor = Janitor.new()
local function wireStation(station: Model)
	local promptPart = station:WaitForChild(WorkbenchConfig.PROMPT_PART_NAME, 10)
	if not promptPart then
		warn(("[WorkbenchGuiController] Station '%s' missing %s"):format(station.Name, WorkbenchConfig.PROMPT_PART_NAME))
		return
	end
	local prompt = promptPart:WaitForChild("ProximityPrompt", 10) :: ProximityPrompt?
	if not prompt then
		warn(("[WorkbenchGuiController] Station '%s' missing ProximityPrompt"):format(station.Name))
		return
	end
	prompt.ActionText = "Open Workbench"
	prompt.ObjectText = "Workbench"
	prompt.MaxActivationDistance = WorkbenchConfig.INTERACT_DISTANCE
	stationJanitor:Add(prompt.Triggered:Connect(function()
		open(station)
	end), "Disconnect")
end

for _, inst in CollectionService:GetTagged(WorkbenchConfig.STATION_TAG) do
	if inst:IsA("Model") then
		wireStation(inst :: Model)
	end
end
CollectionService:GetInstanceAddedSignal(WorkbenchConfig.STATION_TAG):Connect(function(inst)
	if inst:IsA("Model") then
		wireStation(inst :: Model)
	end
end)
CollectionService:GetInstanceRemovedSignal(WorkbenchConfig.STATION_TAG):Connect(function(inst)
	if currentStation == inst then
		close()
	end
end)
