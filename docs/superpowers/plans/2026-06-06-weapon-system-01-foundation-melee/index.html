Weapon System — Plan 1: Foundation + Melee

For agentic workers: REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (- [ ]) syntax for tracking.

Goal: Stand up the shared weapon foundation (config registry, validator, damage router, Tool-driven controller, server authority) and deliver one fully working melee weapon — swing a Tool, server validates the hitbox, a tagged dummy takes damage.

Architecture: Approach B from the design spec. A data-driven WeaponConfig registry + a shared WeaponValidation module feed a client WeaponController (watches the equipped Tool, dispatches to a category handler) and a server WeaponServer (re-runs the hitbox authoritatively). Damage is routed through a single server CombatService that resolves any hit to a Player (→ existing HealthManager) or a Damageable-tagged NPC Humanoid.

Tech Stack: Luau, Rojo, Roblox Tools, Packet (typed networking), Signals (client event bus), Janitor (cleanup), SoundManager, CollectionService tags.

Source spec: docs/superpowers/specs/2026-06-06-weapon-system-design.md

Conventions for this plan (read before starting)​

  • No automated test suite exists and project preferences are to skip Selene, skip rojo build, and not auto-playtest. Each task therefore ends with a manual Studio verification the developer performs (sync via rojo serve, observe the stated result), then a commit.
  • Commits use the Haiku model per project git preference.
  • File header style: every module starts with --!strict (and --!native/--!optimize 2 where the surrounding peers use them) plus a --[=[ @class … ]=] Moonwave doc-comment, matching existing SharedModules.
  • Refinement from spec §10: ReqMeleeSwing carries no payload. The server swings from its own authoritative character CFrame and never trusts a client-supplied aim direction. (The spec's aimDir parameter is dropped as an anti-exploit tightening; later gun packets still carry origin/dir because the server needs the aim line for raycasts.)
  • Weapon Tools are "dumb": just a Handle + a Name matching a WeaponConfig key. All behavior lives in the controller/handlers, not in a per-Tool script (unlike the older GrabHook).

Task 3: WeaponValidation (shared math + target resolution)​

Files:

  • Create: src/ReplicatedStorage/SharedModules/WeaponValidation.luau

  • Step 1: Write the module

--!strict
--!native

--[=[
    @class WeaponValidation
    Pure-ish shared helpers run on BOTH client (prediction) and server
    (authority) so the two agree. Holds the fire-rate gate, the melee hitbox
    query, and part→Humanoid-model resolution. No networking; the only world
    access is the deterministic spatial query in `queryMeleeTargets`.
]=]

local WeaponValidation = {}

-- Fire-rate gate. Returns true when `cooldown` seconds have elapsed since the
-- last fire/swing. `now` is an os.clock() timestamp.
function WeaponValidation.canFire(lastFireTime: number, cooldown: number, now: number): boolean
	return (now - lastFireTime) >= cooldown
end

-- Resolve a hit part to the nearest ancestor Model that owns a Humanoid.
function WeaponValidation.resolveHumanoidModel(part: Instance): (Model?, Humanoid?)
	local model = part:FindFirstAncestorWhichIsA("Model")
	while model do
		local humanoid = model:FindFirstChildOfClass("Humanoid")
		if humanoid then
			return model, humanoid
		end
		model = model:FindFirstAncestorWhichIsA("Model")
	end
	return nil, nil
end

-- Box query in front of `rootCFrame`. Returns the unique Humanoid-bearing
-- Models overlapping the swing volume. `ignore` excludes the attacker's own
-- character (and anything else the caller wants skipped).
function WeaponValidation.queryMeleeTargets(
	rootCFrame: CFrame,
	range: number,
	hitboxSize: Vector3,
	ignore: { Instance }
): { Model }
	local center = rootCFrame * CFrame.new(0, 0, -(range / 2))
	local size = Vector3.new(hitboxSize.X, hitboxSize.Y, range)

	local params = OverlapParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	params.FilterDescendantsInstances = ignore
	params.MaxParts = 50

	local parts = workspace:GetPartBoundsInBox(center, size, params)

	local seen: { [Model]: true } = {}
	local result: { Model } = {}
	for _, part in parts do
		local model = WeaponValidation.resolveHumanoidModel(part)
		if model and not seen[model] then
			seen[model] = true
			table.insert(result, model)
		end
	end
	return result
end

return WeaponValidation

Task 5: CombatService (server damage router)​

Files:

  • Create: src/ServerStorage/SharedModules/CombatService.luau

  • Step 1: Write the module

--!strict

--[=[
    @class CombatService
    @server
    Single server-side damage router. Resolves a hit (part or Model) to a
    target and routes damage: a Player's character goes through `HealthManager`
    (keeps the HUD-tracked record correct); a Model tagged `Damageable` with a
    Humanoid takes direct Humanoid damage. Anything else is ignored. This is the
    ONLY place weapon code applies damage.

    Damageable NPC contract: a Model containing a Humanoid, tagged with the
    CollectionService tag `"Damageable"`. Enemy AI/spawning is supplied
    elsewhere; this module only deals damage.
]=]

local CollectionService = game:GetService("CollectionService")
local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local ServerStorage = game:GetService("ServerStorage")

local SharedModules = ReplicatedStorage.SharedModules
local WeaponValidation = require(SharedModules:WaitForChild("WeaponValidation"))
local WeaponTypes = require(SharedModules:WaitForChild("WeaponTypes"))
local HealthManager = require(ServerStorage.SharedModules:WaitForChild("HealthManager"))

local DAMAGEABLE_TAG = "Damageable"

local CombatService = {}

-- Apply `amount` damage to a resolved Humanoid-bearing Model. Returns true if
-- damage was routed to a valid target.
function CombatService.ApplyToModel(
	model: Model,
	humanoid: Humanoid,
	amount: number,
	attackType: WeaponTypes.DamageType,
	source: Player?,
	aoeVariant: ("Gas" | "Explosion")?
): boolean
	if amount <= 0 then
		return false
	end

	local targetPlayer = Players:GetPlayerFromCharacter(model)
	if targetPlayer then
		HealthManager.ApplyDamage(targetPlayer, amount, attackType, source, aoeVariant)
		return true
	end

	if CollectionService:HasTag(model, DAMAGEABLE_TAG) then
		if humanoid.Health > 0 then
			humanoid:TakeDamage(amount)
		end
		return true
	end

	return false
end

-- Convenience: resolve a hit part then route damage.
function CombatService.ApplyToPart(
	part: Instance,
	amount: number,
	attackType: WeaponTypes.DamageType,
	source: Player?,
	aoeVariant: ("Gas" | "Explosion")?
): boolean
	local model, humanoid = WeaponValidation.resolveHumanoidModel(part)
	if not model or not humanoid then
		return false
	end
	return CombatService.ApplyToModel(model, humanoid, amount, attackType, source, aoeVariant)
end

return CombatService

Note: HealthManager.ApplyDamage(target, amount, attackType, source, aoeVariant) already exists with the signature (Player, number, AttackType, Player?, AOEVariant?) and validates the attack type internally.

Task 6: WeaponState (server per-player cooldowns)​

Files:

  • Create: src/ServerStorage/SharedModules/WeaponState.luau

  • Step 1: Write the module

--!strict

--[=[
    @class WeaponState
    @server
    Authoritative per-player weapon state. Plan 1 tracks last-fire timestamps
    per weapon name for server-side fire-rate gating. Later plans extend the
    record with ammo (`{ mag, reserve }`) and reload timing. Call
    `WeaponState.Initialize()` once from a server bootstrap to clear state on
    player leave.
]=]

local Players = game:GetService("Players")

type Record = {
	lastFire: { [string]: number }, -- weaponName -> os.clock() of last swing/shot
}

local states: { [Player]: Record } = {}

local WeaponState = {}

local function ensure(player: Player): Record
	local record = states[player]
	if not record then
		record = { lastFire = {} }
		states[player] = record
	end
	return record
end

function WeaponState.GetLastFire(player: Player, weaponName: string): number
	local record = states[player]
	if not record then
		return -math.huge
	end
	return record.lastFire[weaponName] or -math.huge
end

function WeaponState.SetLastFire(player: Player, weaponName: string, timestamp: number)
	ensure(player).lastFire[weaponName] = timestamp
end

function WeaponState.Clear(player: Player)
	states[player] = nil
end

function WeaponState.Initialize()
	Players.PlayerRemoving:Connect(function(player)
		WeaponState.Clear(player)
	end)
end

return WeaponState

Task 7: MeleeHandler (client)​

Files:

  • Create: src/StarterPlayer/StarterPlayerScripts/WeaponHandlers/MeleeHandler.luau

The handler implements a fixed interface the WeaponController calls. HandlerContext is defined here and reused by future handlers.

  • Step 1: Write the module
--!strict

--[=[
    @class MeleeHandler
    @client
    Category handler for melee weapons. On activate it gates locally on the
    swing cooldown, plays the swing sound (and optional swing animation),
    predicts the hitbox for an immediate hitmarker via `Signals.WeaponLocalHit`,
    and sends `Packets.ReqMeleeSwing` for the server to apply authoritative
    damage. Owns no per-shot connections; the controller passes a Janitor scoped
    to the equipped Tool.
]=]

local ReplicatedStorage = game:GetService("ReplicatedStorage")

local SharedModules = ReplicatedStorage.SharedModules
local WeaponValidation = require(SharedModules:WaitForChild("WeaponValidation"))
local WeaponTypes = require(SharedModules:WaitForChild("WeaponTypes"))
local Packets = require(SharedModules:WaitForChild("Packets"))
local Signals = require(SharedModules:WaitForChild("Signals"))
local SoundManager = require(SharedModules:WaitForChild("SoundManager"))

local Janitor = require(ReplicatedStorage.Packages:WaitForChild("Janitor"))

export type HandlerContext = {
	player: Player,
	character: Model,
	tool: Tool,
	config: WeaponTypes.WeaponConfig,
	janitor: Janitor.Janitor, -- scoped to this equipped Tool
}

local MeleeHandler = {}

-- Per-character local cooldown clock, keyed by weapon name. Cleared implicitly
-- when the module's consumers drop (new character = new controller scope).
local lastSwing: { [string]: number } = {}

local function getRootPart(character: Model): BasePart?
	return character:FindFirstChild("HumanoidRootPart") :: BasePart?
end

function MeleeHandler.OnEquip(_context: HandlerContext)
	-- No equip-time setup for melee in Plan 1.
end

function MeleeHandler.OnActivate(context: HandlerContext)
	local config = context.config :: WeaponTypes.MeleeWeaponConfig
	local now = os.clock()

	if not WeaponValidation.canFire(lastSwing[config.ToolName] or -math.huge, config.SwingCooldown, now) then
		return
	end
	lastSwing[config.ToolName] = now

	-- Swing sound (positional, replicated to others).
	local rootPart = getRootPart(context.character)
	if rootPart then
		SoundManager.PlaySFXAt(config.SwingSound, rootPart.Position, { Replicate = true })
	end

	-- Optional swing animation via Animator (skeletal anim, not MotionMaster).
	if config.SwingAnimId then
		local humanoid = context.character:FindFirstChildOfClass("Humanoid")
		local animator = humanoid and humanoid:FindFirstChildOfClass("Animator")
		if animator then
			local anim = Instance.new("Animation")
			anim.AnimationId = config.SwingAnimId
			local track = animator:LoadAnimation(anim)
			track:Play()
			anim:Destroy()
		end
	end

	-- Local hit prediction → hitmarker. Damage is server-authoritative.
	if rootPart then
		local hits = WeaponValidation.queryMeleeTargets(rootPart.CFrame, config.Range, config.HitboxSize, { context.character })
		if #hits > 0 then
			Signals.WeaponLocalHit:Fire(hits[1])
		end
	end

	-- Tell the server to validate + apply damage.
	Packets.ReqMeleeSwing:Fire()
end

function MeleeHandler.OnUnequip(_context: HandlerContext)
	-- No teardown beyond the controller's Janitor in Plan 1.
end

return MeleeHandler

Task 8: WeaponController (client bootstrap)​

Files:

  • Create: src/StarterPlayer/StarterPlayerScripts/WeaponController.client.luau

Watches the character (not the Backpack) for equipped Tools — a Tool parented to the character means "equipped", which sidesteps the known Backpack-replacement churn on respawn.

  • Step 1: Write the controller
--!strict

--[=[
    @class WeaponController
    @client
    Bootstrap that drives every equipped weapon. Watches the local character for
    Tool equip (Tool parented to character) / unequip (removed), looks up the
    Tool's `WeaponConfig` entry by name, and dispatches to the category handler.
    Connects `Tool.Activated` to the handler's `OnActivate`. One Janitor per
    equipped Tool; one per character. Fires `Signals.WeaponEquipped/Unequipped`
    for the HUD.
]=]

local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")

local SharedModules = ReplicatedStorage.SharedModules
local WeaponConfig = require(SharedModules.Configs:WaitForChild("WeaponConfig"))
local Signals = require(SharedModules:WaitForChild("Signals"))

local Janitor = require(ReplicatedStorage.Packages:WaitForChild("Janitor"))

local MeleeHandler = require(script.Parent:WaitForChild("WeaponHandlers"):WaitForChild("MeleeHandler"))

type Handler = {
	OnEquip: (MeleeHandler.HandlerContext) -> (),
	OnActivate: (MeleeHandler.HandlerContext) -> (),
	OnUnequip: (MeleeHandler.HandlerContext) -> (),
}

local HANDLERS: { [string]: Handler } = {
	Melee = MeleeHandler,
	-- Gun / Throwable / Placed handlers added in later plans.
}

local player = Players.LocalPlayer

local function onToolEquipped(character: Model, tool: Tool, characterJanitor: Janitor.Janitor)
	local config = WeaponConfig[tool.Name]
	if not config then
		return -- not a managed weapon
	end
	local handler = HANDLERS[config.Category]
	if not handler then
		return
	end

	local toolJanitor = Janitor.new()
	local context: MeleeHandler.HandlerContext = {
		player = player,
		character = character,
		tool = tool,
		config = config,
		janitor = toolJanitor,
	}

	handler.OnEquip(context)
	toolJanitor:Add(tool.Activated:Connect(function()
		handler.OnActivate(context)
	end), "Disconnect")

	Signals.WeaponEquipped:Fire(tool.Name, config.Category)

	-- Tool removed from the character = unequipped (or destroyed).
	toolJanitor:Add(tool.AncestryChanged:Connect(function()
		if tool.Parent ~= character then
			handler.OnUnequip(context)
			Signals.WeaponUnequipped:Fire(tool.Name)
			characterJanitor:Remove(tool) -- triggers toolJanitor cleanup
		end
	end), "Disconnect")

	-- Register the tool's janitor under the character scope, keyed by the Tool,
	-- so character teardown cleans any still-equipped weapon.
	characterJanitor:Add(toolJanitor, "Destroy", tool)
end

local function onCharacterAdded(character: Model)
	local characterJanitor = Janitor.new()

	characterJanitor:Add(character.ChildAdded:Connect(function(child)
		if child:IsA("Tool") then
			onToolEquipped(character, child, characterJanitor)
		end
	end), "Disconnect")

	-- Handle a Tool already equipped at spawn.
	for _, child in character:GetChildren() do
		if child:IsA("Tool") then
			onToolEquipped(character, child, characterJanitor)
		end
	end

	character.AncestryChanged:Connect(function()
		if not character:IsDescendantOf(game) then
			characterJanitor:Destroy()
		end
	end)
end

if player.Character then
	onCharacterAdded(player.Character)
end
player.CharacterAdded:Connect(onCharacterAdded)

characterJanitor:Add(toolJanitor, "Destroy", tool) keys the tool's Janitor by the Tool instance, so characterJanitor:Remove(tool) on unequip both removes it from the character scope and destroys it (running Tool.Activated cleanup). This is the keyed-Janitor pattern from the project's Janitor reference.

    aim or client-claimed hits.
]=]

local ServerStorage = game:GetService("ServerStorage")

local SharedModules = ReplicatedStorage.SharedModules
local WeaponConfig = require(SharedModules.Configs:WaitForChild("WeaponConfig"))
local WeaponValidation = require(SharedModules:WaitForChild("WeaponValidation"))

WeaponState.Initialize()

local function getEquippedWeapon(player: Player): (Tool?, WeaponTypes.WeaponConfig?)
	local character = player.Character
	if not character then
		return nil, nil
	end
	local tool = character:FindFirstChildOfClass("Tool")
	if not tool then
		return nil, nil
	end
	return tool, WeaponConfig[tool.Name]
end

local function onMeleeSwing(player: Player)
	local _tool, config = getEquippedWeapon(player)
	if not config or config.Category ~= "Melee" then
		return
	end
	local melee = config :: WeaponTypes.MeleeWeaponConfig

	-- Server-authoritative cooldown gate.
	local now = os.clock()
	if not WeaponValidation.canFire(WeaponState.GetLastFire(player, melee.ToolName), melee.SwingCooldown, now) then
		return
	end

	local character = player.Character
	local rootPart = character and character:FindFirstChild("HumanoidRootPart") :: BasePart?
	if not character or not rootPart then
		return
	end

	WeaponState.SetLastFire(player, melee.ToolName, now)

	local targets = WeaponValidation.queryMeleeTargets(rootPart.CFrame, melee.Range, melee.HitboxSize, { character })
	for _, model in targets do
		local humanoid = model:FindFirstChildOfClass("Humanoid")
		if humanoid then
			local applied = CombatService.ApplyToModel(model, humanoid, melee.Damage, melee.DamageType, player)
			if applied then
				local hitRoot = model:FindFirstChild("HumanoidRootPart") :: BasePart?
				local pos = if hitRoot then hitRoot.Position else rootPart.Position
				Packets.WeaponImpact:Fire(pos, melee.ToolName)
			end
		end
	end
end

Packets.ReqMeleeSwing.OnServerEvent:Connect(onMeleeSwing)
