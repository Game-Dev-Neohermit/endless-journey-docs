Notification System Integration Implementation Plan

For agentic workers: REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (- [ ]) syntax for tracking.

Goal: Rework the imported NotificationHandler module into a Signal-triggered, config-driven NotificationManager that fixes all audit findings and integrates with SoundManager, MotionMaster, Janitor, and Signals.

Architecture: Mirror the SoundManager pattern — a Configs/NotificationConfig module holds all tuning, a renamed NotificationManager/ module is the sole consumer of two new Signals, and a one-line client bootstrap calls Init(). Audio routes through SoundManager.PlaySFX, animation through MotionMaster, all connections/tweens through Janitor. A single shared Heartbeat (dirty-checked, cached descendants) replaces the old per-notification loops.

Tech Stack: Luau, Rojo, Roblox engine. Dependencies already in-tree: SoundManager, MotionMaster, Signals, howmanysmall/janitor (ReplicatedStorage.Packages.Janitor).

Verification note: This project has no automated test suite — verification happens in Studio (per CLAUDE.md). Per user preferences: do not run selene or rojo build as verification steps; the user syncs live via Rojo and verifies in Studio. "Verify" steps below are Studio checkpoints the user performs, or Luau-LSP type checks.

Spec: docs/superpowers/specs/2026-06-04-notification-system-integration-design.md

--!strict
-- NotificationManager: client toast notifications.
-- Bootstrap with Init() once (see NotificationController.client). Trigger via
-- Signals.RequestNotification / Signals.RequestClearNotifications.
-- Audio -> SoundManager; animation -> MotionMaster; cleanup -> Janitor.

local Players = game:GetService("Players")
local TextService = game:GetService("TextService")
local UserInputService = game:GetService("UserInputService")
local RunService = game:GetService("RunService")
local Workspace = game:GetService("Workspace")
local ReplicatedStorage = game:GetService("ReplicatedStorage")

local SharedModules = ReplicatedStorage.SharedModules
local NotificationConfig = require(SharedModules.Configs.NotificationConfig)
local SoundManager = require(SharedModules.SoundManager)
local MotionMaster = require(SharedModules.MotionMaster)
local Signals = require(SharedModules.Signals)
local Janitor = require(ReplicatedStorage.Packages.Janitor)

local NotificationManager = {}

type Entry = {
	frame: Frame,
	cleaner: any, -- Janitor.Janitor
	tagged: { Instance }, -- cached tagged descendants (transparency updates)
	baseTransparency: number,
	notifType: string,
	lastT: number, -- dirty-check cache for Y-fade transparency
	isExecuted: boolean,
	createdAt: number, -- os.clock() for oldest-first eviction
}

local initialized = false
local player: Player
local playerGui: Instance
local managerJanitor: any
local entries: { Entry } = {}
local layoutCounter = 0
local pendingResize = false

-- ── Settings helpers ─────────────────────────────────────────────

local function getDevice()
	local isMobile = UserInputService.TouchEnabled and not UserInputService.KeyboardEnabled
	return isMobile and NotificationConfig.Device.Mobile or NotificationConfig.Device.Desktop
end

local function getScaled()
	local camera = Workspace.CurrentCamera
	local viewport = camera.ViewportSize
	local fontScale = math.clamp(math.min(viewport.X, viewport.Y) / NotificationConfig.BaseScreenMin, 0.7, 1.3)
	local base = getDevice()
	return {
		MinTextSize = base.MinTextSize,
		MaxTextSize = base.MaxTextSize,
		ImageSize = base.ImageSize * fontScale,
		Padding = base.Padding * fontScale,
		WidthPercentage = base.WidthPercentage,
		MaxWidth = base.MaxWidth,
		Duration = base.Duration,
		CornerRadius = base.CornerRadius * fontScale,
		CanvasWidth = base.CanvasWidth * fontScale,
		FontScale = fontScale,
	}
end

-- ── Sizing (audit #1 real interpolated size, #11 font-metric line height) ──

local function calcSize(settings, message: string)
	local camera = Workspace.CurrentCamera
	local screenWidth = camera.ViewportSize.X
	local fixedWidth = math.min(screenWidth * settings.WidthPercentage, settings.MaxWidth)
	local textAvailWidth = fixedWidth - (settings.ImageSize + settings.CanvasWidth + settings.Padding * 3)

	local scaledMin = settings.MinTextSize * settings.FontScale
	local scaledMax = settings.MaxTextSize * settings.FontScale
	-- #1: interpolate toward max based on how much shorter than the line the
	-- message is; long messages stay near min, short messages grow toward max.
	local probe = TextService:GetTextSize(message, scaledMin, NotificationConfig.Font, Vector2.new(math.huge, math.huge))
	local fitRatio = textAvailWidth > 0 and math.clamp(textAvailWidth / math.max(probe.X, 1), 0, 1) or 0
	local textSize = math.round(math.clamp(scaledMin + (scaledMax - scaledMin) * fitRatio, scaledMin, scaledMax))

	local bounds = TextService:GetTextSize(message, textSize, NotificationConfig.Font, Vector2.new(textAvailWidth, math.huge))
	-- #11: line height from font size, not a throwaway GetTextSize("") call.
	local lineHeight = textSize
	local totalTextHeight = bounds.Y + lineHeight

	return {
		Width = fixedWidth,
		Height = math.ceil(math.max(totalTextHeight + settings.Padding, settings.ImageSize + settings.Padding)),
		TextSize = textSize,
	}
end

-- ── Y-position transparency (audit #6 dirty-checked + cached descendants) ──

local function updateEntryTransparency(entry: Entry)
	local frame = entry.frame
	if not frame.Parent then
		return
	end
	local viewportHeight = Workspace.CurrentCamera.ViewportSize.Y
	local startY = viewportHeight * NotificationConfig.FadeStartY
	local endY = viewportHeight * NotificationConfig.FadeEndY
	local frameBottomY = frame.AbsolutePosition.Y + frame.AbsoluteSize.Y
	local t = frameBottomY > startY and 0 or math.clamp((startY - frameBottomY) / (startY - endY), 0, 1)

	if math.abs(t - entry.lastT) < 1e-3 then
		return -- dirty-check: unchanged this frame
	end
	entry.lastT = t

	local base = entry.baseTransparency
	frame.BackgroundTransparency = base + t * (1 - base)

	for _, child in entry.tagged do
		if child:IsA("ImageLabel") then
			child.ImageTransparency = t
		elseif child:IsA("TextLabel") then
			child.TextTransparency = t
		elseif child:IsA("Frame") then
			child.BackgroundTransparency = t
		end
	end
end

-- ── Container ────────────────────────────────────────────────────

local function ensureContainer(): Frame
	local existing = playerGui:FindFirstChild("NotificationGui")
	if existing then
		local c = existing:FindFirstChild("MainContainer")
		if c and c:IsA("Frame") then
			return c
		end
	end

	local screenGui = Instance.new("ScreenGui")
	screenGui.Name = "NotificationGui"
	screenGui.IgnoreGuiInset = true
	screenGui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
	screenGui.ResetOnSpawn = false
	screenGui.Parent = playerGui
	managerJanitor:Add(screenGui, "Destroy")

	local container = Instance.new("Frame")
	container.Name = "MainContainer"
	container.AnchorPoint = Vector2.new(0.5, 1)
	container.Position = UDim2.fromScale(0.5, 0.97)
	container.Size = UDim2.new(1, 0, 0, 0)
	container.AutomaticSize = Enum.AutomaticSize.Y
	container.BackgroundTransparency = 1
	container.Parent = screenGui

	local layout = Instance.new("UIListLayout")
	layout.SortOrder = Enum.SortOrder.LayoutOrder
	layout.HorizontalAlignment = Enum.HorizontalAlignment.Center
	layout.VerticalAlignment = Enum.VerticalAlignment.Bottom
	layout.Padding = UDim.new(0, getScaled().Padding * 0.4)
	layout.Parent = container

	return container
end

-- ── Resize (audit #10 debounced) ─────────────────────────────────

local function resizeAll()
	local existing = playerGui:FindFirstChild("NotificationGui")
	local container = existing and existing:FindFirstChild("MainContainer")
	if not container then
		return
	end
	local settings = getScaled()

	local layout = container:FindFirstChild("UIListLayout")
	if layout and layout:IsA("UIListLayout") then
		layout.Padding = UDim.new(0, settings.Padding * 0.4)
	end

	for _, entry in entries do
		local frame = entry.frame
		local textLabel = frame:FindFirstChild("MessageLabel")
		local message = (textLabel and textLabel:IsA("TextLabel")) and textLabel.Text or ""
		local sizeInfo = calcSize(settings, message)

		local corner = frame:FindFirstChild("UICorner")
		if corner and corner:IsA("UICorner") then
			corner.CornerRadius = UDim.new(0, settings.CornerRadius)
		end
		local padding = frame:FindFirstChild("UIPadding")
		if padding and padding:IsA("UIPadding") then
			padding.PaddingLeft = UDim.new(0, settings.Padding)
			padding.PaddingRight = UDim.new(0, settings.Padding)
			padding.PaddingTop = UDim.new(0, settings.Padding * 0.5)
			padding.PaddingBottom = UDim.new(0, settings.Padding * 0.5)
		end
		local icon = frame:FindFirstChild("IconLabel")
		if icon and icon:IsA("ImageLabel") then
			icon.Size = UDim2.fromOffset(settings.ImageSize, settings.ImageSize)
		end
		local timeC = frame:FindFirstChild("TimeLeftContainer")
		if timeC and timeC:IsA("Frame") then
			timeC.Size = UDim2.fromOffset(settings.CanvasWidth, settings.ImageSize)
		end
		if textLabel and textLabel:IsA("TextLabel") then
			textLabel.TextSize = sizeInfo.TextSize
			textLabel.Size = UDim2.new(1, -(settings.ImageSize + settings.CanvasWidth + settings.Padding * 3), 1, 0)
		end

		MotionMaster.Play(frame, { Size = UDim2.fromOffset(sizeInfo.Width, sizeInfo.Height) }, {
			Duration = NotificationConfig.AnimDuration,
			Model = NotificationConfig.AnimEasing,
		})
	end
end

local function requestResize()
	if pendingResize then
		return
	end
	pendingResize = true
	task.defer(function()
		pendingResize = false
		resizeAll()
	end)
end

-- ── Close / clear (audit #15 per-entry Janitor) ──────────────────

local function removeEntry(entry: Entry)
	local idx = table.find(entries, entry)
	if idx then
		table.remove(entries, idx)
	end
end

local function closeEntry(entry: Entry)
	if entry.cleaner:Get("closing") then
		return -- already closing
	end
	entry.cleaner:Add(true, nil, "closing") -- mark
	removeEntry(entry)

	local frame = entry.frame
	MotionMaster.Play(frame, { Size = UDim2.fromOffset(0, 0) }, {
		Duration = NotificationConfig.AnimDuration,
		Model = NotificationConfig.AnimEasing,
	}).OnComplete(function()
		entry.cleaner:Destroy() -- destroys frame + tweens + timers
	end)
end

-- ── Show ─────────────────────────────────────────────────────────

local function buildFrame(notifType: string, isExecuted: boolean, settings): Frame
	local frame = Instance.new("Frame")
	frame.BackgroundColor3 = NotificationConfig.Colors[notifType]
	frame.Size = UDim2.fromOffset(0, 0)
	frame.AnchorPoint = Vector2.new(0.5, 0.5)
	frame.BackgroundTransparency = NotificationConfig.BaseTransparency[notifType]
	frame.ClipsDescendants = true
	frame.Name = notifType:sub(1, 1):upper() .. notifType:sub(2) .. "_Notification"
	-- #2/#3: monotonic counter; executed notifications get a high reserved band.
	layoutCounter += 1
	frame.LayoutOrder = isExecuted and (1_000_000_000 + layoutCounter) or layoutCounter

	local corner = Instance.new("UICorner")
	corner.CornerRadius = UDim.new(0, settings.CornerRadius)
	corner.Parent = frame

	local layout = Instance.new("UIListLayout")
	layout.FillDirection = Enum.FillDirection.Horizontal
	layout.SortOrder = Enum.SortOrder.LayoutOrder
	layout.Wraps = true
	layout.Padding = UDim.new(0, settings.Padding * 0.6)
	layout.VerticalAlignment = Enum.VerticalAlignment.Center
	layout.Parent = frame

	local padding = Instance.new("UIPadding")
	padding.PaddingLeft = UDim.new(0, settings.Padding)
	padding.PaddingRight = UDim.new(0, settings.Padding)
	padding.PaddingTop = UDim.new(0, settings.Padding * 0.5)
	padding.PaddingBottom = UDim.new(0, settings.Padding * 0.5)
	padding.Parent = frame

	return frame
end

local function show(notifType: string, message: string, duration: number?, executedFunction: (() -> ())?)
	-- Validation.
	if type(notifType) ~= "string" or type(message) ~= "string" then
		return
	end
	if duration ~= nil and type(duration) ~= "number" then
		return
	end
	if executedFunction ~= nil and type(executedFunction) ~= "function" then
		return
	end
	notifType = notifType:lower()
	if not table.find(NotificationConfig.Types, notifType) then
		return
	end

	local isExecuted = executedFunction ~= nil
	local settings = getScaled()
	local container = ensureContainer()
	local sizeInfo = calcSize(settings, message)

	-- #12: enforce concurrency cap by closing the oldest non-executed entry.
	while #entries >= NotificationConfig.MaxConcurrent do
		local oldest: Entry? = nil
		for _, e in entries do
			if not e.isExecuted and (oldest == nil or e.createdAt < oldest.createdAt) then
				oldest = e
			end
		end
		if oldest then
			closeEntry(oldest)
		else
			break -- all remaining are executed; don't force-close those
		end
	end

	local frame = buildFrame(notifType, isExecuted, settings)
	local cleaner = Janitor.new()
	cleaner:Add(frame, "Destroy")

	local tagged: { Instance } = {}

	-- Icon.
	local icon = Instance.new("ImageLabel")
	icon.Image = NotificationConfig.ImageIds[notifType]
	icon.Size = UDim2.fromOffset(settings.ImageSize, settings.ImageSize)
	icon.BackgroundTransparency = 1
	icon.ImageTransparency = 1
	icon.LayoutOrder = 0
	icon.Name = "IconLabel"
	icon.Parent = frame
	table.insert(tagged, icon)

	-- Time-left bar (only for timed notifications).
	local timeFiller: Frame? = nil
	if not isExecuted then
		local timeContainer = Instance.new("Frame")
		timeContainer.Size = UDim2.fromOffset(settings.CanvasWidth, settings.ImageSize)
		timeContainer.BackgroundColor3 = Color3.fromRGB(175, 175, 175)
		timeContainer.BorderSizePixel = 0
		timeContainer.LayoutOrder = 1
		timeContainer.Name = "TimeLeftContainer"
		table.insert(tagged, timeContainer)

		local filler = Instance.new("Frame")
		filler.AnchorPoint = Vector2.new(0.5, 1)
		filler.Size = UDim2.fromScale(1, 1)
		filler.Position = UDim2.fromScale(0.5, 1)
		filler.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
		filler.BorderSizePixel = 0
		filler.Name = "TimeLeftFiller"
		filler.Parent = timeContainer
		table.insert(tagged, filler)
		timeContainer.Parent = frame
		timeFiller = filler
	end

	-- Text.
	local textLabel = Instance.new("TextLabel")
	textLabel.Text = message
	textLabel.Size = UDim2.new(1, -(settings.ImageSize + settings.CanvasWidth + settings.Padding * 3), 1, 0)
	textLabel.AnchorPoint = Vector2.new(0, 0.5)
	textLabel.BackgroundTransparency = 1
	textLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
	textLabel.Font = NotificationConfig.Font
	textLabel.TextSize = sizeInfo.TextSize
	textLabel.TextTransparency = 1
	textLabel.TextWrapped = true
	textLabel.LayoutOrder = isExecuted and 1 or 2
	textLabel.Name = "MessageLabel"
	textLabel.Parent = frame
	table.insert(tagged, textLabel)

	-- #7/#8: static gradient ramp + single Offset sweep (no per-frame rebuild).
	local textGradient = Instance.new("UIGradient")
	textGradient.Transparency = NumberSequence.new({
		NumberSequenceKeypoint.new(0, 1),
		NumberSequenceKeypoint.new(0.2, 0),
		NumberSequenceKeypoint.new(1, 0),
	})
	textGradient.Offset = Vector2.new(-1, 0)
	textGradient.Parent = textLabel

	frame.Parent = container

	local entry: Entry = {
		frame = frame,
		cleaner = cleaner,
		tagged = tagged,
		baseTransparency = NotificationConfig.BaseTransparency[notifType],
		notifType = notifType,
		lastT = -1, -- force first update
		isExecuted = isExecuted,
		createdAt = os.clock(),
	}
	table.insert(entries, entry)

	-- Audio (#13).
	SoundManager.PlaySFX(NotificationConfig.SfxNames[notifType])

	local notifTime = duration or settings.Duration

	-- Animations (#14): all via MotionMaster, tracked by the entry Janitor.
	cleaner:Add(MotionMaster.Create(textGradient, { Offset = Vector2.new(0, 0) }, {
		Duration = NotificationConfig.GradientSweepDuration,
		Model = NotificationConfig.AnimEasing,
	}):Play(), "Destroy")
	cleaner:Add(MotionMaster.Create(frame, { Size = UDim2.fromOffset(sizeInfo.Width, sizeInfo.Height) }, {
		Duration = NotificationConfig.AnimDuration,
		Model = NotificationConfig.AnimEasing,
	}):Play(), "Destroy")
	cleaner:Add(MotionMaster.Create(icon, { ImageTransparency = 0 }, {
		Duration = NotificationConfig.AnimDuration,
		Model = NotificationConfig.AnimEasing,
	}):Play(), "Destroy")
	cleaner:Add(MotionMaster.Create(textLabel, { TextTransparency = 0 }, {
		Duration = NotificationConfig.AnimDuration,
		Model = NotificationConfig.AnimEasing,
	}):Play(), "Destroy")
	if timeFiller then
		cleaner:Add(MotionMaster.Create(timeFiller, { Size = UDim2.fromScale(1, 0) }, {
			Duration = notifTime,
			Model = "Linear",
		}):Play(), "Destroy")
	end

	-- Close scheduling.
	if isExecuted and executedFunction then
		local timeout = NotificationConfig.ExecutedFunctionTimeout
		local done = false
		task.spawn(function()
			pcall(executedFunction)
			done = true
			closeEntry(entry)
		end)
		if timeout then
			task.delay(timeout, function()
				if not done then
					closeEntry(entry)
				end
			end)
		end
	else
		task.delay(notifTime, function()
			closeEntry(entry)
		end)
	end
end

local function clearAll()
	-- Iterate a copy: closeEntry mutates `entries`.
	for _, entry in table.clone(entries) do
		closeEntry(entry)
	end
end

-- ── Public ───────────────────────────────────────────────────────

function NotificationManager.Init()
	if initialized then
		return
	end
	initialized = true

	player = Players.LocalPlayer
	playerGui = player:WaitForChild("PlayerGui")
