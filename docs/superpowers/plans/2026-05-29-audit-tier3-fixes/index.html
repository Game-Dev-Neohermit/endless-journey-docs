Task 10: [PP3] Share the drop landing-detection loop​

DroppedToolService and CraftMaterialDropService each contain a near-identical "poll until on a tagged surface and at rest, or timeout" loop. Extract it.

Files:

  • Create: src/ServerStorage/SharedModules/DropLandingHelper.luau

  • Modify: src/ServerScriptService/DroppedToolService.server.luau

  • Modify: src/ServerScriptService/CraftMaterialDropService.server.luau

  • Step 1: Create the shared helper

--!strict

local CollectionService = game:GetService("CollectionService")
local ReplicatedStorage = game:GetService("ReplicatedStorage")

local ProximityPickupConfig = require(ReplicatedStorage.SharedModules.Configs:WaitForChild("ProximityPickupConfig"))

local DROPPABLE_SURFACE_TAG = "DroppableSurface"

local DropLandingHelper = {}

-- Polls until `handle` is touching a DroppableSurface part AND at rest, or until it is gone /
-- times out. Returns the outcome so callers can decide what to do.
export type LandingResult = "landed" | "gone" | "timeout"

function DropLandingHelper.waitForLanding(handle: BasePart): LandingResult
	local startTime = os.clock()
	while true do
		task.wait(ProximityPickupConfig.DROP_POLL_INTERVAL)

		if not handle.Parent then
			return "gone"
		end
		if os.clock() - startTime > ProximityPickupConfig.DROP_LANDING_TIMEOUT then
			return "timeout"
		end

		local onSurface = false
		for _, part in ipairs(handle:GetTouchingParts()) do
			if CollectionService:HasTag(part, DROPPABLE_SURFACE_TAG) then
				onSurface = true
				break
			end
		end
		if not onSurface then
			continue
		end
		if handle.AssemblyLinearVelocity.Magnitude >= ProximityPickupConfig.SETTLED_VELOCITY_THRESHOLD then
			continue
		end
		return "landed"
	end
end

return DropLandingHelper
  • Step 2: Use it in DroppedToolService

Replace the body of startLandingLoop(tool, handle) so the polling loop is the helper:

local DropLandingHelper = require(ServerStorage.SharedModules:WaitForChild("DropLandingHelper"))

local function startLandingLoop(tool: Tool, handle: BasePart)
	task.spawn(function()
		local result = DropLandingHelper.waitForLanding(handle)
		if result ~= "landed" then
			return -- gone or never reached a surface
		end

		if tool.Parent ~= workspace then
			return -- re-parented by another system before we could register it
		end
		if CollectionService:HasTag(tool, "PickupItem") then
			return -- already registered by another system
		end

		handle.Anchored = true
		handle.CFrame = CFrame.new(handle.Position) -- identity rotation for clean HoverEffect spin

		local metadata: { [string]: any } = {
			ItemType = tool:GetAttribute("ItemType"),
			ItemTier = tool:GetAttribute("ItemTier"),
		}

		ProximityPickupServer.RegisterItem(tool, metadata, function(player: Player, _meta: { [string]: any })
			tool.Parent = player.Backpack
		end, true)
	end)
end

(ServerStorage is already required at the top of the file.)

  • Step 3: Use it in CraftMaterialDropService

Replace the polling portion of its startLandingLoop (keep the untrackHandle bookkeeping):

local DropLandingHelper = require(ServerStorage.SharedModules:WaitForChild("DropLandingHelper"))

local function startLandingLoop(player: Player, handle: BasePart, pickupCallback: (Player, { [string]: any }) -> ())
	local itemType = handle:GetAttribute("ItemType") :: string
	task.spawn(function()
		local result = DropLandingHelper.waitForLanding(handle)
		if result == "gone" then
			untrackHandle(player, handle)
			return
		end
		if result == "timeout" then
			untrackHandle(player, handle)
			if handle.Parent then
				handle:Destroy()
			end
			return
		end

		-- landed
		if CollectionService:HasTag(handle, "PickupItem") then
			untrackHandle(player, handle)
			return
		end

		handle.Anchored = true
		handle.CanCollide = false
		handle.CFrame = CFrame.new(handle.Position)

		ProximityPickupServer.RegisterItem(handle, {
			ItemType = itemType,
			MaterialType = itemType,
			Quantity = 1,
		}, pickupCallback, false)

		if not handle.Parent then
			untrackHandle(player, handle)
			return -- merged away by _TryMerge
		end

		CollectionService:AddTag(handle, "CraftMaterial")
		untrackHandle(player, handle)
	end)
end
