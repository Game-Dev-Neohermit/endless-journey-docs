NotificationManager Top-Anchor + Sizing Implementation Plan

For agentic workers: REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (- [ ]) syntax for tracking.

Goal: Anchor the toast stack to the top of the screen (below the Roblox top bar, newest on top), and make text sizing reliable so long messages wrap and grow the toast taller instead of being clipped.

Architecture: Two coupled files — NotificationManager/init.luau and Configs/NotificationConfig.luau. Three commits, each leaving the module valid: (1) remove the bottom-stack Y-fade machinery, (2) flip the container to top-anchored newest-on-top, (3) switch to a fixed font size with an exact measured-equals-rendered text width.

Tech Stack: Luau, Rojo, Roblox engine. In-tree deps unchanged: SoundManager, MotionMaster, Signals, howmanysmall/janitor.

Verification note: No automated test suite (per CLAUDE.md). Do not run selene or rojo build; the user syncs live via Rojo and verifies in Studio. "Verify" steps are Luau-LSP type checks or Studio checkpoints.

Spec: docs/superpowers/specs/2026-06-04-notification-top-anchor-and-sizing-design.md

Remove this whole block:

-- ── Y-position transparency (audit #6 dirty-checked + cached descendants) ──

local function updateEntryTransparency(entry: Entry)
	if entry.entering then
		return -- intro animation owns transparency until its size tween completes
	end
	local frame = entry.frame
	if not frame.Parent then
		return
	end
	local viewportHeight = Workspace.CurrentCamera.ViewportSize.Y
	local startY = viewportHeight * NotificationConfig.FadeStartY
	local endY = viewportHeight * NotificationConfig.FadeEndY
	local frameBottomY = frame.AbsolutePosition.Y + frame.AbsoluteSize.Y
	local t = frameBottomY > startY and 0 or math.clamp((startY - frameBottomY) / (startY - endY), 0, 1)

	if math.abs(t - entry.lastT) < 1e-3 then
		return -- dirty-check: unchanged this frame
	end
	entry.lastT = t

	local base = entry.baseTransparency
	frame.BackgroundTransparency = base + t * (1 - base)

	for _, child in entry.tagged do
		if child:IsA("ImageLabel") then
			child.ImageTransparency = t
		elseif child:IsA("TextLabel") then
			child.TextTransparency = t
		elseif child:IsA("Frame") then
			child.BackgroundTransparency = t
		end
	end
end

(Leave the -- ── Container ── section header that follows it intact.)

  • Step 3: Rewrite calcSize for fixed font + exact width

Replace the entire calcSize function and its section header:

-- ── Sizing (audit #1 real interpolated size, #11 font-metric line height) ──

local function calcSize(settings, message: string)
	local camera = Workspace.CurrentCamera
	local screenWidth = camera.ViewportSize.X
	local fixedWidth = math.min(screenWidth * settings.WidthPercentage, settings.MaxWidth)
	local textAvailWidth = fixedWidth - (settings.ImageSize + settings.CanvasWidth + settings.Padding * 3)

	local scaledMin = settings.MinTextSize * settings.FontScale
	local scaledMax = settings.MaxTextSize * settings.FontScale
	-- #1: interpolate toward max based on how much shorter than the line the
	-- message is; long messages stay near min, short messages grow toward max.
	local probe = TextService:GetTextSize(message, scaledMin, NotificationConfig.Font, Vector2.new(math.huge, math.huge))
	local fitRatio = textAvailWidth > 0 and math.clamp(textAvailWidth / math.max(probe.X, 1), 0, 1) or 0
	local textSize = math.round(math.clamp(scaledMin + (scaledMax - scaledMin) * fitRatio, scaledMin, scaledMax))

	local bounds = TextService:GetTextSize(message, textSize, NotificationConfig.Font, Vector2.new(textAvailWidth, math.huge))
	-- bounds.Y already includes the full wrapped height; Padding provides breathing room.
	local totalTextHeight = bounds.Y

	return {
		Width = fixedWidth,
		Height = math.ceil(math.max(totalTextHeight + settings.Padding, settings.ImageSize + settings.Padding)),
		TextSize = textSize,
	}
end

with:

-- ── Sizing (fixed font; measured text width == rendered width) ──

local function calcSize(settings, message: string, isExecuted: boolean)
	local screenWidth = Workspace.CurrentCamera.ViewportSize.X
	local frameWidth = math.min(screenWidth * settings.WidthPercentage, settings.MaxWidth)

	-- Exact text area: subtract the frame UIPadding (both sides), the icon, the
	-- time bar (timed toasts only), and the UIListLayout gaps between items, so
	-- the wrap width we measure equals the width the label is actually given.
	local horizPadding = settings.Padding * 2
	local gap = settings.Padding * 0.6
	local reserved = settings.ImageSize + gap
	if not isExecuted then
		reserved += settings.CanvasWidth + gap
	end
	local textWidth = math.max(frameWidth - horizPadding - reserved, 1)

	local textSize = settings.TextSize
	local bounds =
		TextService:GetTextSize(message, textSize, NotificationConfig.Font, Vector2.new(textWidth, math.huge))
	local height = math.ceil(math.max(bounds.Y, settings.ImageSize) + settings.Padding)

	return {
		Width = frameWidth,
		Height = height,
		TextSize = textSize,
		TextWidth = textWidth,
	}
end
