Craft Material Drop Overhaul — Implementation Plan

For agentic workers: REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (- [ ]) syntax for tracking.

Goal: Replace Tool-as-world-object with Handle-mesh-as-world-object so craft materials never trigger Roblox auto-equip, and add a custom G-key drop that replaces the disabled Backspace drop.

Architecture: Shredder and G-drop both place a plain Model wrapper (PrimaryPart = cloned Handle BasePart) in workspace.DroppedMaterials. ProximityPickupServer.RegisterItem receives this wrapper with keepModel = false; the pickup callback clones a fresh Tool from the ReplicatedStorage.CraftingMaterials template and gives it to the player — no Tool ever lives in workspace. The landing detection loop (GetTouchingParts + velocity threshold) already exists in DroppedToolService and is replicated verbatim in the new CraftMaterialDropService.

Tech Stack: Luau, Roblox engine, MotionMaster (SplineMove), ProximityPickupServer, CollectionService, ContextActionService

Task 4: Create CraftMaterialDropService.server.luau​

Server-side handler for G-drop. Validates the request, clones the Handle from the template, applies impulse, waits for landing, then registers the item for proximity pickup.

Files:

  • Create: src/ServerScriptService/CraftMaterialDropService.server.luau

Context you need:

  • ProximityPickupConfig.DROP_POLL_INTERVAL = 0.1 (seconds between landing polls)

  • ProximityPickupConfig.DROP_LANDING_TIMEOUT = 60 (seconds before giving up)

  • ProximityPickupConfig.SETTLED_VELOCITY_THRESHOLD = 0.5 (studs/s)

  • CollectionService:GetTagged("DroppableSurface") returns the tagged surface parts

  • DroppedMaterials folder in workspace (created by ShredderServer on startup; wait for it)

  • The collision group "DroppedMaterial" is registered by DroppedToolService on startup (that service runs before this one)

  • Step 14: Create the file

--!strict

local CollectionService = game:GetService("CollectionService")
local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local ServerStorage = game:GetService("ServerStorage")

local SharedModules = ReplicatedStorage.SharedModules
local Packets = require(SharedModules:WaitForChild("Packets"))
local ProximityPickupConfig = require(SharedModules.Configs:WaitForChild("ProximityPickupConfig"))
local ShredderConfig = require(SharedModules.Configs:WaitForChild("ShredderConfig"))
local ProximityPickupServer = require(ServerStorage.SharedModules:WaitForChild("ProximityPickupServer"))

local DROPPABLE_SURFACE_TAG = "DroppableSurface"
local DROP_IMPULSE = 20 -- magnitude multiplier applied with AssemblyMass; tune up/down for arc size

-- Rate-limit: one drop per player per DROP_COOLDOWN seconds.
local DROP_COOLDOWN = 1
local lastDropTime: { [Player]: number } = {}

-- Folder that ShredderServer creates on startup.
local droppedFolder: Folder = workspace:WaitForChild(ShredderConfig.DROPPED_FOLDER_NAME) :: Folder

local materialsFolder: Folder = ReplicatedStorage:WaitForChild(ShredderConfig.MATERIALS_FOLDER) :: Folder

local function getTemplate(itemType: string): Tool?
	local t = materialsFolder:FindFirstChild(itemType)
	if not t or not t:IsA("Tool") then
		return nil
	end
	return t :: Tool
end

local function startLandingLoop(mat: Model, handle: BasePart, pickupCallback: (Player, { [string]: any }) -> ())
	local itemType = mat:GetAttribute("ItemType") :: string
	task.spawn(function()
		local startTime = os.clock()

		while true do
			task.wait(ProximityPickupConfig.DROP_POLL_INTERVAL)

			if not handle.Parent then
				return -- model was cleaned up
			end
			if os.clock() - startTime > ProximityPickupConfig.DROP_LANDING_TIMEOUT then
				-- Fell off the world — destroy the orphaned model.
				if mat.Parent then
					mat:Destroy()
				end
				return
			end

			local touching = handle:GetTouchingParts()
			local onSurface = false
			for _, part in ipairs(touching) do
				if CollectionService:HasTag(part, DROPPABLE_SURFACE_TAG) then
					onSurface = true
					break
				end
			end

			if not onSurface then
				continue
			end
			if handle.AssemblyLinearVelocity.Magnitude >= ProximityPickupConfig.SETTLED_VELOCITY_THRESHOLD then
				continue
			end

			break -- on tagged surface and at rest
		end

		if not handle.Parent then
			return
		end
		if CollectionService:HasTag(mat, "PickupItem") then
			return -- already registered by another path
		end

		handle.Anchored = true
		handle.CanCollide = false
		handle.CFrame = CFrame.new(handle.Position) -- reset rotation for clean HoverEffect spin

		ProximityPickupServer.RegisterItem(
			mat,
			{ ItemType = itemType, MaterialType = itemType, Quantity = 1 },
			pickupCallback,
			false
		)

		if not mat.Parent then
			return -- merged away by _TryMerge
		end

		CollectionService:AddTag(mat, "CraftMaterial")
	end)
end

Packets.RequestDrop.OnServerEvent:Connect(function(player: Player)
	-- Rate limit.
	local now = tick()
	local last = lastDropTime[player]
	if last and (now - last) < DROP_COOLDOWN then
		return
	end

	local character = player.Character
	if not character then
		return
	end

	local hrp = character:FindFirstChild("HumanoidRootPart") :: BasePart?
	if not hrp then
		return
	end

	-- Active hotbar slot = currently equipped Tool.
	local tool = character:FindFirstChildWhichIsA("Tool")
	if not tool then
		return
	end
	if not tool:GetAttribute("IsCraftMaterial") then
		return
	end

	local itemType = tool:GetAttribute("ItemType") :: string?
	if not itemType or itemType == "" then
		return
	end

	local template = getTemplate(itemType)
	if not template then
		warn(("[CraftMaterialDropService] No template for ItemType '%s'"):format(itemType))
		return
	end

	local handleTemplate = template:FindFirstChild("Handle") :: BasePart?
	if not handleTemplate or not handleTemplate:IsA("BasePart") then
		warn(("[CraftMaterialDropService] Template '%s' has no Handle BasePart"):format(itemType))
		return
	end

	lastDropTime[player] = now

	-- Remove the Tool from the character before spawning the world mesh.
	tool:Destroy()

	-- Build the world representation: a Model wrapper + cloned Handle.
	local handle = handleTemplate:Clone()
	local mat = Instance.new("Model")
	mat.Name = itemType
	mat:SetAttribute("MaterialType", itemType)
	mat:SetAttribute("ItemType", itemType)
	mat.PrimaryPart = handle
	handle.CollisionGroup = "DroppedMaterial"
	handle.CanCollide = true
	handle.CanTouch = true
	handle.Anchored = false
	-- Spawn just in front of the player's torso.
	handle.CFrame = hrp.CFrame * CFrame.new(0, 0, -2)
	handle.Parent = mat
	mat.Parent = droppedFolder

	-- Propel forward relative to the player's look direction.
	handle:ApplyImpulse(hrp.CFrame.LookVector * DROP_IMPULSE * handle.AssemblyMass)

	local capturedItemType = itemType
	local capturedTemplate = template

	local function pickupCallback(claimedPlayer: Player, meta: { [string]: any })
		local qty = (meta.Quantity or 1) :: number
		for _ = 1, qty do
			local newTool = capturedTemplate:Clone()
			newTool:SetAttribute("MaterialType", capturedItemType)
			newTool:SetAttribute("ItemType", capturedItemType)
			newTool:SetAttribute("IsCraftMaterial", true)
			newTool.CanBeDropped = false
			newTool.Parent = claimedPlayer.Backpack
		end
	end

	startLandingLoop(mat, handle, pickupCallback)
end)

-- Clean up rate-limit table when player leaves.
Players.PlayerRemoving:Connect(function(player: Player)
	lastDropTime[player] = nil
end)
