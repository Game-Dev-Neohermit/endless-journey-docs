Perk Client API Implementation Plan

For agentic workers: REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (- [ ]) syntax for tracking.

Goal: Replace the code-built perk GUI with a GUI-agnostic client data layer (PerkClient) that future perk screens consume.

Architecture: Invert the dependency. PerkClient (a client-facing singleton in SharedModules, like SoundManager) caches the authoritative PerkStateSync snapshot, watches the replicated PartyPerkBuffs folder, exposes read queries returning PerkView descriptors + commands with optimistic prediction + a Changed signal. PerkController.client.luau shrinks to a one-line bootstrap that Init()s it. The perk system owns zero GUI instances; the two future screens (authored in Studio) require PerkClient and subscribe to Changed.

Tech Stack: Roblox / Luau, Rojo sync. Uses the project's Packet network layer, Signal (SignalPlus) primitive, and Janitor cleanup. No new packets, no server changes, no external libraries.

Global Constraints​

  • No GUI instances in the perk system. PerkClient and the bootstrap must not call Instance.new for any GUI object. Screens are authored separately.
  • Cleanup via Janitor only — never hand-rolled connection arrays (project convention).
  • All client↔server traffic through Packets — reuse ReqAllocatePerk, ReqRespecPerks, PerkStateSync; add no new channels.
  • Changed is an in-module signal, not the global Signals bus — do not register it in Signals/init.luau.
  • Verification is in Studio. There is no automated test suite. Do not run selene or rojo build as verification steps (user syncs live with Studio). "Tests" below are Studio Play-test checks and command-bar probes.
  • EffectText formatting must match the old controller exactly: math.floor(PerLevel * level * 100 + 0.5) substituted into Description via :format.

Files:

  • Create: src/ReplicatedStorage/SharedModules/PerkClient.luau

Interfaces:

  • Consumes: Packets.PerkStateSync (OnClientEvent(levels, personalAvailable, partyAvailable)); ReplicatedStorage/PartyPerkBuffs folder attributes; Signal constructor; Janitor.

  • Produces (used by later tasks): PerkClient.Init(), PerkClient.IsReady(): boolean, PerkClient.GetAvailablePoints(): { Personal: number, Party: number }, PerkClient.Changed: Signal.Signal<>, and module-scope state (perkLevels, personalAvailable, partyAvailable) that Tasks 3 and 4 read/write.

  • Step 1: Create the module with state, helpers, sync, and Init

Create src/ReplicatedStorage/SharedModules/PerkClient.luau with exactly this content:

--!strict

--[=[
	@class PerkClient
	@client
	Client-side read model + command surface for the perk system. Owns NO GUI.
	Caches the authoritative `PerkStateSync` snapshot, watches the replicated
	`PartyPerkBuffs` folder, exposes queries returning `PerkView` descriptors,
	commands (`Allocate`/`Respec`) with optimistic prediction, and a `Changed`
	signal consumers subscribe to in order to repaint. Call `Init()` once from a
	bootstrap; any GUI then `require`s this module and reuses the singleton.
]=]

local ReplicatedStorage = game:GetService("ReplicatedStorage")

local SharedModules = ReplicatedStorage.SharedModules
local PerkConfig = require(SharedModules.Configs:WaitForChild("PerkConfig"))
local Packets = require(SharedModules:WaitForChild("Packets"))
local Signal = require(SharedModules:WaitForChild("Signals"):WaitForChild("Signal"))
local Janitor = require(ReplicatedStorage.Packages.Janitor)

local PerkClient = {}

-- Fires on every read-model change (sync, optimistic predict, party folder change).
PerkClient.Changed = Signal() :: Signal.Signal<>

-- Cached server-authoritative state (Tasks 3 and 4 read/write these).
local perkLevels: { [string]: number } = {}
local personalAvailable = 0
local partyAvailable = 0
local ready = false

local janitor = Janitor.new()
local initialized = false

-- ── Helpers (used by Task 3) ─────────────────────────────────────────────────
-- Returns (rawFraction, formattedText) for `perk` at `level`, matching the old
-- controller exactly: floor(PerLevel * level * 100 + 0.5) into Description.
local function effectFor(perk: PerkConfig.Perk, level: number): (number, string)
	local value = perk.PerLevel * level
	local percent = math.floor(value * 100 + 0.5)
	return value, (perk.Description:format(percent))
end

-- Server-wide active level of a Party perk, read from the replicated folder.
local function activePartyLevel(perk: PerkConfig.Perk): number
	local folder = ReplicatedStorage:FindFirstChild("PartyPerkBuffs")
	local raw = (folder and (folder:GetAttribute(perk.BuffKey) :: number?)) or 0
	return math.floor(raw / perk.PerLevel + 0.5)
end

-- ── Queries (reactive core) ──────────────────────────────────────────────────
function PerkClient.IsReady(): boolean
	return ready
end

function PerkClient.GetAvailablePoints(): { Personal: number, Party: number }
	return { Personal = personalAvailable, Party = partyAvailable }
end

-- ── Lifecycle ────────────────────────────────────────────────────────────────
function PerkClient.Init()
	if initialized then
		return
	end
	initialized = true

	janitor:Add(Packets.PerkStateSync.OnClientEvent:Connect(function(levels, personal, party)
		perkLevels = levels or {}
		personalAvailable = personal or 0
		partyAvailable = party or 0
		ready = true
		PerkClient.Changed:Fire()
	end), "Disconnect")

	-- The PartyPerkBuffs folder may not exist yet at join; bind when present and
	-- repaint on any attribute change (a party perk became / stopped being active).
	local function bindFolder(folder: Instance)
		janitor:Add(folder.AttributeChanged:Connect(function()
			PerkClient.Changed:Fire()
		end), "Disconnect", "PartyFolderAttr")
	end

	local existing = ReplicatedStorage:FindFirstChild("PartyPerkBuffs")
	if existing then
		bindFolder(existing)
	end
	janitor:Add(ReplicatedStorage.ChildAdded:Connect(function(child)
		if child.Name == "PartyPerkBuffs" then
			bindFolder(child)
			PerkClient.Changed:Fire()
		end
	end), "Disconnect")
end

return PerkClient

  • Step 2: Add the view builder and sort helper

Add these two functions after the activePartyLevel helper (before the -- ── Queries section):

-- Build the GUI-facing descriptor for one perk from current cached state.
local function buildView(perkId: string): PerkView
	local perk = PerkConfig.Perks[perkId]
	local level = perkLevels[perkId] or 0
	local effectValue, effectText = effectFor(perk, level)

	local avail = if perk.Category == "Personal" then personalAvailable else partyAvailable
	local view: PerkView = {
		Id = perkId,
		DisplayName = perk.DisplayName,
		Category = perk.Category,
		Level = level,
		MaxLevel = perk.MaxLevel,
		PerLevel = perk.PerLevel,
		Description = perk.Description,
		EffectText = effectText,
		EffectValue = effectValue,
		Icon = perk.Icon,
		CanAllocate = level < perk.MaxLevel and avail >= 1,
		ActivePartyLevel = nil,
		ActiveEffectText = nil,
		ActiveEffectValue = nil,
	}

	if perk.Category == "Party" then
		local activeLvl = activePartyLevel(perk)
		local activeValue, activeText = effectFor(perk, activeLvl)
		view.ActivePartyLevel = activeLvl
		view.ActiveEffectValue = activeValue
		view.ActiveEffectText = activeText
	end

	return view
end

-- Perk ids in display order; optionally filtered to one category. When both
-- categories are returned, Personal sorts before Party, each by its Order.
local function sortedIds(category: PerkConfig.PerkCategory?): { string }
	local ids = {}
	for perkId, perk in PerkConfig.Perks do
		if category == nil or perk.Category == category then
			table.insert(ids, perkId)
		end
	end
	table.sort(ids, function(a, b)
		local pa, pb = PerkConfig.Perks[a], PerkConfig.Perks[b]
		if pa.Category ~= pb.Category then
			return pa.Category < pb.Category
		end
		return pa.Order < pb.Order
	end)
	return ids
end
  • Step 3: Add the query functions

Add after PerkClient.GetAvailablePoints (inside the Queries section):

function PerkClient.GetPerks(category: PerkConfig.PerkCategory?): { PerkView }
	local out = {}
	for _, perkId in sortedIds(category) do
		table.insert(out, buildView(perkId))
	end
	return out
end

-- Perks currently providing a benefit. "Active" is category-aware: a Personal
-- perk is active when the player's Level > 0; a Party perk is active when its
-- server-wide ActivePartyLevel > 0 (buffing the server regardless of allocator).
function PerkClient.GetActivePerks(category: PerkConfig.PerkCategory?): { PerkView }
	local out = {}
	for _, view in PerkClient.GetPerks(category) do
		local isActive = if view.Category == "Party"
			then (view.ActivePartyLevel or 0) > 0
			else view.Level > 0
		if isActive then
			table.insert(out, view)
		end
	end
	return out
end

function PerkClient.GetPerk(perkId: string): PerkView?
	if not PerkConfig.Perks[perkId] then
		return nil
	end
	return buildView(perkId)
end

-- ── Commands ─────────────────────────────────────────────────────────────────
-- Optimistically predicts the spend (so the UI is instant), fires Changed, then
-- sends the request. The server is authoritative; the next PerkStateSync
-- reconciles. Returns whether the predict ran (false = doomed request, skipped).
function PerkClient.Allocate(perkId: string): boolean
	local perk = PerkConfig.Perks[perkId]
	if not perk then
		return false
	end
	local level = perkLevels[perkId] or 0
	local avail = if perk.Category == "Personal" then personalAvailable else partyAvailable
	if level >= perk.MaxLevel or avail < 1 then
		return false
	end

	perkLevels[perkId] = level + 1
	if perk.Category == "Personal" then
		personalAvailable -= 1
	else
		partyAvailable -= 1
	end
	PerkClient.Changed:Fire()
	Packets.ReqAllocatePerk:Fire(perkId)
	return true
end

-- Refund all points in one pool. Reconciliation arrives via PerkStateSync.
function PerkClient.Respec(category: PerkConfig.PerkCategory)
	local pool = if category == "Personal" then 1 else 2
	Packets.ReqRespecPerks:Fire(pool)
end

-- Whether any points are spent in `category` (lets the GUI disable Respec).
function PerkClient.CanRespec(category: PerkConfig.PerkCategory): boolean
	for perkId, level in perkLevels do
		local perk = PerkConfig.Perks[perkId]
		if perk and perk.Category == category and level > 0 then
			return true
		end
	end
	return false
end
