Tech Tree System Implementation Plan

For agentic workers: REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (- [ ]) syntax for tracking.

Goal: Build a cooperative, server-wide tech tree unlocked by pooling craftable materials (Wood/Metal) at a single fixed bench, with auto-unlock on full funding and Shredder-style ground-pile refunds on cancel.

Architecture: Client-predicts / server-authoritative, mirroring Placement and Shredder. A pure shared validator (TechTreeState) encodes the rules; TechTreeService owns the authoritative server-wide state and consumes material Tools; a config-driven TechEffectsDispatcher applies node effects; a shared MaterialEjector performs refund arcs; TechTreeGuiController drives a data-generated GUI.

Tech Stack: Luau, Rojo, Roblox engine, Packets (typed networking), Signals (in-process bus), Janitor (cleanup), MotionMaster (animation), ProximityPickupServer (ground pickups).

Testing & verification note​

This project has no automated test suite — verification happens in Studio (per CLAUDE.md), and the workflow skips selene, skips rojo build, and skips agent-driven playtesting (per saved preferences). Therefore tasks here replace the skill's pytest/TDD loop with: implement → check the stated acceptance criteria (code-level, read by the implementer) → commit. Three STUDIO CHECKPOINT blocks mark points where the user runs a live Studio playtest before continuing.

The user syncs src/ into Studio live via rojo serve, so no build step is needed.

Task 4: TechTreeState (pure shared validator)​

Files:

  • Create: src/ReplicatedStorage/SharedModules/TechTreeState.luau

This module has no side effects — both client and server call it. State is a plain table the caller owns.

  • Step 1: Write the module.
--!strict

local ReplicatedStorage = game:GetService("ReplicatedStorage")
local SharedModules = ReplicatedStorage.SharedModules
local TechTreeConfig = require(SharedModules.Configs:WaitForChild("TechTreeConfig"))
local Codes = require(SharedModules:WaitForChild("TechTreeResultCodes"))

export type State = {
	unlocked: { [string]: boolean },
	pools: { [string]: { [string]: number } }, -- nodeId -> matType -> submitted
}

local TechTreeState = {}

function TechTreeState.newState(): State
	return { unlocked = {}, pools = {} }
end

function TechTreeState.getNode(id: string): TechTreeConfig.Node?
	return TechTreeConfig.Nodes[id]
end

function TechTreeState.isUnlocked(state: State, id: string): boolean
	return state.unlocked[id] == true
end

-- Eligible = node exists, not yet unlocked, and prereq is nil or unlocked.
function TechTreeState.isEligible(state: State, id: string): boolean
	local node = TechTreeConfig.Nodes[id]
	if not node then
		return false
	end
	if state.unlocked[id] then
		return false
	end
	if node.Prereq ~= nil and not state.unlocked[node.Prereq] then
		return false
	end
	return true
end

function TechTreeState.submitted(state: State, id: string, matType: string): number
	local pool = state.pools[id]
	return (pool and pool[matType]) or 0
end

-- Remaining amount of one material before the node is fully funded.
function TechTreeState.remaining(state: State, id: string, matType: string): number
	local node = TechTreeConfig.Nodes[id]
	if not node then
		return 0
	end
	local need = node.Cost[matType] or 0
	return math.max(0, need - TechTreeState.submitted(state, id, matType))
end

-- True when every material in the node's cost is fully submitted.
function TechTreeState.isComplete(state: State, id: string): boolean
	local node = TechTreeConfig.Nodes[id]
	if not node then
		return false
	end
	for matType in node.Cost do
		if TechTreeState.remaining(state, id, matType) > 0 then
			return false
		end
	end
	return true
end

-- Authoritative submit validation. Returns (ok, reasonCode).
-- Does NOT check whether the player physically holds the materials — that is a
-- server-only concern layered on top (the client can't be trusted for it anyway).
function TechTreeState.validateSubmit(state: State, id: string, matType: string, amount: number): (boolean, number)
	local node = TechTreeConfig.Nodes[id]
	if not node then
		return false, Codes.UNKNOWN_NODE
	end
	if state.unlocked[id] then
		return false, Codes.ALREADY_UNLOCKED
	end
	if not TechTreeState.isEligible(state, id) then
		return false, Codes.PREREQ_NOT_MET
	end
	if node.Cost[matType] == nil then
		return false, Codes.INVALID_MATERIAL
	end
	if amount <= 0 then
		return false, Codes.INVALID_AMOUNT
	end
	if amount > TechTreeState.remaining(state, id, matType) then
		return false, Codes.EXCEEDS_REMAINING
	end
	return true, Codes.NONE
end

return TechTreeState

  • Step 2: Acceptance check (trace by hand).

    • newState() → {unlocked={}, pools={}}.
    • With fresh state: isEligible(state, "Durability_T1") → true; isEligible(state, "Durability_T2") → false (prereq not unlocked).
    • validateSubmit(state, "Durability_T1", "Metal", 25) → (true, 0); ..., "Metal", 26) → (false, EXCEEDS_REMAINING); ..., "Wood", 5) → (false, INVALID_MATERIAL) (T1 cost is Metal only).
    • After setting state.unlocked.Durability_T1 = true: isEligible(state,"Durability_T2") → true.

  • Step 3: Commit.

Task 7: MaterialEjector (shared server module)​

Files:

  • Create: src/ServerStorage/SharedModules/MaterialEjector.luau

Extracts the Shredder's arc-eject-as-ground-pile mechanism. Takes a bench Model and a { [matType]: count } map; arcs each material from the bench's SpawnPoint to a scattered LandingPoint, registers it for proximity pickup, and auto-collects unclaimed pieces after a delay. No tech-tree knowledge.

  • Step 1: Write the module.
--!strict

-- ┌─────────────────────────────────────────────────────────────────────────┐
-- │  MaterialEjector                                                        │
-- │  Shared server mechanism: arc a set of craft-material Tools out of a    │
-- │  bench onto the floor as a ProximityPickup pile (Shredder-style).       │
-- │  Used by the Tech Tree refund path; the Shredder may migrate onto this  │
-- │  later (out of scope here).                                             │
-- └─────────────────────────────────────────────────────────────────────────┘

local CollectionService = game:GetService("CollectionService")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local ServerStorage = game:GetService("ServerStorage")

local SharedModules = ReplicatedStorage.SharedModules
local MotionMaster = require(SharedModules:WaitForChild("MotionMaster"))
local ProximityPickupServer = require(ServerStorage.SharedModules:WaitForChild("ProximityPickupServer"))

local MaterialEjector = {}

type ArcConfig = {
	TRAJECTORY_MODEL_NAME: string,
	SPAWN_POINT_NAME: string,
	LANDING_POINT_NAME: string,
	MATERIALS_FOLDER: string,
	DROPPED_FOLDER_NAME: string,
	ARC_DURATION: number,
	ARC_HEIGHT_BOOST: number,
	LANDING_SCATTER_RADIUS: number,
	ITEM_EJECT_DELAY: number,
	MAX_ANIMATED_EJECT_ITEMS: number,
	AUTO_COLLECT_DELAY: number,
}

local function getDroppedFolder(name: string): Folder
	local f = workspace:FindFirstChild(name)
	if not f then
		f = Instance.new("Folder")
		f.Name = name
		f.Parent = workspace
	end
	return f :: Folder
end

local function getTemplate(folderName: string, matType: string): Tool?
	local folder = ReplicatedStorage:FindFirstChild(folderName)
	if not folder then
		return nil
	end
	local t = folder:FindFirstChild(matType)
	if not t or not t:IsA("Tool") then
		return nil
	end
	return t :: Tool
end

-- Eject `counts` (matType -> quantity) from `bench` onto the floor.
-- Pieces beyond MAX_ANIMATED_EJECT_ITEMS are skipped visually but still registered
-- at the landing point (no infinite arc spam).
function MaterialEjector.Eject(bench: Model, counts: { [string]: number }, config: ArcConfig)
	local trajectory = bench:FindFirstChild(config.TRAJECTORY_MODEL_NAME)
	local spawnPart = trajectory and trajectory:FindFirstChild(config.SPAWN_POINT_NAME) :: BasePart?
	local landingPart = trajectory and trajectory:FindFirstChild(config.LANDING_POINT_NAME) :: BasePart?

	-- Degrade gracefully: if trajectory missing, drop at the bench pivot with no arc.
	local P0: Vector3 = spawnPart and spawnPart.Position or bench:GetPivot().Position
	local baseLanding: Vector3 = landingPart and landingPart.Position or P0

	local droppedFolder = getDroppedFolder(config.DROPPED_FOLDER_NAME)

	-- Flatten counts into a list of material names.
	local sequence: { string } = {}
	for matType, qty in counts do
		for _ = 1, qty do
			table.insert(sequence, matType)
		end
	end

	task.spawn(function()
		local animatedCap = config.MAX_ANIMATED_EJECT_ITEMS
		for index, matType in sequence do
			local template = getTemplate(config.MATERIALS_FOLDER, matType)
			if not template then
				continue
			end
			local handleTemplate = template:FindFirstChild("Handle") :: BasePart?
			if not handleTemplate then
				continue
			end

			local handle = handleTemplate:Clone()
			handle.Name = matType
			handle:SetAttribute("MaterialType", matType)
			handle:SetAttribute("ItemType", matType)
			handle:SetAttribute("IsCraftMaterial", true)
			handle.CollisionGroup = "DroppedMaterial"
			handle.Anchored = true
			handle.CanCollide = false
			handle.CFrame = CFrame.new(P0)
			handle.Parent = droppedFolder

			-- Uniform disk scatter around the landing point.
			local angle = math.random() * math.pi * 2
			local r = config.LANDING_SCATTER_RADIUS * math.sqrt(math.random())
			local landingPos = baseLanding + Vector3.new(math.cos(angle) * r, 0, math.sin(angle) * r)

			local capturedTemplate = template
			local capturedMat = matType
			local function pickupCallback(claimer: Player)
				local tool = capturedTemplate:Clone()
				tool:SetAttribute("MaterialType", capturedMat)
				tool:SetAttribute("ItemType", capturedMat)
				tool:SetAttribute("IsCraftMaterial", true)
				tool.CanBeDropped = false
				tool.Parent = claimer.Backpack
			end

			local function register()
				if not handle.Parent then
					return
				end
				ProximityPickupServer.RegisterItem(
					handle,
					{ MaterialType = capturedMat, ItemType = capturedMat, Quantity = 1 },
					pickupCallback,
					false
				)
				if handle.Parent then
					CollectionService:AddTag(handle, "CraftMaterial")
				end
			end

			if index <= animatedCap then
				local arcPeak = (P0 + landingPos) * 0.5 + Vector3.new(0, config.ARC_HEIGHT_BOOST, 0)
				local tw = MotionMaster.SplineMove(handle, { P0, arcPeak, landingPos }, {
					Duration = config.ARC_DURATION,
					Model = "EaseInOutSine",
				})
				tw:OnComplete(function()
					tw:Destroy()
					register()
				end):Play()
				if config.ITEM_EJECT_DELAY > 0 and index < #sequence then
					task.wait(config.ITEM_EJECT_DELAY)
				end
			else
				-- Overflow: place at landing point and register without an arc.
				handle.CFrame = CFrame.new(landingPos)
				register()
			end
		end
	end)
end

return MaterialEjector

  • Step 2: Acceptance check. Eject tolerates a missing TrajectoryPoints (uses bench pivot, no error). Each produced handle gets IsCraftMaterial/ItemType/MaterialType attributes, DroppedMaterial collision group, registers with ProximityPickupServer.RegisterItem(handle, meta, cb, false), and is tagged CraftMaterial only after registration (matches Shredder ordering). Pickup callback clones the matching template into the claimer's Backpack.

  • Step 3: Commit.

Files:

  • Create: src/ServerScriptService/TechTreeService.server.luau

  • Step 1: Write the service.

--!strict

--[=[
	@class TechTreeService
	@server
	Authoritative owner of the server-wide tech tree. Holds the shared
	`unlocked`/`pools` state, validates and processes `ReqTechSubmit` /
	`ReqTechCancel`, consumes material Tools from the submitter's backpack,
	auto-unlocks nodes on full funding (firing `Signals.OnNodeUnlocked`), and
	refunds cancelled pools via `MaterialEjector`. Broadcasts per-node deltas
	with `TechNodeUpdate` and answers `ReqTechSnapshot` for late joiners.
	Session-only: no persistence.
]=]

local CollectionService = game:GetService("CollectionService")
local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local ServerStorage = game:GetService("ServerStorage")

local SharedModules = ReplicatedStorage.SharedModules
local TechTreeConfig = require(SharedModules.Configs:WaitForChild("TechTreeConfig"))
local TechTreeState = require(SharedModules:WaitForChild("TechTreeState"))
local Codes = require(SharedModules:WaitForChild("TechTreeResultCodes"))
local Packets = require(SharedModules:WaitForChild("Packets"))
local Signals = require(SharedModules:WaitForChild("Signals"))
local MaterialEjector = require(ServerStorage.SharedModules:WaitForChild("MaterialEjector"))

local ReqTechSubmit = Packets.ReqTechSubmit
local ReqTechCancel = Packets.ReqTechCancel
local TechNodeUpdate = Packets.TechNodeUpdate
local ReqTechSnapshot = Packets.ReqTechSnapshot

-- ── Authoritative server-wide state ──
local state: TechTreeState.State = TechTreeState.newState()

-- ── Bench lookup ──
local function findBench(): Model?
	for _, inst in CollectionService:GetTagged(TechTreeConfig.BENCH_TAG) do
		if inst:IsA("Model") then
			return inst :: Model
		end
	end
	return nil
end

local function isNearBench(player: Player, bench: Model): boolean
	local char = player.Character
	local hrp = char and char:FindFirstChild("HumanoidRootPart") :: BasePart?
	local bp = bench.PrimaryPart
	if not hrp or not bp then
		return false
	end
	local limit = TechTreeConfig.INTERACT_DISTANCE + TechTreeConfig.INTERACT_DISTANCE_TOLERANCE
	return (hrp.Position - bp.Position).Magnitude <= limit
end

-- ── Material Tool helpers ──
local function countMaterial(player: Player, matType: string): number
	local count = 0
	local bp = player:FindFirstChildOfClass("Backpack")
	if bp then
		for _, item in bp:GetChildren() do
			if item:IsA("Tool") and item:GetAttribute("IsCraftMaterial") and item:GetAttribute("ItemType") == matType then
				count += 1
			end
		end
	end
	local char = player.Character
	if char then
		local equipped = char:FindFirstChildWhichIsA("Tool")
		if equipped and equipped:GetAttribute("IsCraftMaterial") and equipped:GetAttribute("ItemType") == matType then
			count += 1
		end
	end
	return count
end

-- Destroys exactly `amount` matching Tools. Assumes count was already checked.
local function consumeMaterial(player: Player, matType: string, amount: number)
	local remaining = amount
	local function tryConsume(container: Instance?)
		if not container or remaining <= 0 then
			return
		end
		for _, item in container:GetChildren() do
			if remaining <= 0 then
				break
			end
			if item:IsA("Tool") and item:GetAttribute("IsCraftMaterial") and item:GetAttribute("ItemType") == matType then
				item:Destroy()
				remaining -= 1
			end
		end
	end
	tryConsume(player:FindFirstChildOfClass("Backpack"))
	tryConsume(player.Character)
end

-- ── Broadcast helpers ──
local function poolCounts(id: string): (number, number)
	return TechTreeState.submitted(state, id, "Metal"), TechTreeState.submitted(state, id, "Wood")
end

local function broadcastNode(id: string)
	local metal, wood = poolCounts(id)
	TechNodeUpdate:Fire(id, metal, wood, TechTreeState.isUnlocked(state, id))
end

local function serializeState(): { unlocked: { [string]: boolean }, pools: { [string]: { [string]: number } } }
	return { unlocked = state.unlocked, pools = state.pools }
end

-- ── Unlock ──
local function unlockNode(id: string)
	if state.unlocked[id] then
		return
	end
	state.unlocked[id] = true
	state.pools[id] = nil -- pool consumed
	Signals.OnNodeUnlocked:Fire(id)
	broadcastNode(id)
end

-- ── ReqTechSubmit ──
ReqTechSubmit.OnServerInvoke = function(player: Player, nodeId: string, matType: string, amount: number): (boolean, number)
	if typeof(nodeId) ~= "string" or typeof(matType) ~= "string" or typeof(amount) ~= "number" then
		return false, Codes.INVALID_AMOUNT
	end
	amount = math.floor(amount)

	local bench = findBench()
	if not bench then
		return false, Codes.NO_BENCH
	end
	if not isNearBench(player, bench) then
		return false, Codes.TOO_FAR
	end

	local ok, code = TechTreeState.validateSubmit(state, nodeId, matType, amount)
	if not ok then
		return false, code
	end

	if countMaterial(player, matType) < amount then
		return false, Codes.INSUFFICIENT_MATERIALS
	end

	-- Consume then credit the pool.
	consumeMaterial(player, matType, amount)
	local pool = state.pools[nodeId]
	if not pool then
		pool = {}
		state.pools[nodeId] = pool
	end
	pool[matType] = (pool[matType] or 0) + amount

	if TechTreeState.isComplete(state, nodeId) then
		unlockNode(nodeId)
	else
		broadcastNode(nodeId)
	end

	return true, Codes.NONE
end

-- ── ReqTechCancel ──
ReqTechCancel.OnServerInvoke = function(player: Player, nodeId: string): (boolean, number)
	if typeof(nodeId) ~= "string" then
		return false, Codes.UNKNOWN_NODE
	end
	if TechTreeConfig.Nodes[nodeId] == nil then
		return false, Codes.UNKNOWN_NODE
	end
	if state.unlocked[nodeId] then
		return false, Codes.ALREADY_UNLOCKED
	end

	local bench = findBench()
	if not bench then
		return false, Codes.NO_BENCH
	end
	if not isNearBench(player, bench) then
		return false, Codes.TOO_FAR
	end

	local pool = state.pools[nodeId]
	if not pool or next(pool) == nil then
		return false, Codes.NOTHING_TO_REFUND
	end

	-- Snapshot + clear before ejecting (ejection is async).
	local refund: { [string]: number } = {}
	for matType, qty in pool do
		if qty > 0 then
			refund[matType] = qty
		end
	end
	state.pools[nodeId] = nil
	broadcastNode(nodeId)

	MaterialEjector.Eject(bench, refund, TechTreeConfig)

	return true, Codes.NONE
end

-- ── Snapshot (join / GUI init) ──
ReqTechSnapshot.OnServerInvoke = function(_player: Player)
	return serializeState()
end
