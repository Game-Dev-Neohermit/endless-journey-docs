Sound System (SoundManager) Implementation Plan

For agentic workers: REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (- [ ]) syntax for tracking.

Goal: Replace the scattered, duplicated sound code with one config-driven SoundManager package handling SFX, BGM (playlist/shuffle), and ambience (global 2D beds), routed through Master→SFX/BGM/Ambience volume buses.

Architecture: A foldered client-side package ReplicatedStorage/SharedModules/SoundManager/ (facade init + Groups/Pool/SFX/Music/Ambience submodules) plus a central Configs/SoundConfig.luau. A thin client bootstrap calls SoundManager.Init() once; every other client script and the GrabHook tool call the facade. The server relay derives its replication allowlist from config. Spec: docs/superpowers/specs/2026-05-30-sound-system-design.md.

Tech Stack: Luau / Roblox, Rojo sync, GenericPoolInstance/GenericPoolObject (pooling), MotionMaster (fades — never TweenService), Packets (networking).

Conventions for this plan​

  • No rojo build, no selene — the user live-syncs via rojo serve; verification happens in Studio.
  • No automated test suite exists. Each task ends with a Studio/structural verification instead of unit tests. Where useful, verify game-tree structure with the Roblox Studio MCP (search_game_tree, inspect_instance) — do not use MCP to read/write scripts; edit the Rojo-synced files directly.
  • Commits use Claude Haiku (project preference). When a step says "Commit", perform the commit with the Haiku model.
  • Empty-string Id = no-op everywhere (preserves the existing "leave blank to disable" convention).
  • Verify after each task by confirming the file syncs without Luau errors in the Studio Output; the user can spot-check audio.

--!strict
-- Central configuration + registry for all game audio (SFX, BGM, ambience).
-- Single source of truth: edit IDs/volumes/pitch here, not in feature configs.

local SoundConfig = {}

export type SoundDef = {
	Id: string,
	Volume: number?,
	Pitch: number?,
	PitchRange: { number }?, -- {min, max}; random pitch per play (overrides Pitch)
	Looped: boolean?,
	Group: string?,          -- "SFX" | "BGM" | "Ambience"; default "SFX"
	Replicate: boolean?,     -- positional broadcast to other clients allowed
	Debounce: number?,       -- min seconds between plays of this name (anti-spam)
}

export type PlaylistDef = {
	Tracks: { string },
	Shuffle: boolean?,
	Gap: number?,       -- seconds between tracks
	Volume: number?,
	Crossfade: number?, -- seconds
}

-- Default effective volume per bus (multiplies through Master). 0..1 (engine allows up to 10).
SoundConfig.GroupVolumes = {
	Master = 1,
	SFX = 1,
	BGM = 0.6,
	Ambience = 0.5,
}

SoundConfig.Sfx = {
	PlacementPlace   = { Id = "rbxassetid://116021254194480" },
	PlacementRotate  = { Id = "" },
	PlaceableMove    = { Id = "" },
	PlaceableStash   = { Id = "" },
	ProximityPickup  = { Id = "rbxassetid://91026773425948", Volume = 0.2, PitchRange = { 0.92, 1.06 }, Debounce = 0.1 },
	GrabHookFire     = { Id = "rbxassetid://70375067578450", Volume = 0.5, PitchRange = { 1.0, 1.0 }, Replicate = true },
	GrabHookLoop     = { Id = "rbxassetid://83230748776604", Volume = 0.5, Looped = true, Replicate = true },
	GrabHookCooldown = { Id = "rbxassetid://93475294170115", Volume = 1.0 },
} :: { [string]: SoundDef }

SoundConfig.Music = {
	-- Tracks empty for now -> PlayMusic no-ops until real IDs are added.
	Playlists = {
		Default = { Tracks = {}, Shuffle = true, Gap = 1.0, Volume = 0.6, Crossfade = 1.5 },
	} :: { [string]: PlaylistDef },
}

SoundConfig.Ambience = {
	Beds = {
		-- TrainHum = { Id = "", Volume = 0.4 },
	} :: { [string]: SoundDef },
	Autostart = {} :: { string }, -- bed names to fade in on Init
}

-- Set of sound IDs that may be relayed positionally to other clients.
function SoundConfig.GetReplicatedIds(): { [string]: true }
	local ids: { [string]: true } = {}
	for _, def in SoundConfig.Sfx do
		if def.Replicate and type(def.Id) == "string" and def.Id ~= "" then
			ids[def.Id] = true
		end
	end
	return ids
end

-- All non-empty asset IDs (for preloading).
function SoundConfig.GetPreloadIds(): { string }
	local ids = {}
	for _, def in SoundConfig.Sfx do
		if type(def.Id) == "string" and def.Id ~= "" then
			table.insert(ids, def.Id)
		end
	end
	for _, bed in SoundConfig.Ambience.Beds do
		if type(bed.Id) == "string" and bed.Id ~= "" then
			table.insert(ids, bed.Id)
		end
	end
	for _, playlist in SoundConfig.Music.Playlists do
		for _, id in playlist.Tracks do
			if id ~= "" then
				table.insert(ids, id)
			end
		end
	end
	return ids
end

return SoundConfig

--!strict
-- Owns the SoundGroup bus hierarchy: Master -> {SFX, BGM, Ambience}.
-- SoundGroups nest via the `.SoundGroup` property; SoundGroup.Volume multiplies.
-- (Verified against the Roblox Engine API: "SoundGroups can be nested".)

local SoundService = game:GetService("SoundService")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local SoundConfig = require(ReplicatedStorage.SharedModules.Configs.SoundConfig)

local Groups = {}

local BUS_NAMES = { "SFX", "BGM", "Ambience" }

local master: SoundGroup? = nil
local buses: { [string]: SoundGroup } = {}

local function getOrCreate(name: string): SoundGroup
	local existing = SoundService:FindFirstChild(name)
	if existing and existing:IsA("SoundGroup") then
		return existing
	end
	local g = Instance.new("SoundGroup")
	g.Name = name
	g.Parent = SoundService
	return g
end

local function ensure()
	if master then
		return
	end
	master = getOrCreate("Master")
	(master :: SoundGroup).Volume = SoundConfig.GroupVolumes.Master or 1
	for _, name in BUS_NAMES do
		local bus = getOrCreate(name)
		bus.Volume = SoundConfig.GroupVolumes[name] or 1
		bus.SoundGroup = master
		buses[name] = bus
	end
end

function Groups.Ensure()
	ensure()
end

-- Returns the SoundGroup for a bus name ("SFX" when nil/unknown).
function Groups.Get(name: string?): SoundGroup
	ensure()
	return buses[name or "SFX"] or buses.SFX
end

function Groups.SetVolume(name: string, volume: number)
	ensure()
	if name == "Master" then
		(master :: SoundGroup).Volume = volume
		return
	end
	local bus = buses[name]
	if bus then
		bus.Volume = volume
	end
end

function Groups.GetVolume(name: string): number
	ensure()
	if name == "Master" then
		return (master :: SoundGroup).Volume
	end
	local bus = buses[name]
	return if bus then bus.Volume else 0
end

return Groups

--!strict
-- Pooled instances for positional one-shot SFX: a hidden anchored part to
-- carry the Sound, a reusable Sound, and an optional PitchShiftSoundEffect.

local ReplicatedStorage = game:GetService("ReplicatedStorage")
local SharedModules = ReplicatedStorage.SharedModules
local InstancePool = require(SharedModules.GenericPoolInstance)
local ObjectPool = require(SharedModules.GenericPoolObject)

local Pool = {}

local MAX_SIZE = 20
local INITIAL = 5
local EXPAND = 5

local staticPartPool
local soundPool
local pitchPool

local function ensure()
	if staticPartPool then
		return
	end
	local template = ReplicatedStorage:WaitForChild("StaticPart")
	assert(template:IsA("Part"), "StaticPart template is invalid")
	local poolLocation = ReplicatedStorage:WaitForChild("Pools")
	staticPartPool = InstancePool.new(template, INITIAL, MAX_SIZE, EXPAND, poolLocation)
	soundPool = ObjectPool.new(Instance, { "Sound" }, 0, MAX_SIZE, EXPAND)
	pitchPool = ObjectPool.new(Instance, { "PitchShiftSoundEffect" }, 0, MAX_SIZE, EXPAND)
end

function Pool.Ensure()
	ensure()
end

-- Acquire a part positioned at `position` with a Sound parented to it.
function Pool.Acquire(position: Vector3): (Part, Sound)
	ensure()
	local part = staticPartPool:RetrieveObject(CFrame.new(position))
	local sound = soundPool:RetrieveObject()
	part.Parent = workspace
	sound.Parent = part
	return part, sound
end

function Pool.AcquirePitch(): PitchShiftSoundEffect
	ensure()
	return pitchPool:RetrieveObject()
end

-- Return a part + sound (+optional pitch effect) to their pools.
function Pool.Release(part: Part, sound: Sound, pitch: PitchShiftSoundEffect?)
	ensure()
	if pitch then
		pitchPool:InsertObject(pitch)
	end
	task.defer(function()
		soundPool:InsertObject(sound)
		task.defer(function()
			staticPartPool:InsertObject(part)
		end)
	end)
end

return Pool

Task 4: SFX.luau (one-shots, positional, looped, replication)​

Files:

  • Create: src/ReplicatedStorage/SharedModules/SoundManager/SFX.luau

  • Step 1: Write the module

--!strict
local SoundService = game:GetService("SoundService")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local SharedModules = ReplicatedStorage.SharedModules

local SoundConfig = require(SharedModules.Configs.SoundConfig)
local Packets = require(SharedModules.Packets)
local MotionMaster = require(SharedModules.MotionMaster)
local Groups = require(script.Parent.Groups)
local Pool = require(script.Parent.Pool)

local SFX = {}

export type PlayOptions = {
	Volume: number?,
	Pitch: number?,
	PitchRange: { number }?,
	Replicate: boolean?,
	Debounce: number?,
	Position: Vector3?, -- used by PlayLooped's replicate-on-start
	FadeIn: number?,    -- PlayLooped: fade from 0 to target over N seconds
}

export type LoopHandle = {
	Sound: Sound,
	FadeTo: (self: LoopHandle, volume: number, duration: number?) -> (),
	Stop: (self: LoopHandle, fadeDuration: number?) -> (),
	Destroy: (self: LoopHandle) -> (),
}

local lastPlayed: { [string]: number } = {}

local function getDef(name: string): SoundConfig.SoundDef?
	local def = SoundConfig.Sfx[name]
	if not def then
		warn(("[SoundManager] Unknown SFX '%s'"):format(tostring(name)))
	end
	return def
end

local function passesDebounce(name: string, def: SoundConfig.SoundDef, opts: PlayOptions?): boolean
	local window = (opts and opts.Debounce) or def.Debounce
	if not window or window <= 0 then
		return true
	end
	local now = os.clock()
	local last = lastPlayed[name]
	if last and (now - last) < window then
		return false
	end
	lastPlayed[name] = now
	return true
end

local function resolvePitch(def: SoundConfig.SoundDef, opts: PlayOptions?): number
	if opts and opts.Pitch then
		return opts.Pitch
	end
	local range = (opts and opts.PitchRange) or def.PitchRange
	if range and #range == 2 then
		return range[1] + math.random() * (range[2] - range[1])
	end
	return def.Pitch or 1.0
end

local function shouldReplicate(def: SoundConfig.SoundDef, opts: PlayOptions?): boolean
	if opts and opts.Replicate ~= nil then
		return opts.Replicate
	end
	return def.Replicate == true
end

-- 2D one-shot (transient; uses PlaybackSpeed for pitch).
function SFX.Play(name: string, opts: PlayOptions?)
	local def = getDef(name)
	if not def or def.Id == "" then
		return
	end
	if not passesDebounce(name, def, opts) then
		return
	end
	local sound = Instance.new("Sound")
	sound.SoundId = def.Id
	sound.Volume = (opts and opts.Volume) or def.Volume or 0.5
	sound.PlaybackSpeed = resolvePitch(def, opts)
	sound.SoundGroup = Groups.Get(def.Group)
	sound.Parent = SoundService
	sound:Play()
	sound.Ended:Once(function()
		sound:Destroy()
	end)
end

-- Internal: pooled positional play (pitch via PitchShiftSoundEffect, matching the old SoundController).
local function playPositional(id: string, position: Vector3, volume: number, pitch: number)
	local part, sound = Pool.Acquire(position)
	sound.SoundId = id
	sound.Volume = volume
	sound.PlaybackSpeed = 1.0
	sound.SoundGroup = Groups.Get("SFX")

	local pitchEffect: PitchShiftSoundEffect? = nil
	if pitch and pitch ~= 1.0 then
		pitchEffect = Pool.AcquirePitch()
		;(pitchEffect :: PitchShiftSoundEffect).Octave = pitch
		;(pitchEffect :: PitchShiftSoundEffect).Parent = sound
	end

	sound:Play()
	if not sound.IsLoaded then
		sound.Loaded:Wait()
	end
	task.delay(sound.TimeLength + 0.05, function()
		Pool.Release(part, sound, pitchEffect)
	end)
end

-- Positional one-shot played locally; opts.Replicate (or def.Replicate) ALSO broadcasts.
function SFX.PlayAt(name: string, position: Vector3, opts: PlayOptions?)
	local def = getDef(name)
	if not def or def.Id == "" then
		return
	end
	if not passesDebounce(name, def, opts) then
		return
	end
	local volume = (opts and opts.Volume) or def.Volume or 0.5
	local pitch = resolvePitch(def, opts)

	task.spawn(playPositional, def.Id, position, volume, pitch)

	if shouldReplicate(def, opts) then
		Packets.ReplicateSoundAtLocation:Fire({
			Location = position,
			SoundConfig = { Id = def.Id, Volume = volume, Pitch = pitch },
		})
	end
end

-- Looped sound with a control handle (used by GrabHook's grapple loop).
function SFX.PlayLooped(name: string, opts: PlayOptions?): LoopHandle?
	local def = getDef(name)
	if not def or def.Id == "" then
		return nil
	end
	local targetVolume = (opts and opts.Volume) or def.Volume or 0.5
	local fadeIn = opts and opts.FadeIn

	local sound = Instance.new("Sound")
	sound.SoundId = def.Id
	sound.Looped = true
	sound.PlaybackSpeed = resolvePitch(def, opts)
	sound.SoundGroup = Groups.Get(def.Group)
	sound.Volume = if fadeIn and fadeIn > 0 then 0 else targetVolume
	sound.Parent = SoundService
	sound:Play()

	if fadeIn and fadeIn > 0 then
		MotionMaster.Play(sound, { Volume = targetVolume }, { Duration = fadeIn })
	end

	-- Parity with prior behavior: fire a single positional play to other clients on start.
	if shouldReplicate(def, opts) and opts and opts.Position then
		Packets.ReplicateSoundAtLocation:Fire({
			Location = opts.Position,
			SoundConfig = { Id = def.Id, Volume = targetVolume, Pitch = 1.0 },
		})
	end

	local handle = {} :: LoopHandle
	handle.Sound = sound
	function handle:FadeTo(volume: number, duration: number?)
		if duration and duration > 0 then
			MotionMaster.Play(sound, { Volume = volume }, { Duration = duration })
		else
			sound.Volume = volume
		end
	end
	function handle:Stop(fadeDuration: number?)
		if fadeDuration and fadeDuration > 0 then
			MotionMaster.Play(sound, { Volume = 0 }, { Duration = fadeDuration })
			task.delay(fadeDuration, function()
				if sound.Parent then
					sound:Stop()
					sound:Destroy()
				end
			end)
		else
			sound:Stop()
			sound:Destroy()
		end
	end
	function handle:Destroy()
		sound:Destroy()
	end
	return handle
end

-- Receive replicated positional sounds from other players.
function SFX.ConnectReplication()
	Packets.ReplicateSoundAtLocation.OnClientEvent:Connect(function(data)
		if typeof(data) ~= "table" or typeof(data.SoundConfig) ~= "table" then
			return
		end
		local cfg = data.SoundConfig
		if type(cfg.Id) ~= "string" or cfg.Id == "" then
			return
		end
		local position = data.Location or Vector3.zero
		task.spawn(playPositional, cfg.Id, position, cfg.Volume or 0.5, cfg.Pitch or 1.0)
	end)
end

return SFX

--!strict
local SoundService = game:GetService("SoundService")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local SharedModules = ReplicatedStorage.SharedModules

local SoundConfig = require(SharedModules.Configs.SoundConfig)
local MotionMaster = require(SharedModules.MotionMaster)
local Groups = require(script.Parent.Groups)

local Music = {}

local currentPlaylistName: string? = nil
local order: { string } = {}
local index = 0
local currentSound: Sound? = nil
local generation = 0 -- bumping invalidates pending async advances

local function buildOrder(playlist: SoundConfig.PlaylistDef): { string }
	local tracks = table.clone(playlist.Tracks)
	if playlist.Shuffle then
		for i = #tracks, 2, -1 do
			local j = math.random(i)
			tracks[i], tracks[j] = tracks[j], tracks[i]
		end
	end
	return tracks
end

local function fadeOutAndDestroy(sound: Sound, fade: number)
	if fade > 0 then
		MotionMaster.Play(sound, { Volume = 0 }, { Duration = fade })
		task.delay(fade, function()
			if sound.Parent then
				sound:Stop()
				sound:Destroy()
			end
		end)
	else
		sound:Stop()
		sound:Destroy()
	end
end

local function playIndex(playlist: SoundConfig.PlaylistDef, gen: number)
	if gen ~= generation or #order == 0 then
		return
	end
	index = (index % #order) + 1
	local id = order[index]
	local previous = currentSound

	if id == "" then
		currentSound = nil
		if previous then
			fadeOutAndDestroy(previous, playlist.Crossfade or 0)
		end
		return
	end

	local target = playlist.Volume or 0.6
	local crossfade = playlist.Crossfade or 0
	local sound = Instance.new("Sound")
	sound.SoundId = id
	sound.Looped = false
	sound.SoundGroup = Groups.Get("BGM")
	sound.Volume = if crossfade > 0 then 0 else target
	sound.Parent = SoundService
	sound:Play()
	currentSound = sound

	if crossfade > 0 then
		MotionMaster.Play(sound, { Volume = target }, { Duration = crossfade })
		if previous then
			fadeOutAndDestroy(previous, crossfade)
		end
	elseif previous then
		previous:Stop()
		previous:Destroy()
	end

	sound.Ended:Once(function()
		if gen ~= generation then
			return
		end
		task.delay(playlist.Gap or 0, function()
			playIndex(playlist, gen)
		end)
	end)
end

-- Start (or replace) the active playlist. No-op when the playlist has no tracks.
function Music.Play(playlistName: string?)
	local name = playlistName or "Default"
	local playlist = SoundConfig.Music.Playlists[name]
	if not playlist then
		warn(("[SoundManager] Unknown playlist '%s'"):format(name))
		return
	end
	if #playlist.Tracks == 0 then
		return
	end
	generation += 1
	currentPlaylistName = name
	order = buildOrder(playlist)
	index = 0
	playIndex(playlist, generation)
end

function Music.Skip()
	if not currentPlaylistName then
		return
	end
	local playlist = SoundConfig.Music.Playlists[currentPlaylistName]
	if not playlist then
		return
	end
	generation += 1
	playIndex(playlist, generation)
end

function Music.Stop(fade: number?)
	generation += 1
	currentPlaylistName = nil
	local sound = currentSound
	currentSound = nil
	if sound then
		fadeOutAndDestroy(sound, fade or 0)
	end
end

return Music

--!strict
local SoundService = game:GetService("SoundService")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local SharedModules = ReplicatedStorage.SharedModules

local SoundConfig = require(SharedModules.Configs.SoundConfig)
local MotionMaster = require(SharedModules.MotionMaster)
local Groups = require(script.Parent.Groups)

local Ambience = {}

local DEFAULT_FADE = 1.0
local active: { [string]: Sound } = {}

local function startBed(name: string, fade: number)
	if active[name] then
		return
	end
	local bed = SoundConfig.Ambience.Beds[name]
	if not bed or bed.Id == "" then
		return
	end
	local target = bed.Volume or 0.5
	local sound = Instance.new("Sound")
	sound.SoundId = bed.Id
	sound.Looped = true
	sound.SoundGroup = Groups.Get("Ambience")
	sound.Volume = if fade > 0 then 0 else target
	sound.Parent = SoundService
	sound:Play()
	if fade > 0 then
		MotionMaster.Play(sound, { Volume = target }, { Duration = fade })
	end
	active[name] = sound
end

local function stopBed(name: string, fade: number)
	local sound = active[name]
	if not sound then
		return
	end
	active[name] = nil
	if fade > 0 then
		MotionMaster.Play(sound, { Volume = 0 }, { Duration = fade })
		task.delay(fade, function()
			if sound.Parent then
				sound:Stop()
				sound:Destroy()
			end
		end)
	else
		sound:Stop()
		sound:Destroy()
	end
end

function Ambience.Add(name: string, fade: number?)
	startBed(name, fade or DEFAULT_FADE)
end

function Ambience.Remove(name: string, fade: number?)
	stopBed(name, fade or DEFAULT_FADE)
end

-- Crossfade to exactly the given set of beds.
function Ambience.Set(names: { string }, fade: number?)
	local f = if fade ~= nil then fade else DEFAULT_FADE
	local want: { [string]: true } = {}
	for _, n in names do
		want[n] = true
	end
	for name in active do
		if not want[name] then
			stopBed(name, f)
		end
	end
	for _, name in names do
		startBed(name, f)
	end
end

function Ambience.StopAll(fade: number?)
	local f = if fade ~= nil then fade else DEFAULT_FADE
	for name in active do
		stopBed(name, f)
	end
end

return Ambience

--!strict
-- SoundManager: single entry point for all game audio (SFX, BGM, ambience).
-- Call .Init() once from a client bootstrap; other client code/tools call the
-- Play* / Music / Ambience / volume methods. Fades use MotionMaster.

local ContentProvider = game:GetService("ContentProvider")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local SharedModules = ReplicatedStorage.SharedModules

local SoundConfig = require(SharedModules.Configs.SoundConfig)
local Groups = require(script.Groups)
local Pool = require(script.Pool)
local SFX = require(script.SFX)
local Music = require(script.Music)
local Ambience = require(script.Ambience)

local SoundManager = {}

export type PlayOptions = SFX.PlayOptions
export type LoopHandle = SFX.LoopHandle

local initialized = false

function SoundManager.Init()
	if initialized then
		return
	end
	initialized = true

	Groups.Ensure()
	Pool.Ensure()
	SFX.ConnectReplication()

	-- Preload all configured audio assets.
	task.spawn(function()
		local ids = SoundConfig.GetPreloadIds()
		if #ids > 0 then
			ContentProvider:PreloadAsync(ids)
		end
	end)

	-- Autostart ambience beds (no-op when none configured).
	if #SoundConfig.Ambience.Autostart > 0 then
