Craft Material Drop Landing & Pickup Registration — Implementation Plan

For agentic workers: REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (- [ ]) syntax for tracking.

Goal: Dropped craft material Tools fall and physically land on tagged surfaces, then enter the hover+spin pickup state via a single packet-driven path that replaces the existing CollectionService tag listener for all pickup items.

Architecture: A new DroppedToolService registers PhysicsService collision groups and runs a per-tool polling loop to detect landing on DroppableSurface-tagged parts. ProximityPickupServer.RegisterItem fires a new PickupItemReady packet to all clients on every registration (shredder and drop). ProximityPickupController strips its CollectionService hover wiring entirely and listens to PickupItemReady instead.

Tech Stack: Luau (strict), Roblox PhysicsService, CollectionService, in-tree Packets module (typed RemoteEvent wrapper), HoverEffect, MotionMaster. No automated test suite — verification is done via Studio playtesting.

Spec: docs/superpowers/specs/2026-05-19-craft-material-drop-landing-design.md

local HOVER_FLOOR_CLEARANCE = 0.15

local function trackModel(model: Model)
	local itemId = model:GetAttribute("PickupItemId")
	if typeof(itemId) == "string" and not trackedItems[itemId] then
		trackedItems[itemId] = model

		local primaryPart = model.PrimaryPart
		if not primaryPart then
			warn("Pickup item has no PrimaryPart:", model:GetFullName())
			return
		end

		local floorOffset = primaryPart.Size.Y + HOVER_FLOOR_CLEARANCE
		local ok, handle = pcall(
			HoverEffect.Apply,
			primaryPart,
			ProximityPickupConfig.HOVER_HEIGHT,
			ProximityPickupConfig.HOVER_DURATION,
			{ FloorOffset = floorOffset, SpinDuration = ProximityPickupConfig.SPIN_DURATION }
		)
		if ok then
			hoverHandles[itemId] = handle
		else
			warn("Failed to apply hover effect to", model:GetFullName(), ":", handle)
		end

		model.AncestryChanged:Connect(function(child, parent)
			if parent == nil then
				stopHoverEffect(itemId)
				trackedItems[itemId] = nil
			end
		end)
	end
end

for _, model in ipairs(CollectionService:GetTagged("PickupItem")) do
	trackModel(model :: Model)
end

CollectionService:GetInstanceAddedSignal("PickupItem"):Connect(trackModel)

CollectionService:GetInstanceRemovedSignal("PickupItem"):Connect(function(model: Instance)
	local itemId = model:GetAttribute("PickupItemId") :: string?
	if itemId then
		stopHoverEffect(itemId)
		trackedItems[itemId] = nil
	end
end)
  • Step 3: Add HOVER_FLOOR_CLEARANCE constant and PickupItemReady handler

In the space left by the removed code (after stopHoverEffect and before the pickup sound section), add:

local HOVER_FLOOR_CLEARANCE = 0.15

-- ─────────────────────────────────────────────────────────────────────────
-- Hover registration
-- Replaces the old CollectionService tag listener. The server fires
-- PickupItemReady after every RegisterItem call (shredder and drops),
-- and replays it to late-joining players via PlayerAdded.
-- ─────────────────────────────────────────────────────────────────────────

Packets.PickupItemReady.OnClientEvent:Connect(function(handle: BasePart, itemId: string)
	if trackedItems[itemId] then return end -- guard: late-joiner replay + normal fire can both arrive

	local model = handle.Parent :: Model
	trackedItems[itemId] = model

	local floorOffset = handle.Size.Y + HOVER_FLOOR_CLEARANCE
	local ok, hoverHandle = pcall(
		HoverEffect.Apply,
		handle,
		ProximityPickupConfig.HOVER_HEIGHT,
		ProximityPickupConfig.HOVER_DURATION,
		{ FloorOffset = floorOffset, SpinDuration = ProximityPickupConfig.SPIN_DURATION }
	)
	if ok then
		hoverHandles[itemId] = hoverHandle
	else
		warn("Failed to apply hover effect to", handle:GetFullName(), ":", hoverHandle)
	end

	model.AncestryChanged:Connect(function(_, parent)
		if parent == nil then
			stopHoverEffect(itemId)
			trackedItems[itemId] = nil
		end
	end)
end)

Task 5: Create DroppedToolService​

Files:

  • Create: src/ServerScriptService/DroppedToolService.server.luau

Before playtesting this task: Complete the Manual Studio Setup steps in the spec (docs/superpowers/specs/2026-05-19-craft-material-drop-landing-design.md → Manual Studio Setup section). Specifically: set CanBeDropped = true on the CraftingMaterials Tool templates, and tag at least one platform floor part with DroppableSurface.

  • Step 1: Create the file

Create src/ServerScriptService/DroppedToolService.server.luau with the following content:

--!strict

local CollectionService = game:GetService("CollectionService")
local PhysicsService = game:GetService("PhysicsService")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local ServerStorage = game:GetService("ServerStorage")

local SharedModules = ReplicatedStorage.SharedModules
local ProximityPickupConfig = require(SharedModules.Configs:WaitForChild("ProximityPickupConfig"))
local ProximityPickupServer = require(ServerStorage.SharedModules:WaitForChild("ProximityPickupServer"))

local DROPPED_MATERIAL_GROUP = "DroppedMaterial"
local DROPPABLE_SURFACE_TAG = "DroppableSurface"

-- ─────────────────────────────────────────────────────────────────────────
-- Collision group setup
-- DroppedMaterial passes through Default group (characters, walls, untagged
-- geometry) and only physically interacts with DroppableSurface parts.
-- ─────────────────────────────────────────────────────────────────────────

PhysicsService:RegisterCollisionGroup(DROPPED_MATERIAL_GROUP)
PhysicsService:RegisterCollisionGroup(DROPPABLE_SURFACE_TAG)
PhysicsService:CollisionGroupSetCollidable(DROPPED_MATERIAL_GROUP, "Default", false)
PhysicsService:CollisionGroupSetCollidable(DROPPED_MATERIAL_GROUP, DROPPABLE_SURFACE_TAG, true)

local function assignSurfaceGroup(instance: Instance)
	if instance:IsA("BasePart") then
		PhysicsService:SetPartCollisionGroup(instance :: BasePart, DROPPABLE_SURFACE_TAG)
	end
end

for _, instance in ipairs(CollectionService:GetTagged(DROPPABLE_SURFACE_TAG)) do
	assignSurfaceGroup(instance)
end

CollectionService:GetInstanceAddedSignal(DROPPABLE_SURFACE_TAG):Connect(assignSurfaceGroup)

-- ─────────────────────────────────────────────────────────────────────────
-- Landing detection loop
-- Polls GetTouchingParts every DROP_POLL_INTERVAL seconds. Exits when the
-- handle is touching a DroppableSurface part and its linear velocity is
-- below SETTLED_VELOCITY_THRESHOLD, or when the timeout expires.
-- ─────────────────────────────────────────────────────────────────────────

local function startLandingLoop(tool: Tool, handle: BasePart)
	task.spawn(function()
		local startTime = os.clock()

		while true do
			task.wait(ProximityPickupConfig.DROP_POLL_INTERVAL)

			if not handle.Parent then
				return -- tool removed while falling (e.g. studio cleanup)
			end

			if os.clock() - startTime > ProximityPickupConfig.DROP_LANDING_TIMEOUT then
				return -- item fell off the train or otherwise never reached a surface
			end

			local touching = handle:GetTouchingParts()
			local onSurface = false
			for _, part in ipairs(touching) do
				if CollectionService:HasTag(part, DROPPABLE_SURFACE_TAG) then
					onSurface = true
					break
				end
			end

			if not onSurface then
				continue
			end

			if handle.AssemblyLinearVelocity.Magnitude >= ProximityPickupConfig.SETTLED_VELOCITY_THRESHOLD then
				continue
			end

			break -- both conditions met: on a tagged surface and at rest
		end

		if not handle.Parent then
			return
		end

		-- Landing sequence: freeze, reset rotation, register for pickup
		handle.Anchored = true
		handle.CFrame = CFrame.new(handle.Position) -- identity rotation so HoverEffect spin starts clean

		local metadata: { [string]: any } = {
			ItemType = tool:GetAttribute("ItemType"),
			ItemTier = tool:GetAttribute("ItemTier"),
		}

		ProximityPickupServer.RegisterItem(
			tool,
			metadata,
			function(player: Player, _meta: { [string]: any })
				tool.Parent = player.Backpack
			end,
			true -- keepModel: tool IS the inventory item
		)
	end)
end

-- ─────────────────────────────────────────────────────────────────────────
-- Drop detection
-- Watches for Tool instances added as direct children of workspace that
-- carry the IsCraftMaterial attribute. Direct-child-of-workspace is the
-- tell for Roblox's default Backspace drop; ShredderServer places items
-- in workspace.DroppedMaterials (a subfolder), so it is not caught here.
-- ─────────────────────────────────────────────────────────────────────────

workspace.DescendantAdded:Connect(function(instance: Instance)
	if not instance:IsA("Tool") then return end
	if instance.Parent ~= workspace then return end
	if not instance:GetAttribute("IsCraftMaterial") then return end

	local tool = instance :: Tool
	local handle = tool:FindFirstChild("Handle") :: BasePart?
	if not handle then
		warn("[DroppedToolService] Dropped craft material has no Handle:", tool:GetFullName())
		return
	end

	PhysicsService:SetPartCollisionGroup(handle, DROPPED_MATERIAL_GROUP)
	handle.CanCollide = true
	handle.CanTouch = true
	handle.Anchored = false

	startLandingLoop(tool, handle)
end)
