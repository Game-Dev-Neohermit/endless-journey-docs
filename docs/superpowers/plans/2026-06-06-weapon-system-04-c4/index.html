Weapon System — Plan 4: C4 (Placed Explosive)

For agentic workers: REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (- [ ]) syntax for tracking.

Goal: Add C4 — a placeable explosive the player sticks to a surface within range, that arms after a delay and detonates all of the player's charges on command, dealing AOE damage. Completes v1. Depends on Plans 1–3.

Architecture: Extends the foundation. A client PlacedHandler requests placement (activate) and detonation (a key). The server validates range/max-active, spawns a server-owned prop (which replicates to all clients automatically — no render packet needed), tracks armed charges in WeaponState, and on detonate reuses CombatService.ApplyAOE + ExplosionAt from Plan 3.

Tech Stack: Luau, Rojo, Roblox Tools, Packet, Signals, Janitor, SoundManager, CollectionService.

Source spec: docs/superpowers/specs/2026-06-06-weapon-system-design.md (§3 C4 flow, §5 placed config, §13 anti-exploit).

Conventions: No automated tests (verify in Studio), commits use Haiku, --!strict + Moonwave header.

function WeaponState.TakeAllPlaced(player: Player): { PlacedCharge }
	local record = states[player]
	if not record then
		return {}
	end
	local charges = record.placed
	record.placed = {}
	return charges

Task 5: PlacedHandler (client)​

Files: Create src/StarterPlayer/StarterPlayerScripts/WeaponHandlers/PlacedHandler.luau

  • Step 1: Write the module:
--!strict

--[=[
    @class PlacedHandler
    @client
    Category handler for placed explosives (C4). On activate it raycasts under the
    crosshair to a nearby surface and requests placement; the server validates and
    spawns the (replicated) prop. Detonation is triggered by the controller's
    detonate input, which calls OnDetonate → `ReqDetonateC4`.
]=]

local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")

local SharedModules = ReplicatedStorage.SharedModules
local WeaponTypes = require(SharedModules:WaitForChild("WeaponTypes"))
local Packets = require(SharedModules:WaitForChild("Packets"))
local SoundManager = require(SharedModules:WaitForChild("SoundManager"))

local MeleeHandler = require(script.Parent:WaitForChild("MeleeHandler"))
export type HandlerContext = MeleeHandler.HandlerContext

local player = Players.LocalPlayer

local PlacedHandler = {}

function PlacedHandler.OnEquip(_context: HandlerContext) end

function PlacedHandler.OnActivate(context: HandlerContext)
	local config = context.config :: WeaponTypes.PlacedWeaponConfig
	local character = context.character

	local camera = workspace.CurrentCamera
	local viewport = camera.ViewportSize
	local ray = camera:ViewportPointToRay(viewport.X / 2, viewport.Y / 2)
	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	params.FilterDescendantsInstances = { character }
	local result = workspace:Raycast(ray.Origin, ray.Direction * (config.PlaceRange + 6), params)
	if not result then
		return
	end

	Packets.ReqPlaceC4:Fire(result.Position, result.Normal):andThen(function(success: boolean, _reason: number)
		if success then
			SoundManager.PlaySFXAt(config.PlaceSound, result.Position, { Replicate = true })
		end
	end)
end

-- Called by WeaponController on the detonate input.
function PlacedHandler.OnDetonate(_context: HandlerContext)
	Packets.ReqDetonateC4:Fire()
end

function PlacedHandler.OnUnequip(_context: HandlerContext) end

return PlacedHandler

Same :Response note as Plan 2: if the local Packet returns values directly instead of an andThen chain, use local success, reason = Packets.ReqPlaceC4:Fire(...). Confirm against ReqShredderStart usage.

Task 8: C4Server (placement + detonation authority)​

Files: Create src/ServerScriptService/C4Server.server.luau

  • Step 1: Write the server script:
--!strict

--[=[
    @class C4Server
    @server
    Authoritative C4 placement and detonation. On `ReqPlaceC4` it validates the
    placement point is within `PlaceRange` of the player and enforces `MaxActive`,
    then spawns a server-owned prop (which replicates to all clients) and records
    it with an arm time. On `ReqDetonateC4` it detonates every armed charge for
    that player via `CombatService.ApplyAOE` + `Packets.ExplosionAt`, destroying
    the props. Reuses the shared combat + state modules.
]=]

local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local ServerStorage = game:GetService("ServerStorage")

local SharedModules = ReplicatedStorage.SharedModules
local WeaponConfig = require(SharedModules.Configs:WaitForChild("WeaponConfig"))
local WeaponTypes = require(SharedModules:WaitForChild("WeaponTypes"))
local Packets = require(SharedModules:WaitForChild("Packets"))

local CombatService = require(ServerStorage.SharedModules:WaitForChild("CombatService"))
local WeaponState = require(ServerStorage.SharedModules:WaitForChild("WeaponState"))

local propsFolder = ServerStorage:WaitForChild("WeaponProps")

-- Spawned C4 props live here so they replicate and stay out of harm's filtering.
local placedFolder = Instance.new("Folder")
placedFolder.Name = "PlacedC4"
placedFolder.Parent = workspace

local REASON_OK = 0
local REASON_OUT_OF_RANGE = 1
local REASON_MAX_ACTIVE = 2
local REASON_INVALID = 3

local function getEquippedC4(player: Player): WeaponTypes.PlacedWeaponConfig?
	local character = player.Character
	local tool = character and character:FindFirstChildOfClass("Tool")
	if not tool then
		return nil
	end
	local config = WeaponConfig[tool.Name]
	if config and config.Category == "Placed" then
		return config :: WeaponTypes.PlacedWeaponConfig
	end
	return nil
end

local function onPlaceC4(player: Player, position: Vector3, normal: Vector3): (boolean, number)
	local config = getEquippedC4(player)
	if not config then
		return false, REASON_INVALID
	end

	local character = player.Character
	local root = character and character:FindFirstChild("HumanoidRootPart") :: BasePart?
	if not character or not root then
		return false, REASON_INVALID
	end
	if (position - root.Position).Magnitude > config.PlaceRange then
		return false, REASON_OUT_OF_RANGE
	end
	if WeaponState.GetPlacedCount(player) >= config.MaxActive then
		return false, REASON_MAX_ACTIVE
	end

	local template = propsFolder:FindFirstChild(config.PropName)
	if not template or not template:IsA("Model") then
		warn("[C4Server] Missing prop template:", config.PropName)
		return false, REASON_INVALID
	end

	local prop = template:Clone() :: Model
	-- Orient the prop flat against the surface (normal = up).
	prop:PivotTo(CFrame.lookAlong(position, normal))
	prop:AddTag("PlacedC4")
	prop.Parent = placedFolder

	WeaponState.AddPlaced(player, prop, os.clock() + config.ArmDelay)
	Packets.C4Placed:Fire(position)
	return true, REASON_OK
end

local function onDetonateC4(player: Player)
	local config = getEquippedC4(player)
	-- Allow detonation even if the player swapped tools: use C4 config from registry directly.
	local c4 = config or (WeaponConfig.C4 :: WeaponTypes.PlacedWeaponConfig)

	local now = os.clock()
	local charges = WeaponState.TakeAllPlaced(player)
	for _, charge in charges do
		if charge.model and charge.model.Parent and now >= charge.armedAt then
			local center = charge.model:GetPivot().Position
			CombatService.ApplyAOE(center, c4.AOERadius, c4.AOEDamage, "AOE", player, {})
			Packets.ExplosionAt:Fire(center, c4.AOERadius, "C4")
			charge.model:Destroy()
		elseif charge.model and charge.model.Parent then
			-- Not yet armed: put it back so it can detonate later.
			WeaponState.AddPlaced(player, charge.model, charge.armedAt)
		end
	end
end

Packets.ReqPlaceC4.OnServerInvoke = function(player: Player, position: Vector3, normal: Vector3)
	return onPlaceC4(player, position, normal)
end
Packets.ReqDetonateC4.OnServerEvent:Connect(onDetonateC4)

Players.PlayerRemoving:Connect(function(player)
	-- WeaponState.Clear (called by WeaponServer's WeaponState.Initialize) destroys
	-- remaining props; nothing extra needed here.
end)

CFrame.lookAlong(position, normal) orients the prop's look vector along the surface normal. If you want the C4 to lie flat (face-on the wall) adjust the offset CFrame when building the prop. The detonate handler re-adds not-yet-armed charges so an early trigger doesn't waste them.
