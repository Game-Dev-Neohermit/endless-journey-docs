Enemy Spawner Implementation Plan

For agentic workers: REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (- [ ]) syntax for tracking.

Goal: Build a generic, server-authoritative combatant spawner that places enemy/friendly NPC Models at attachment-authored spawn points (including on a moving ship) with radial jitter, placement verification, adaptive point selection, drip-feed waves, and tracked wave lifecycle.

Architecture: Three new modules — EnemySpawnerConfig (tuning + archetype registry), SpawnVerification (pure geometry validator), EnemySpawner (server service) — plus an EnemyWave handle module and a thin system loader. Spawned Models are tagged EnemyNPC/FriendlyNPC so the existing CombatService/NpcCollisionService pick them up with no further wiring. The spawner is a pure service; orchestrators call its API and react to WaveHandle.OnCleared.

Tech Stack: Luau (Roblox), Rojo project, howmanysmall/janitor (cleanup), SignalPlus-based Signals, CollectionService tags, workspace:Raycast / GetPartBoundsInBox for verification.

Testing note: This project has no automated test suite — verification is done in Roblox Studio (per CLAUDE.md). Do not run selene or rojo build; the user syncs live with Studio. Each task therefore ends with a Studio verification checkpoint (manual steps + expected result) and a commit, in place of automated tests. Use the Command Bar snippets provided to drive checks.

Spec: docs/superpowers/specs/2026-06-15-enemy-spawner-design.md

--!strict
--!native
--!optimize 2

--[=[
    @class SpawnVerification
    Stateless placement validator for spawning Models. Verifies a candidate
    world position is on solid ground (snapping Y onto the floor), clear of
    walls/ceiling, and not crowding existing enemies. Pure geometry — no
    server-only deps — so it lives in SharedModules and is independently
    testable. All v1 callers are server-side.
]=]

local ReplicatedStorage = game:GetService("ReplicatedStorage")
local EnemySpawnerConfig = require(ReplicatedStorage.SharedModules.Configs.EnemySpawnerConfig)

local SpawnVerification = {}

export type VerifyResult = {
    ok: boolean,
    position: Vector3?,   -- ground-snapped position when ok == true
    reason: string?,      -- "void" | "blocked" | "crowded" when ok == false
}

local DOWN = Vector3.new(0, -1, 0)

--[=[
    @param candidate Vector3 -- jittered world position (XZ matters; Y is a ceiling to cast from)
    @param bodySize Vector3 -- enemy body / headroom box used for overlap + Y snap
    @param ignoreList { Instance } -- excluded from ground + overlap (group model, ActiveEnemies, etc.)
    @param livingEnemies { Model } -- existing enemies tested for spacing
    @return VerifyResult
]=]
function SpawnVerification.Verify(
    candidate: Vector3,
    bodySize: Vector3,
    ignoreList: { Instance },
    livingEnemies: { Model }
): VerifyResult
    local cfg = EnemySpawnerConfig.Verification

    -- 1. GROUND BELOW: cast down from a little above the candidate.
    local rayParams = RaycastParams.new()
    rayParams.FilterType = Enum.RaycastFilterType.Exclude
    rayParams.FilterDescendantsInstances = ignoreList
    rayParams.IgnoreWater = true

    local origin = candidate + Vector3.new(0, 2, 0)
    local hit = workspace:Raycast(origin, DOWN * (cfg.MaxGroundDrop + 2), rayParams)
    if not hit then
        return { ok = false, reason = "void" }
    end

    -- Snap Y so the body sits on the floor.
    local snapped = Vector3.new(candidate.X, hit.Position.Y + bodySize.Y / 2, candidate.Z)

    -- 2. CLEAR OF WALLS / HEADROOM: overlap box centered on the snapped position.
    local overlapParams = OverlapParams.new()
    overlapParams.FilterType = Enum.RaycastFilterType.Exclude
    overlapParams.FilterDescendantsInstances = ignoreList

    local boxCF = CFrame.new(snapped)
    local parts = workspace:GetPartBoundsInBox(boxCF, EnemySpawnerConfig.Verification.HeadroomSize, overlapParams)
    for _, part in ipairs(parts) do
        if part:IsA("BasePart") and part.CanCollide then
            return { ok = false, reason = "blocked" }
        end
    end

    -- 3. SPACING vs living enemies.
    local minSpacingSq = cfg.MinEnemySpacing * cfg.MinEnemySpacing
    for _, enemy in ipairs(livingEnemies) do
        if enemy.PrimaryPart then
            local delta = enemy:GetPivot().Position - snapped
            if delta.Magnitude * delta.Magnitude <= minSpacingSq then
                return { ok = false, reason = "crowded" }
            end
        end
    end

    return { ok = true, position = snapped }
end

return SpawnVerification

Task 3: EnemyWave lifecycle handle​

Files:

  • Create: src/ServerStorage/SharedModules/EnemyWave.luau

  • Step 1: Write the wave handle module

--!strict

--[=[
    @class EnemyWave
    @server
    Factory for a WaveHandle: a Janitor-backed tracker over a set of spawned
    enemy Models. Watches each enemy's Humanoid.Died and removal; when the live
    count reaches 0 AND the wave has been marked complete (all drip-feed spawns
    attempted), fires OnCleared exactly once. Returned by EnemySpawner.SpawnWave.
]=]

local ReplicatedStorage = game:GetService("ReplicatedStorage")
local Janitor = require(ReplicatedStorage.Packages.Janitor)
local Signal = require(ReplicatedStorage.SharedModules.Signals.Signal)

local EnemyWave = {}
EnemyWave.__index = EnemyWave

export type WaveHandle = {
    OnCleared: any,                              -- Signal fired once when last enemy dies
    Register: (self: WaveHandle, model: Model) -> (),
    MarkComplete: (self: WaveHandle) -> (),       -- called by SpawnWave after all spawns attempted
    GetAlive: (self: WaveHandle) -> number,
    GetEnemies: (self: WaveHandle) -> { Model },
    Despawn: (self: WaveHandle) -> (),
    Destroy: (self: WaveHandle) -> (),
}

function EnemyWave.new(): WaveHandle
    local self = setmetatable({}, EnemyWave)
    self._janitor = Janitor.new()
    self._alive = {} :: { [Model]: true }
    self._complete = false
    self._cleared = false
    self.OnCleared = self._janitor:Add(Signal(), "Destroy")
    return (self :: any) :: WaveHandle
end

function EnemyWave:_checkCleared()
    if self._cleared or not self._complete then
        return
    end
    if next(self._alive) == nil then
        self._cleared = true
        self.OnCleared:Fire()
    end
end

function EnemyWave:_remove(model: Model)
    if self._alive[model] then
        self._alive[model] = nil
        self:_checkCleared()
    end
end

function EnemyWave:Register(model: Model)
    self._alive[model] = true
    local humanoid = model:FindFirstChildOfClass("Humanoid")
    if humanoid then
        self._janitor:Add(humanoid.Died:Connect(function()
            self:_remove(model)
        end), "Disconnect")
    end
    -- Also handle a model leaving the world (despawn/destroy) without a Died event.
    self._janitor:Add(model.AncestryChanged:Connect(function(_, parent)
        if parent == nil then
            self:_remove(model)
        end
    end), "Disconnect")
end

function EnemyWave:MarkComplete()
    self._complete = true
    self:_checkCleared()
end

function EnemyWave:GetAlive(): number
    local n = 0
    for _ in pairs(self._alive) do
        n += 1
    end
    return n
end

function EnemyWave:GetEnemies(): { Model }
    local list = {}
    for model in pairs(self._alive) do
        table.insert(list, model)
    end
    return list
end

function EnemyWave:Despawn()
    for model in pairs(self._alive) do
        model:Destroy()
    end
    table.clear(self._alive)
    self:_checkCleared()
end

function EnemyWave:Destroy()
    self:Despawn()
    self._janitor:Destroy()
end

return EnemyWave

Task 5: EnemySpawner service​

Files:

  • Create: src/ServerStorage/SharedModules/EnemySpawner.luau

  • Step 1: Write the service module

--!strict

--[=[
    @class EnemySpawner
    @server
    Generic, server-authoritative combatant spawner. Places enemy/friendly NPC
    Models at attachment-authored spawn points (tag EnemySpawnGroup / point tag
    EnemySpawnPoint), with radial jitter, SpawnVerification, cumulative-inverse
    weighted point selection, drip-feed waves, and a tracked WaveHandle. Spawned
    Models are tagged EnemyNPC/FriendlyNPC so CombatService/NpcCollisionService
    pick them up automatically. Owns spawning only — orchestrators decide when.
]=]

local ReplicatedStorage = game:GetService("ReplicatedStorage")
local ServerStorage = game:GetService("ServerStorage")
local CollectionService = game:GetService("CollectionService")

local Shared = ReplicatedStorage.SharedModules
local Config = require(Shared.Configs.EnemySpawnerConfig)
local SpawnVerification = require(Shared.SpawnVerification)
local Factions = require(Shared.Factions)
local Signals = require(Shared.Signals)
local EnemyWave = require(ServerStorage.SharedModules.EnemyWave)

local EnemySpawner = {}

type PointEntry = {
    attachment: Attachment,
    radius: number,
    weight: number,
}

type GroupEntry = {
    model: Model,
    points: { PointEntry },
    counts: { [Attachment]: number },   -- cumulative spawns this encounter
}

export type SpawnComposition = string | { [string]: number }

export type SpawnRequest = {
    count: number,
    composition: SpawnComposition,
    interval: number?,   -- seconds between spawns; nil/0 = all at once
    seed: number?,       -- optional deterministic RNG
}

local groups: { [string]: GroupEntry } = {}
local rng = Random.new()
local activeFolder: Folder

-- [[ HELPERS ]]

local function getActiveFolder(): Folder
    if not activeFolder then
        local existing = workspace:FindFirstChild(Config.ActiveEnemiesFolderName)
        if existing and existing:IsA("Folder") then
            activeFolder = existing
        else
            local f = Instance.new("Folder")
            f.Name = Config.ActiveEnemiesFolderName
            f.Parent = workspace
            activeFolder = f
        end
    end
    return activeFolder
end

local function groupNameOf(model: Model): string
    local attr = model:GetAttribute("GroupName")
    if typeof(attr) == "string" then
        return attr
    end
    return model.Name
end

local function buildPoints(model: Model): { PointEntry }
    local points: { PointEntry } = {}
    for _, desc in ipairs(model:GetDescendants()) do
        if desc:IsA("Attachment") and CollectionService:HasTag(desc, Config.PointTag) then
            local radius = desc:GetAttribute("Radius")
            local weight = desc:GetAttribute("Weight")
            table.insert(points, {
                attachment = desc,
                radius = if typeof(radius) == "number" then radius else Config.DEFAULT_POINT_RADIUS,
                weight = if typeof(weight) == "number" then weight else 1,
            })
        end
    end
    return points
end

local function resolveGroup(groupName: string): GroupEntry?
    local cached = groups[groupName]
    if cached then
        return cached
    end
    for _, model in ipairs(CollectionService:GetTagged(Config.GroupTag)) do
        if model:IsA("Model") and groupNameOf(model) == groupName then
            local entry: GroupEntry = {
                model = model,
                points = buildPoints(model),
                counts = {},
            }
            groups[groupName] = entry
            return entry
        end
    end
    warn(string.format("EnemySpawner: no EnemySpawnGroup named '%s'", groupName))
    return nil
end

-- Weighted pick favoring points with the fewest cumulative spawns.
local function selectPoint(group: GroupEntry): PointEntry?
    local total = 0
    local weights: { number } = {}
    for i, p in ipairs(group.points) do
        local count = group.counts[p.attachment] or 0
        local eff = p.weight / (count + 1) ^ Config.WeightFalloff
        weights[i] = eff
        total += eff
    end
    if total <= 0 then
        return nil
    end
    local ticket = rng:NextNumber() * total
    local acc = 0
    for i, p in ipairs(group.points) do
        acc += weights[i]
        if ticket <= acc then
            return p
        end
    end
    return group.points[#group.points]
end

local function jitter(origin: Vector3, radius: number): Vector3
    if radius <= 0 then
        return origin
    end
    local angle = rng:NextNumber() * math.pi * 2
    local dist = math.sqrt(rng:NextNumber()) * radius   -- uniform over the disc
    return origin + Vector3.new(math.cos(angle) * dist, 0, math.sin(angle) * dist)
end

local function bodySizeOf(template: Model): Vector3
    local hum = template:FindFirstChildOfClass("Humanoid")
    if hum then
        return Vector3.new(3, hum.HipHeight + 4, 3)
    end
    return Config.Verification.HeadroomSize
end

local function livingEnemies(): { Model }
    local list = {}
    for _, child in ipairs(getActiveFolder():GetChildren()) do
        if child:IsA("Model") then
            table.insert(list, child)
        end
    end
    return list
end

local function atGlobalCap(): boolean
    return #getActiveFolder():GetChildren() >= Config.GLOBAL_MAX_ENEMIES
end

local function applyArchetype(model: Model, archetype: Config.Archetype)
    local hum = model:FindFirstChildOfClass("Humanoid")
    if hum then
        hum.MaxHealth = archetype.Health
        hum.Health = archetype.Health
        hum.WalkSpeed = archetype.WalkSpeed
    end
    local tag = if archetype.Faction == "Friendly" then Factions.FriendlyTag else Factions.EnemyTag
    CollectionService:AddTag(model, tag)
end

-- Place one verified enemy at a chosen point. Returns the Model or nil.
local function placeAt(point: PointEntry, archetypeName: string, group: GroupEntry): Model?
    if atGlobalCap() then
        warn("EnemySpawner: GLOBAL_MAX_ENEMIES reached; skipping spawn")
        return nil
    end
    local archetype = Config.Archetypes[archetypeName]
    if not archetype then
        warn(string.format("EnemySpawner: unknown archetype '%s'", archetypeName))
        return nil
    end
    local template = ServerStorage:FindFirstChild("EnemyAssets")
        and ServerStorage.EnemyAssets:FindFirstChild(archetype.Model)
    if not template or not template:IsA("Model") then
        warn(string.format("EnemySpawner: missing EnemyAssets model '%s'", archetype.Model))
        return nil
    end

    local bodySize = bodySizeOf(template)
    local ignore = { group.model, getActiveFolder() }
    local origin = point.attachment.WorldPosition

    for _ = 1, Config.Verification.MaxRetries do
        local candidate = jitter(origin, point.radius)
        local result = SpawnVerification.Verify(candidate, bodySize, ignore, livingEnemies())
        if result.ok and result.position then
            local model = template:Clone()
            applyArchetype(model, archetype)
            model:PivotTo(CFrame.new(result.position))
            model.Parent = getActiveFolder()
            group.counts[point.attachment] = (group.counts[point.attachment] or 0) + 1
            Signals.OnEnemySpawned:Fire(result.position, archetypeName)
            return model
        end
    end
    warn(string.format("EnemySpawner: no valid spot for '%s' after %d tries; skipping", archetypeName, Config.Verification.MaxRetries))
    return nil
end

local function rollArchetype(composition: SpawnComposition): string?
    if typeof(composition) == "string" then
        return composition
    end
    local total = 0
    for _, w in pairs(composition) do
        total += w
    end
    if total <= 0 then
        return nil
    end
    local ticket = rng:NextNumber() * total
    local acc = 0
    for name, w in pairs(composition) do
        acc += w
        if ticket <= acc then
            return name
        end
    end
    return nil
end

-- [[ PUBLIC API ]]

function EnemySpawner.Init()
    getActiveFolder()
    -- Invalidate a cached group if its model is untagged/removed at runtime.
    CollectionService:GetInstanceRemovedSignal(Config.GroupTag):Connect(function(model)
        for name, entry in pairs(groups) do
            if entry.model == model then
                groups[name] = nil
            end
        end
    end)
end

function EnemySpawner.ResetGroup(groupName: string)
    local entry = groups[groupName]
    if entry then
        table.clear(entry.counts)
    end
end

-- Spawn one archetype at a specific named point (verified). Returns Model or nil.
function EnemySpawner.SpawnAt(groupName: string, pointName: string, archetypeName: string): Model?
    local group = resolveGroup(groupName)
    if not group then
        return nil
    end
    for _, p in ipairs(group.points) do
        if p.attachment.Name == pointName then
            return placeAt(p, archetypeName, group)
        end
    end
    warn(string.format("EnemySpawner: group '%s' has no point named '%s'", groupName, pointName))
    return nil
end

function EnemySpawner.SpawnWave(groupName: string, request: SpawnRequest): EnemyWave.WaveHandle
    local wave = EnemyWave.new()
    local group = resolveGroup(groupName)
    if not group or #group.points == 0 then
        warn(string.format("EnemySpawner: group '%s' unresolved or has no points", groupName))
        wave:MarkComplete()
        return wave
    end

    if request.seed then
        rng = Random.new(request.seed)
    end

    local interval = request.interval or 0

    local function spawnOne()
        local point = selectPoint(group)
        if not point then
            return
        end
        local archetypeName = rollArchetype(request.composition)
        if not archetypeName then
            return
        end
        local model = placeAt(point, archetypeName, group)
        if model then
            wave:Register(model)
        end
    end

    if interval <= 0 then
        for _ = 1, request.count do
            spawnOne()
        end
        wave:MarkComplete()
    else
        task.spawn(function()
            for i = 1, request.count do
                spawnOne()
                if i < request.count then
                    task.wait(interval)
                end
            end
            wave:MarkComplete()
        end)
    end

    return wave
end

return EnemySpawner
