Replicated Server Projectiles + C4/Grenade Redesign — Implementation Plan

For agentic workers: REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (- [ ]) syntax for tracking.

Goal: Replace the per-client cosmetic projectile model with one server-spawned, replicated instance per projectile (rocket, grenade, C4) that flies → rests → detonates, and merge C4 into the Throwable category as a thrown, manually-detonated explosive.

Architecture: A new server ServerProjectile module owns each replicated instance under workspace/ServerProjectiles. Throwables use a physics mover (unanchored, massless, server network ownership → free bounce/settle, smooth interpolation); the rocket uses the existing authoritative FastCast2 cast stepping an anchored part. Damage/AOE stays server-authoritative. The old per-client ProjectileVisuals path, the C4 raycast-placement path, and three cosmetic packets are deleted.

Tech Stack: Roblox Luau, Rojo, FastCast2 (vendored), Janitor (Wally howmanysmall/janitor), Packet (vendored networking), MotionMaster.

Spec: docs/superpowers/specs/2026-06-07-replicated-projectile-redesign-design.md

--!strict

--[=[
    @class ServerProjectile
    @server
    Spawns and owns server-authoritative, REPLICATED projectile instances cloned
    from ReplicatedStorage/WeaponAssets into workspace/ServerProjectiles. Two
    movers: a physics mover for throwables (unanchored, massless, server network
    ownership — free bounce/settle and smooth interpolation to every client) and a
    rocket spawner returning an anchored part the caller steps along its
    authoritative FastCast2 cast. Damage/explosions are handled by the caller
    (WeaponServer), never here.
]=]

local ReplicatedStorage = game:GetService("ReplicatedStorage")

local Janitor = require(ReplicatedStorage.Packages:WaitForChild("Janitor"))

-- Studs to spawn a thrown projectile ahead of the muzzle so it never self-collides
-- with the thrower at launch (physics parts can't ignore an instance the way a
-- raycast filter can).
local SPAWN_OFFSET = 2.5

-- Resolved lazily (NOT a blocking top-level WaitForChild): the WeaponAssets folder
-- is Studio-authored and may not exist yet. Until it does, spawns no-op safely.
local assetsFolder: Instance? = ReplicatedStorage:FindFirstChild("WeaponAssets")

-- All replicated projectiles live here so they stay out of unrelated systems' way.
local container = Instance.new("Folder")
container.Name = "ServerProjectiles"
container.Parent = workspace

local ServerProjectile = {}
ServerProjectile.Container = container

local function getTemplate(name: string): Instance?
	if not assetsFolder then
		assetsFolder = ReplicatedStorage:FindFirstChild("WeaponAssets")
	end
	if not assetsFolder then
		return nil
	end
	return assetsFolder:FindFirstChild(name)
end

-- Root assembly part: the part itself, or a Model's PrimaryPart.
local function rootOf(inst: Instance): BasePart?
	if inst:IsA("BasePart") then
		return inst
	end
	if inst:IsA("Model") then
		return inst.PrimaryPart
	end
	return nil
end

local function setMassless(inst: Instance)
	if inst:IsA("BasePart") then
		inst.Massless = true
	end
	for _, d in inst:GetDescendants() do
		if d:IsA("BasePart") then
			d.Massless = true
		end
	end
end

local function placeAt(inst: Instance, cframe: CFrame)
	if inst:IsA("BasePart") then
		inst.CFrame = cframe
	elseif inst:IsA("Model") then
		inst:PivotTo(cframe)
	end
end

--[=[
    Spawn a physics-driven, replicated throwable. Returns the cloned instance
    (BasePart or Model) or nil if the template is missing/invalid.

    @param rest "Bounce" lets it bounce + settle (grenade); "Stick" freezes it dead
                on first contact that isn't the thrower (C4).
    @param ignore instances whose descendants don't count as a "Stick" surface
                  (e.g. the thrower's character).
]=]
function ServerProjectile.spawnThrowable(
	templateName: string,
	origin: Vector3,
	direction: Vector3,
	speed: number,
	rest: string,
	ignore: { Instance }
): Instance?
	local template = getTemplate(templateName)
	if not template then
		warn("[ServerProjectile] Missing template:", templateName)
		return nil
	end

	local clone = template:Clone()
	local root = rootOf(clone)
	if not root then
		warn("[ServerProjectile] Template has no root part (set PrimaryPart on Models):", templateName)
		clone:Destroy()
		return nil
	end

	setMassless(clone)
	root.Anchored = false
	root.CanCollide = true
	placeAt(clone, CFrame.lookAlong(origin + direction * SPAWN_OFFSET, direction))
	clone.Parent = container

	-- Keep the SERVER as network owner: authoritative position reads + an exploiter
	-- cannot seize ownership to teleport the charge.
	pcall(function()
		root:SetNetworkOwner(nil)
	end)
	root.AssemblyLinearVelocity = direction * speed

	if rest == "Stick" then
		local janitor = Janitor.new()
		janitor:Add(root.Touched:Connect(function(hit: BasePart)
			for _, ig in ignore do
				if ig and hit:IsDescendantOf(ig) then
					return -- ignore the thrower's own body
				end
			end
			if not root.Anchored then
				root.AssemblyLinearVelocity = Vector3.zero
				root.Anchored = true -- rest immediately, no bounce; sticks where it hit
			end
			janitor:Cleanup()
		end), "Disconnect")
		-- Self-clean if the part is destroyed (detonated / player left) before contact.
		root.Destroying:Once(function()
			janitor:Destroy()
		end)
	end

	return clone
end

--[=[
    Spawn an anchored, replicated rocket cosmetic at `cframe`. The caller (WeaponServer)
    steps its CFrame along the authoritative FastCast cast and destroys it on hit.
    CanQuery=false so the authoritative ray never hits the cosmetic itself.
]=]
function ServerProjectile.spawnRocket(templateName: string, cframe: CFrame): BasePart?
	local template = getTemplate(templateName)
	if not (template and template:IsA("BasePart")) then
		warn("[ServerProjectile] Rocket template missing or not a BasePart:", templateName)
		return nil
	end
	local clone = template:Clone() :: BasePart
	clone.Anchored = true
	clone.CanCollide = false
	clone.CanQuery = false
	clone.CanTouch = false
	clone.CFrame = cframe
	clone.Parent = container
	return clone
end

return ServerProjectile

local THROW_MAX_ORIGIN_DISTANCE = 14

local function onThrowProjectile(player: Player, origin: Vector3, direction: Vector3)
	local _tool, config = getEquippedWeapon(player)
	if not config or config.Category ~= "Throwable" then
		return
	end
	local throwable = config :: WeaponTypes.ThrowableWeaponConfig

	local now = os.clock()
	if not WeaponValidation.canFire(WeaponState.GetLastFire(player, throwable.ToolName), throwable.CooldownTime, now) then
		return
	end
	local character = player.Character
	local root = character and character:FindFirstChild("HumanoidRootPart") :: BasePart?
	if not character or not root then
		return
	end
	-- Anti-exploit: finite origin near the character (NaN-safe).
	local originDist = (origin - root.Position).Magnitude
	if originDist ~= originDist or originDist > THROW_MAX_ORIGIN_DISTANCE then
		return
	end
	-- Reject non-finite / zero-length throw direction before normalizing.
	local dirMag = direction.Magnitude
	if dirMag ~= dirMag or dirMag < 1e-4 then
		return
	end
	local unit = direction / dirMag

	-- Manual (C4): enforce MaxActive before spawning. Yield-free until AddActive below.
	if throwable.Detonation == "Manual" then
		local maxActive = throwable.MaxActive or 1
		if WeaponState.GetActiveCount(player) >= maxActive then
			return
		end
	end

	WeaponState.SetLastFire(player, throwable.ToolName, now)

	local part = ServerProjectile.spawnThrowable(
		throwable.CosmeticTemplate,
		origin,
		unit,
		throwable.ThrowSpeed,
		throwable.Rest,
		{ character }
	)
	if not part then
		return
	end

	if throwable.Detonation == "Fuse" then
		local fuse = throwable.FuseTime or 3
		task.delay(fuse, function()
			if part.Parent == nil then
				return
			end
			local center = part:GetPivot().Position
			CombatService.ApplyAOE(center, throwable.AOERadius, throwable.AOEDamage, "AOE", player, {})
			Packets.ExplosionAt:Fire(center, throwable.AOERadius, throwable.ToolName)
			part:Destroy()
		end)
	else
		WeaponState.AddActive(player, part)
	end
end

local function onDetonate(player: Player)
	local actives = WeaponState.TakeAllActive(player)
	local c4 = WeaponConfig.C4 :: WeaponTypes.ThrowableWeaponConfig
	for _, charge in actives do
		local part = charge.model
		if part and part.Parent then
			local center = part:GetPivot().Position
			CombatService.ApplyAOE(center, c4.AOERadius, c4.AOEDamage, "AOE", player, {})
			Packets.ExplosionAt:Fire(center, c4.AOERadius, c4.ToolName)
			part:Destroy()
		end
	end
end

local function getThrowOriginDir(character: Model, tool: Tool): (Vector3, Vector3)
	local handle = tool:FindFirstChild("Handle") :: BasePart?
	local origin = handle and handle.Position or character:GetPivot().Position
	local camera = workspace.CurrentCamera
	if not camera then
		-- No camera yet: throw straight along the character's facing.
		return origin, character:GetPivot().LookVector
	end
	local viewport = camera.ViewportSize
	local ray = camera:ViewportPointToRay(viewport.X / 2, viewport.Y / 2)
	-- Bias the throw slightly upward for a natural lob.
	local direction = (ray.Direction + Vector3.new(0, 0.25, 0)).Unit
	return origin, direction
end

function ThrowableHandler.OnActivate(context: HandlerContext)
	local config = context.config :: WeaponTypes.ThrowableWeaponConfig
	local now = os.clock()
	if not WeaponValidation.canFire(lastThrow[config.ToolName] or -math.huge, config.CooldownTime, now) then
		return
	end
	lastThrow[config.ToolName] = now

	local origin, direction = getThrowOriginDir(context.character, context.tool)
	SoundManager.PlaySFXAt(config.ThrowSound, origin, { Replicate = true })
	Packets.ReqThrowProjectile:Fire(origin, direction)
end

-- Manual-detonation throwables (C4): bound to G by WeaponController.
function ThrowableHandler.OnDetonate(_context: HandlerContext)
	Packets.ReqDetonate:Fire()
end

function ThrowableHandler.OnUnequip(_context: HandlerContext) end

return ThrowableHandler
