Weapon Ammo & Consumable Throwables Implementation Plan

For agentic workers: REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (- [ ]) syntax for tracking.

Goal: Give guns an inventory-backed numeric ammo pool (reload draws from it), make C4/grenades consumed from the backpack on use, and expose a clean client read API for the HUD.

Architecture: A new server AmmoManager (in-memory, session-only, mirrors CoinManager) owns per-player ammo counts keyed by ammo type (AmmoConfig). Guns keep their per-gun magazine in WeaponState; the pool replaces per-gun ReserveMax and is consumed on reload. Throwables stay physical Tools, destroyed server-side on throw, with C4 detonation decoupled from holding the Tool. Ammo + throwables are crafted at the existing workbench. A client AmmoApi caches mag/pool from sync packets and derives throwable counts from the Backpack, firing Signals for the HUD.

Tech Stack: Luau, Rojo, Packets (typed networking), Signals (event bus), Janitor (cleanup). No automated test suite — verification is manual in Roblox Studio with live Rojo sync (do not run rojo build or selene).

Conventions for this plan​

  • No test runner exists. Each task's verification is a Studio check (play-test with live sync) and/or a require-without-error check, not an automated test.
  • Commits: Conventional Commits. Stage only the files the task names.
  • The refactor is staged so the game keeps loading after every task. The old WeaponAmmoSync packet and WeaponAmmoChanged signal are kept until Task 13 (cleanup), after all consumers have switched.
  • Mid-refactor window: Guns are not fully wired end-to-end until Task 9; the first full gun Studio verification is in Task 9. Throwables verify in Task 11; crafting in Task 12.

Task 3: AmmoManager (server authority)​

Files:

  • Create: src/ServerStorage/SharedModules/AmmoManager.luau

  • Step 1: Create the module

--!strict

--[=[
	@class AmmoManager
	@server
	Authoritative, in-memory (session-only) per-player ammo pool keyed by ammo
	type (see Configs/AmmoConfig). Guns draw from it on reload; crafting adds to
	it. Mirrors the CoinManager shape but WITHOUT persistence. Every mutation
	fires AmmoPoolSync to the owning client. Call Initialize() once from a server
	bootstrap; call EnsureStarting(player) on each spawn to (re)seed the pool.
]=]

local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")

local SharedModules = ReplicatedStorage.SharedModules
local AmmoConfig = require(SharedModules.Configs:WaitForChild("AmmoConfig"))
local Packets = require(SharedModules:WaitForChild("Packets"))

local pools: { [Player]: { [string]: number } } = {}

local AmmoManager = {}
local initialized = false

local function ensure(player: Player): { [string]: number }
	local p = pools[player]
	if not p then
		p = {}
		pools[player] = p
	end
	return p
end

--[=[ @within AmmoManager Current pool for an ammo type, 0 if none. ]=]
function AmmoManager.GetAmmo(player: Player, ammoType: string): number
	local p = pools[player]
	return (p and p[ammoType]) or 0
end

-- Push the current count for one ammo type to the owning client.
local function replicate(player: Player, ammoType: string)
	Packets.AmmoPoolSync:FireClient(player, ammoType, AmmoManager.GetAmmo(player, ammoType))
end

--[=[ @within AmmoManager Re-send the current pool for an ammo type (e.g. on equip). ]=]
function AmmoManager.Sync(player: Player, ammoType: string)
	replicate(player, ammoType)
end

--[=[ @within AmmoManager Add ammo; no-op on amount<=0 or unknown type; clamps to Capacity. ]=]
function AmmoManager.AddAmmo(player: Player, ammoType: string, amount: number)
	if amount <= 0 then
		return
	end
	local def = AmmoConfig.AmmoTypes[ammoType]
	if not def then
		return
	end
	local p = ensure(player)
	p[ammoType] = math.min((p[ammoType] or 0) + amount, def.Capacity)
	replicate(player, ammoType)
end

--[=[ @within AmmoManager Atomic spend; false with no mutation if amount<=0 or insufficient. ]=]
function AmmoManager.ConsumeAmmo(player: Player, ammoType: string, amount: number): boolean
	if amount <= 0 then
		return false
	end
	local p = pools[player]
	local have = (p and p[ammoType]) or 0
	if have < amount then
		return false
	end
	p[ammoType] = have - amount
	replicate(player, ammoType)
	return true
end

--[=[ @within AmmoManager (Re)seed every ammo type to its Starting amount. Per-life. ]=]
function AmmoManager.EnsureStarting(player: Player)
	local p = ensure(player)
	for ammoType, def in AmmoConfig.AmmoTypes do
		p[ammoType] = def.Starting
		replicate(player, ammoType)
	end
end

function AmmoManager.Clear(player: Player)
	pools[player] = nil
end

--[=[ @within AmmoManager Wire per-player cleanup. Call exactly once per server. ]=]
function AmmoManager.Initialize()
	if initialized then
		return
	end
	initialized = true
	Players.PlayerRemoving:Connect(function(player)
		AmmoManager.Clear(player)
	end)
end

return AmmoManager

  • Step 1: Create AmmoApi.luau
--!strict

--[=[
	@class AmmoApi
	@client
	Single read surface for the HUD over weapon ammo + throwable counts. Caches
	the per-gun magazine (from WeaponMagSync + client prediction) and the per-type
	ammo pool (from AmmoPoolSync + prediction), and derives throwable counts live
	from the Backpack/Character. Fires Signals.WeaponMagChanged / AmmoPoolChanged /
	ThrowableCountChanged so the HUD reacts without polling. Call Init() once from
	a client bootstrap. Static maxima come from config, not the network.
]=]

local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")

local SharedModules = ReplicatedStorage.SharedModules
local Packets = require(SharedModules:WaitForChild("Packets"))
local Signals = require(SharedModules:WaitForChild("Signals"))
local AmmoConfig = require(SharedModules.Configs:WaitForChild("AmmoConfig"))
local WeaponConfig = require(SharedModules.Configs:WaitForChild("WeaponConfig"))
local WeaponTypes = require(SharedModules:WaitForChild("WeaponTypes"))
local Janitor = require(ReplicatedStorage.Packages:WaitForChild("Janitor"))

local player = Players.LocalPlayer

local magCache: { [string]: number } = {} -- weaponName -> mag
local poolCache: { [string]: number } = {} -- ammoType -> amount

local AmmoApi = {}
local initialized = false

--[=[ @within AmmoApi (mag, magSize) for a gun; (0,0) if not a known gun. ]=]
function AmmoApi.GetMag(weaponName: string): (number, number)
	local cfg = WeaponConfig[weaponName]
	local magSize = 0
	if cfg and cfg.Category == "Gun" then
		magSize = (cfg :: WeaponTypes.GunWeaponConfig).MagazineSize
	end
	return magCache[weaponName] or 0, magSize
end

--[=[ @within AmmoApi (amount, capacity) for an ammo type. ]=]
function AmmoApi.GetPool(ammoType: string): (number, number)
	local def = AmmoConfig.AmmoTypes[ammoType]
	return poolCache[ammoType] or 0, (def and def.Capacity) or 0
end

-- Count a throwable Tool across Backpack + equipped slot.
local function countTool(toolName: string): number
	local count = 0
	local bp = player:FindFirstChildOfClass("Backpack")
	if bp then
		for _, item in bp:GetChildren() do
			if item:IsA("Tool") and item.Name == toolName then
				count += 1
			end
		end
	end
	local char = player.Character
	if char then
		local equipped = char:FindFirstChildWhichIsA("Tool")
		if equipped and equipped.Name == toolName then
			count += 1
		end
	end
	return count
end

--[=[ @within AmmoApi Number of a throwable Tool currently carried. ]=]
function AmmoApi.GetThrowableCount(toolName: string): number
	return countTool(toolName)
end

--[=[ @within AmmoApi (mag, magSize, poolAmount, poolCapacity) for the equipped gun, or nil. ]=]
function AmmoApi.GetEquippedAmmo(): (number, number, number, number)?
	local char = player.Character
	local tool = char and char:FindFirstChildWhichIsA("Tool")
	if not tool then
		return nil
	end
	local cfg = WeaponConfig[tool.Name]
	if not cfg or cfg.Category ~= "Gun" then
		return nil
	end
	local mag, magSize = AmmoApi.GetMag(tool.Name)
	local ammoType = AmmoConfig.WeaponAmmo[tool.Name]
	local pool, cap = 0, 0
	if ammoType then
		pool, cap = AmmoApi.GetPool(ammoType)
	end
	return mag, magSize, pool, cap
end

--[=[ @within AmmoApi Client prediction setter for a gun's magazine. ]=]
function AmmoApi.SetMag(weaponName: string, mag: number)
	magCache[weaponName] = mag
	Signals.WeaponMagChanged:Fire(weaponName, mag)
end

--[=[ @within AmmoApi Client prediction setter for an ammo pool. ]=]
function AmmoApi.SetPoolLocal(ammoType: string, amount: number)
	poolCache[ammoType] = amount
	Signals.AmmoPoolChanged:Fire(ammoType, amount)
end

local function isThrowable(toolName: string): boolean
	local cfg = WeaponConfig[toolName]
	return cfg ~= nil and cfg.Category == "Throwable"
end

--[=[ @within AmmoApi Connect sync packets + backpack watchers. Call once. ]=]
function AmmoApi.Init()
	if initialized then
		return
	end
	initialized = true

	Packets.WeaponMagSync.OnClientEvent:Connect(function(weaponName: string, mag: number)
		magCache[weaponName] = mag
		Signals.WeaponMagChanged:Fire(weaponName, mag)
	end)
	Packets.AmmoPoolSync.OnClientEvent:Connect(function(ammoType: string, amount: number)
		poolCache[ammoType] = amount
		Signals.AmmoPoolChanged:Fire(ammoType, amount)
	end)

	-- Backpack is replaced per character: rebind watchers each spawn via a Janitor.
	local backpackJanitor = Janitor.new()
	local function watch(container: Instance)
		local function onChange(child: Instance)
			if child:IsA("Tool") and isThrowable(child.Name) then
				Signals.ThrowableCountChanged:Fire(child.Name, countTool(child.Name))
			end
		end
		backpackJanitor:Add(container.ChildAdded:Connect(onChange), "Disconnect")
		backpackJanitor:Add(container.ChildRemoved:Connect(onChange), "Disconnect")
	end
	local function rebind(character: Model)
		backpackJanitor:Cleanup()
		local bp = player:FindFirstChildOfClass("Backpack") or player:WaitForChild("Backpack")
		watch(bp)
		watch(character)
	end

	player.CharacterAdded:Connect(rebind)
	if player.Character then
		rebind(player.Character)
	end
end

return AmmoApi

-- Pure ammo transition: reload from a shared pool. Moves rounds from the pool
-- into the magazine up to `magazineSize`. Returns (newMag, taken).
function WeaponValidation.ammoReloadFromPool(mag: number, poolAvailable: number, magazineSize: number): (number, number)
	local needed = math.max(0, magazineSize - mag)
	local taken = math.min(needed, poolAvailable)
	return mag + taken, taken
end

	local mag = WeaponState.GetMag(player, gun.ToolName)
	local pool = AmmoManager.GetAmmo(player, gun.AmmoType)
	if mag >= gun.MagazineSize or pool <= 0 then
		return false, 0
	end

	WeaponState.SetReloadEnd(player, gun.ToolName, now + gun.ReloadTime)

	task.delay(gun.ReloadTime, function()
		if player.Parent == nil then
			return
		end
		local curMag = WeaponState.GetMag(player, gun.ToolName)
		local curPool = AmmoManager.GetAmmo(player, gun.AmmoType)
		local newMag, taken = WeaponValidation.ammoReloadFromPool(curMag, curPool, gun.MagazineSize)
		if taken > 0 then
			AmmoManager.ConsumeAmmo(player, gun.AmmoType, taken) -- fires AmmoPoolSync
			WeaponState.SetMag(player, gun.ToolName, newMag)
			Packets.WeaponMagSync:FireClient(player, gun.ToolName, newMag)
		end
		WeaponState.SetReloadEnd(player, gun.ToolName, 0)
	end)

	return true, gun.ReloadTime
end
